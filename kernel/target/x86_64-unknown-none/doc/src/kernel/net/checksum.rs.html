1/// RFC 1071 인터넷 체크섬 (ones-complement sum의 ones-complement)
2///
3/// IP 헤더, ICMP, UDP 등 모든 인터넷 프로토콜 체크섬에 사용.
4/// 체크섬 필드를 0으로 채운 상태로 계산해서 넣으면 됨.
5pub fn internet_checksum(data: &[u8]) -> u16 {
6    let mut sum: u32 = 0;
7    let mut i = 0;
8    while i + 1 < data.len() {
9        sum += ((data[i] as u32) << 8) | (data[i + 1] as u32);
10        i += 2;
11    }
12    if i < data.len() {
13        sum += (data[i] as u32) << 8; // 홀수 byte → 하위 byte = 0 padding
14    }
15    // carry fold: 최대 2번이면 수렴
16    while sum >> 16 != 0 {
17        sum = (sum & 0xFFFF) + (sum >> 16);
18    }
19    !(sum as u16)
20}
