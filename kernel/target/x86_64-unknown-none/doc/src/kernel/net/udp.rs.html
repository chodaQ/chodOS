5use alloc::vec::Vec;

7/// UDP 데이터그램(헤더+페이로드) 빌드.
8pub fn build(src_port: u16, dst_port: u16, payload: &[u8]) -> Vec<u8> {
9    let length = (8 + payload.len()) as u16;
10    let mut seg = Vec::with_capacity(8 + payload.len());
11    seg.push((src_port >> 8) as u8); seg.push(src_port as u8);
12    seg.push((dst_port >> 8) as u8); seg.push(dst_port as u8);
13    seg.push((length  >> 8) as u8);  seg.push(length as u8);
14    seg.push(0); seg.push(0);        // checksum = 0 (optional in IPv4)
15    seg.extend_from_slice(payload);
16    seg
17}

19/// UDP 데이터그램 파싱. (src_port, dst_port, payload) 반환.
20pub fn parse(data: &[u8]) -> Option<(u16, u16, &[u8])> {
21    if data.len() < 8 { return None; }
22    let src  = ((data[0] as u16) << 8) | data[1] as u16;
23    let dst  = ((data[2] as u16) << 8) | data[3] as u16;
24    let len  = ((data[4] as u16) << 8) | data[5] as u16;
25    let plen = (len as usize).checked_sub(8)?;
26    if data.len() < 8 + plen { return None; }
27    Some((src, dst, &data[8..8 + plen]))
28}
