1/// ICMP Echo Request / Reply (IPv4)
2///
3/// ICMP over IPv4, Protocol = 1.
4/// Echo Request: type=8, code=0
5/// Echo Reply:   type=0, code=0
6
7use alloc::vec::Vec;
8use super::checksum::internet_checksum;
9
10pub const TYPE_ECHO_REPLY:   u8 = 0;
11pub const TYPE_ECHO_REQUEST: u8 = 8;
12
13/// ICMP Echo Request 메시지(헤더+페이로드) 빌드. 체크섬 자동 계산.
14pub fn build_echo_request(id: u16, seq: u16, payload: &[u8]) -> Vec<u8> {
15    let mut msg = Vec::with_capacity(8 + payload.len());
16    msg.push(TYPE_ECHO_REQUEST); // type
17    msg.push(0);                 // code
18    msg.push(0); msg.push(0);   // checksum placeholder
19    msg.push((id  >> 8) as u8); msg.push(id  as u8);
20    msg.push((seq >> 8) as u8); msg.push(seq as u8);
21    msg.extend_from_slice(payload);
22    let csum = internet_checksum(&msg);
23    msg[2] = (csum >> 8) as u8;
24    msg[3] = csum as u8;
25    msg
26}
27
28/// ICMP Echo Reply 파싱. type=0 이면 (id, seq) 반환.
29pub fn parse_echo_reply(data: &[u8]) -> Option<(u16, u16)> {
30    if data.len() < 8 { return None; }
31    if data[0] != TYPE_ECHO_REPLY { return None; }
32    let id  = ((data[4] as u16) << 8) | data[5] as u16;
33    let seq = ((data[6] as u16) << 8) | data[7] as u16;
34    Some((id, seq))
35}
