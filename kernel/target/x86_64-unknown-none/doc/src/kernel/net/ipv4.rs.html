11/// IPv4 헤더(20 bytes)를 빌드하고 체크섬을 채워 반환.
12///
13/// `total_len` = IP 헤더(20) + 페이로드 길이.
14pub fn build_header(
15    proto:       u8,
16    src:         [u8; 4],
17    dst:         [u8; 4],
18    total_len:   u16,
19    id:          u16,
20) -> [u8; 20] {
21    let mut h = [0u8; 20];
22    h[0] = 0x45;                          // version=4, IHL=5 (20 bytes)
23    h[1] = 0x00;                          // DSCP/ECN
24    h[2] = (total_len >> 8) as u8;
25    h[3] = total_len as u8;
26    h[4] = (id >> 8) as u8;
27    h[5] = id as u8;
28    h[6] = 0x40; h[7] = 0x00;            // flags=DF, frag_offset=0
29    h[8] = 64;                            // TTL
30    h[9] = proto;
31    // h[10..12] = checksum (0 for calculation)
32    h[12..16].copy_from_slice(&src);
33    h[16..20].copy_from_slice(&dst);
34    let csum = internet_checksum(&h);
35    h[10] = (csum >> 8) as u8;
36    h[11] = csum as u8;
37    h
38}

40/// Ethernet 페이로드(IP 헤더 + 데이터)에서 (proto, src_ip, dst_ip, ihl)을 파싱.
41///
42/// `ihl`은 IP 헤더 길이(bytes). 페이로드는 `data[ihl..]`.
43pub fn parse_header(data: &[u8]) -> Option<(u8, [u8; 4], [u8; 4], usize)> {
44    if data.len() < 20 { return None; }
45    if data[0] >> 4 != 4 { return None; } // IPv4만 처리
46    let ihl = ((data[0] & 0xF) as usize) * 4;
47    if data.len() < ihl { return None; }
48    let proto = data[9];
49    let mut src = [0u8; 4]; src.copy_from_slice(&data[12..16]);
50    let mut dst = [0u8; 4]; dst.copy_from_slice(&data[16..20]);
51    Some((proto, src, dst, ihl))
52}
