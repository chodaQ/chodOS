1//! PS/2 마우스 드라이버 (i8042 컨트롤러, IRQ12)
2//!
3//! 패킷 형식 (Set 1, 3바이트):
4//!   byte0: [Yo|Xo|Ys|Xs|1|M|R|L]  (Ys/Xs=부호, Yo/Xo=오버플로, bit3=항상 1)
5//!   byte1: X 이동량 (부호: byte0[4])
6//!   byte2: Y 이동량 (부호: byte0[5], 위 = 양수)
7
8use core::sync::atomic::{AtomicBool, AtomicI32, AtomicU8, AtomicUsize, Ordering};
9
10const DATA_PORT: u16 = 0x60;
11const CMD_PORT:  u16 = 0x64;
12
13pub static MOUSE_X:   AtomicI32 = AtomicI32::new(0);
14pub static MOUSE_Y:   AtomicI32 = AtomicI32::new(0);
15pub static MOUSE_BTN: AtomicU8  = AtomicU8::new(0);
16
17// 클릭 엣지 감지용: 이전 버튼 상태
18static PREV_BTN: AtomicU8 = AtomicU8::new(0);
19
20// 3바이트 패킷 누적
21static PKT:     [AtomicU8; 3] = [const { AtomicU8::new(0) }; 3];
22static PKT_IDX: AtomicUsize   = AtomicUsize::new(0);
23
24// 초기화 완료 플래그
25static READY: AtomicBool = AtomicBool::new(false);
26
27// ── 저수준 i8042 I/O ──────────────────────────────────────────────────────────
28
29unsafe fn wait_write() {
30    let mut i = 0u32;
31    loop {
32        let s: u8;
33        core::arch::asm!("in al, dx", out("al") s, in("dx") CMD_PORT, options(nomem, nostack));
34        if s & 0x02 == 0 { break; }
35        i += 1;
36        if i > 200_000 { break; }
37    }
38}
39
40unsafe fn wait_read() {
41    let mut i = 0u32;
42    loop {
43        let s: u8;
44        core::arch::asm!("in al, dx", out("al") s, in("dx") CMD_PORT, options(nomem, nostack));
45        if s & 0x01 != 0 { break; }
46        i += 1;
47        if i > 200_000 { break; }
48    }
49}
50
51unsafe fn cmd(c: u8) {
52    wait_write();
53    core::arch::asm!("out dx, al", in("dx") CMD_PORT, in("al") c, options(nomem, nostack));
54}
55
56unsafe fn write_data(d: u8) {
57    wait_write();
58    core::arch::asm!("out dx, al", in("dx") DATA_PORT, in("al") d, options(nomem, nostack));
59}
60
61unsafe fn read_data() -> u8 {
62    wait_read();
63    let v: u8;
64    core::arch::asm!("in al, dx", out("al") v, in("dx") DATA_PORT, options(nomem, nostack));
65    v
66}

68// 마우스로 바이트 전송 (0xD4 명령 → 0x60에 데이터)
69unsafe fn mouse_send(byte: u8) -> u8 {
70    cmd(0xD4);
71    write_data(byte);
72    read_data() // ACK
73}

75// ── 초기화 ────────────────────────────────────────────────────────────────────
76
77pub fn init() {
78    unsafe {
79        // 1. 기존 출력 버퍼 플러시 (오래된 데이터 제거)
80        for _ in 0..16 {
81            let s: u8;
82            core::arch::asm!("in al, dx", out("al") s, in("dx") CMD_PORT, options(nomem, nostack));
83            if s & 0x01 == 0 { break; }
84            let _: u8;
85            core::arch::asm!("in al, dx", out("al") _, in("dx") DATA_PORT, options(nomem, nostack));
86        }
87
88        // 2. 마우스 포트 활성화 (0xA8)
89        cmd(0xA8);
90
91        // 3. CCB 읽기 → IRQ12(bit1=1) + 마우스 클럭(bit5=0) 설정
92        cmd(0x20);
93        let ccb = read_data();
94        let ccb_new = (ccb | 0x02) & !0x20;
95        cmd(0x60);
96        write_data(ccb_new);
97
98        // 4. 마우스 리셋 (0xFF) — ACK 유무와 상관없이 응답 버퍼 비우기
99        cmd(0xD4); write_data(0xFF);
100        // 짧은 지연 후 버퍼 flush (ACK + BAT + ID 최대 3바이트)
101        for _ in 0..3 {
102            let s: u8;
103            core::arch::asm!("in al, dx", out("al") s, in("dx") CMD_PORT, options(nomem, nostack));
104            if s & 0x01 == 0 { break; }
105            let _: u8;
106            core::arch::asm!("in al, dx", out("al") _, in("dx") DATA_PORT, options(nomem, nostack));
107            // 짧은 대기 (~100µs 수준)
108            for _ in 0u32..10_000 {}
109        }
110
111        // 5. 기본 설정 (0xF6) — ACK 무시
112        cmd(0xD4); write_data(0xF6);
113        for _ in 0u32..50_000 {}
114        // 버퍼 flush
115        let s: u8;
116        core::arch::asm!("in al, dx", out("al") s, in("dx") CMD_PORT, options(nomem, nostack));
117        if s & 0x01 != 0 {
118            let _: u8;
119            core::arch::asm!("in al, dx", out("al") _, in("dx") DATA_PORT, options(nomem, nostack));
120        }
121
122        // 6. 스트리밍 활성화 (0xF4)
123        cmd(0xD4); write_data(0xF4);
124        for _ in 0u32..50_000 {}
125        // ACK 읽기 시도 (있으면 읽고, 없으면 무시)
126        let s: u8;
127        core::arch::asm!("in al, dx", out("al") s, in("dx") CMD_PORT, options(nomem, nostack));
128        if s & 0x01 != 0 {
129            let _: u8;
130            core::arch::asm!("in al, dx", out("al") _, in("dx") DATA_PORT, options(nomem, nostack));
131        }
132    }
133
134    // 커서를 화면 중앙으로 초기화
135    let w = crate::fb::width();
136    let h = crate::fb::height();
137    if w > 0 && h > 0 {
138        MOUSE_X.store(w as i32 / 2, Ordering::Relaxed);
139        MOUSE_Y.store(h as i32 / 2, Ordering::Relaxed);
140    }
141
142    READY.store(true, Ordering::Relaxed);
143    crate::serial_println!("[mouse] PS/2 mouse ready — cursor at ({}, {})",
144        MOUSE_X.load(Ordering::Relaxed), MOUSE_Y.load(Ordering::Relaxed));
145}

147// ── IRQ12 핸들러 ──────────────────────────────────────────────────────────────
148
149pub fn handle_irq() {
150    let byte: u8;
151    unsafe {
152        core::arch::asm!("in al, dx", out("al") byte, in("dx") DATA_PORT, options(nomem, nostack));
153    }
154
155    let idx = PKT_IDX.load(Ordering::Relaxed);
156
157    // 첫 번째 바이트 동기화: bit3이 반드시 1이어야 함
158    if idx == 0 && (byte & 0x08) == 0 { return; }
159
160    PKT[idx].store(byte, Ordering::Relaxed);
161    let next = idx + 1;
162
163    if next == 3 {
164        PKT_IDX.store(0, Ordering::Relaxed);
165        process_packet();
166    } else {
167        PKT_IDX.store(next, Ordering::Relaxed);
168    }
169}
170
171fn process_packet() {
172    if !READY.load(Ordering::Relaxed) { return; }
173
174    let b0 = PKT[0].load(Ordering::Relaxed);
175    let b1 = PKT[1].load(Ordering::Relaxed);
176    let b2 = PKT[2].load(Ordering::Relaxed);
177
178    // X/Y 오버플로 비트 — 패킷이 손상됐으므로 무시
179    if b0 & 0xC0 != 0 { return; }
180
181    let dx: i32 = if b0 & 0x10 != 0 { b1 as i32 - 256 } else { b1 as i32 };
182    let dy: i32 = if b0 & 0x20 != 0 { b2 as i32 - 256 } else { b2 as i32 };
183
184    let w = crate::fb::width() as i32;
185    let h = crate::fb::height() as i32;
186    if w == 0 || h == 0 { return; }
187
188    let x = (MOUSE_X.load(Ordering::Relaxed) + dx).max(0).min(w - 1);
189    let y = (MOUSE_Y.load(Ordering::Relaxed) - dy).max(0).min(h - 1); // Y 반전
190    MOUSE_X.store(x, Ordering::Relaxed);
191    MOUSE_Y.store(y, Ordering::Relaxed);
192
193    let btn = b0 & 0x07; // [M|R|L]
194    MOUSE_BTN.store(btn, Ordering::Relaxed);
195
196    let prev = PREV_BTN.load(Ordering::Relaxed);
197
198    // 왼쪽 버튼 rising edge → 클릭
199    if btn & 0x01 != 0 && prev & 0x01 == 0 {
200        on_click(x, y);
201    }
202    // 왼쪽 버튼 홀드 + 이동 → 드래그
203    if btn & 0x01 != 0 && (dx != 0 || dy != 0) {
204        crate::wm::on_drag(x, y);
205    }
206    // 왼쪽 버튼 falling edge → 드래그 종료
207    if prev & 0x01 != 0 && btn & 0x01 == 0 {
208        crate::wm::on_release();
209    }
210
211    PREV_BTN.store(btn, Ordering::Relaxed);
212
213    // 커서 다시 그리기 (드래그 중엔 on_drag 내부에서 이미 redraw하므로 중복되지만 무해)
214    crate::fb::draw_cursor(x, y);
215}

217fn on_click(x: i32, y: i32) {
218    crate::serial_println!("[mouse] click @ ({}, {})", x, y);
219    crate::wm::on_click(x as u32, y as u32);
