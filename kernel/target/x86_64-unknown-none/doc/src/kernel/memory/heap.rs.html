1//! 커널 힙 할당자 (Kernel Heap Allocator)
2//!
3//! ## 역할
4//! Rust의 `alloc` 크레이트(Box, Vec, String 등)가 사용하는
5//! 전역 힙 메모리를 관리함.
6//!
7//! ## #[global_allocator]
8//! Rust 런타임은 `Box::new()`, `Vec::push()` 등이 메모리를 요청할 때
9//! `#[global_allocator]`로 지정된 타입의 `GlobalAlloc` 구현을 호출함.
10//! 우리는 `linked_list_allocator::LockedHeap`을 사용함.
11//!
12//! ## linked_list_allocator 동작 방식
13//! - 힙을 하나의 큰 연속 메모리 블록으로 받음
14//! - 해제된 블록들을 연결 리스트로 관리
15//! - 할당 시: 리스트에서 크기가 맞는 블록을 찾아 분할
16//! - 해제 시: 블록을 리스트에 추가하고 인접한 블록과 병합(coalescing)
17//!
18//! ## 한계 (향후 개선 포인트)
19//! - 단편화: 많은 할당/해제 반복 시 작은 블록들이 흩어짐
20//! - Milestone 5 이후 Slab Allocator로 교체 예정
21
22use linked_list_allocator::LockedHeap;
23use super::frame;
24
25/// 전역 힙 할당자
26///
27/// `LockedHeap`: 스핀락으로 보호된 linked list 힙 할당자.
28/// `empty()`: 아직 초기화 전 상태. `init()` 호출 전 할당 시도 → panic.
29///
30/// `#[global_allocator]`: Rust에게 이 타입을 전역 메모리 할당자로 등록.
31/// 이 속성이 있으면 `alloc::alloc::alloc()`, `alloc::alloc::dealloc()` 등이
32/// 이 구현으로 라우팅됨.
33#[global_allocator]
34static ALLOCATOR: LockedHeap = LockedHeap::empty();
35
36/// 힙에 할당할 연속 프레임 수 (4MB = 1024 × 4KB)
37///
38/// 초기 힙 크기로 충분한 양.
39/// 추후 동적 힙 확장 기능을 추가할 수 있음 (힙이 가득 차면 프레임 추가).
40const HEAP_FRAMES: usize = 1024; // 4MB
41
42/// 커널 힙 초기화
43///
44/// 1. 프레임 할당자에서 연속된 `HEAP_FRAMES`개의 물리 프레임을 확보
45/// 2. HHDM 오프셋을 더해 가상 주소로 변환
46/// 3. `LockedHeap`에 해당 가상 주소 범위를 힙으로 등록
47///
48/// # Panics
49/// 연속 프레임이 부족하면 panic (부팅 실패).
50pub fn init(hhdm_offset: u64) {
51    let heap_size = HEAP_FRAMES * frame::PAGE_SIZE as usize;
52
53    // 연속된 물리 프레임 확보.
54    // linked_list_allocator는 연속된 메모리 범위가 필요함.
55    // alloc_contiguous()는 비트맵에서 연속 free 프레임을 탐색.
56    let heap_phys = frame::alloc_contiguous(HEAP_FRAMES)
57        .expect("[heap] failed: not enough contiguous physical frames");
58
59    // 물리 주소 → 가상 주소 변환
60    // HHDM(Higher Half Direct Map): limine이 설정한 직접 매핑
61    //   가상 주소 = 물리 주소 + HHDM 오프셋
62    // 이 범위는 limine이 이미 페이지 테이블에 매핑해 두었음
63    let heap_virt = (heap_phys + hhdm_offset) as usize;
64
65    crate::serial_println!(
66        "[mem] heap: {} MB at phys=0x{:x} virt=0x{:x}",
67        heap_size / (1024 * 1024),
68        heap_phys,
69        heap_virt
70    );
71
72    // 힙 할당자에 메모리 범위 등록
73    // Safety: 이 가상 주소 범위가 유효하고, 다른 용도로 사용되지 않음을 보장해야 함.
74    // - phys_start는 frame allocator가 used로 마킹했으므로 이중 할당 없음
75    // - HHDM 매핑은 limine이 보장
76    // - 이 함수는 커널 초기화 중 단 한 번만 호출됨
77    unsafe {
78        ALLOCATOR.lock().init(heap_virt as *mut u8, heap_size);
79    }
80}
