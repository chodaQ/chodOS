1//! 비트맵 물리 프레임 할당자 (Bitmap Physical Frame Allocator)
2//!
3//! ## 설계 원칙
4//!
5//! 물리 메모리를 4KB 페이지(프레임) 단위로 관리.
6//! 비트맵의 각 비트가 프레임 하나를 나타냄:
7//!   1 = free  (사용 가능)
8//!   0 = used  (사용 중 또는 접근 불가)
9//!
10//! ## 왜 비트맵인가?
11//! - 구현 단순, 이해하기 쉬움
12//! - 최대 8GB 메모리를 256KB 비트맵으로 관리 (8GB / 4KB / 8 bits = 256KB)
13//! - 할당: O(n/64) — 64비트 단위로 스캔하므로 n/64배 빠름
14//! - 해제: O(1)
15//!
16//! ## 한계 (향후 개선 포인트)
17//! - Buddy System 대비 내부 단편화 없지만, 연속 할당 탐색이 느림
18//! - 멀티코어 환경에서 락 없이 unsafe하게 접근 중 (단일코어 가정)
19//! - Milestone 3에서 Buddy Allocator로 교체 예정
20
21use core::sync::atomic::{AtomicU64, Ordering};
22use limine::memmap;
23
24// ==================== 상수 ====================
25
26/// 페이지(프레임) 크기: 4KB
27pub const PAGE_SIZE: u64 = 4096;
28
29/// 비트맵으로 관리 가능한 최대 프레임 수
30/// 32768(워드 수) × 64(비트/워드) = 2,097,152 프레임 = 8GiB
31const BITMAP_WORDS: usize = 32768;
32const MAX_FRAMES: usize = BITMAP_WORDS * 64;
33
34// ==================== 전역 상태 ====================
35
36/// 프레임 비트맵 (1 = free, 0 = used)
37///
38/// `static mut`: 커널 초기화 단계에서 단일 코어가 초기화 전용으로 접근.
39/// 멀티코어 지원 시 뮤텍스로 교체 필요.
40///
41/// `.bss` 섹션에 배치 → limine이 0으로 채워줌 → 초기 상태: 전부 used
42static mut BITMAP: [u64; BITMAP_WORDS] = [0u64; BITMAP_WORDS];
43
44/// 현재 free 프레임 수 (통계용)
45static FREE_FRAMES: AtomicU64 = AtomicU64::new(0);
46
47/// 전체 usable 프레임 수 (통계용)
48static TOTAL_USABLE: AtomicU64 = AtomicU64::new(0);
49
50// ==================== 비트맵 헬퍼 ====================
51
52/// 특정 프레임을 free로 마킹
53#[inline]
54unsafe fn set_free(frame: usize) {
55    BITMAP[frame / 64] |= 1u64 << (frame % 64);
56}
57
58/// 특정 프레임을 used로 마킹
59#[inline]
60unsafe fn set_used(frame: usize) {
61    BITMAP[frame / 64] &= !(1u64 << (frame % 64));
62}
63
64/// 특정 프레임이 free인지 확인
65#[inline]
66unsafe fn is_free(frame: usize) -> bool {
67    BITMAP[frame / 64] & (1u64 << (frame % 64)) != 0
68}
69
70// ==================== 초기화 ====================
71
72/// 프레임 할당자 초기화
73///
74/// limine 메모리 맵을 스캔해서 `MEMMAP_USABLE` 영역의 프레임만 free로 마킹.
75/// 나머지(커널, 펌웨어, ACPI, limine 등)는 이미 0(used)이므로 건드리지 않음.
76///
77/// limine이 메모리 맵을 이미 정리해주기 때문에:
78/// - `MEMMAP_EXECUTABLE_AND_MODULES` = 커널 + 모듈 (건드리지 않음)
79/// - `MEMMAP_BOOTLOADER_RECLAIMABLE` = 나중에 회수 가능 (지금은 건드리지 않음)
80/// - `MEMMAP_USABLE` = 우리가 자유롭게 쓸 수 있는 RAM
81pub fn init(memmap_entries: &[&memmap::Entry]) {
82    let mut total: u64 = 0;
83    let mut free: u64 = 0;
84
85    for entry in memmap_entries {
86        if entry.type_ != memmap::MEMMAP_USABLE {
87            continue;
88        }
89
90        // 이 usable 영역의 시작/끝 프레임 인덱스
91        // 끝 주소는 반드시 PAGE_SIZE 배수여야 함 (limine이 보장)
92        let start_frame = (entry.base / PAGE_SIZE) as usize;
93        let end_frame = ((entry.base + entry.length) / PAGE_SIZE) as usize;
94
95        for frame in start_frame..end_frame.min(MAX_FRAMES) {
96            unsafe { set_free(frame) };
97            free += 1;
98        }
99        total += entry.length / PAGE_SIZE;
100    }
101
102    FREE_FRAMES.store(free, Ordering::Relaxed);
103    TOTAL_USABLE.store(total, Ordering::Relaxed);
104
105    crate::serial_println!(
106        "[mem] frame allocator: {} MB usable ({} frames free)",
107        free * PAGE_SIZE / (1024 * 1024),
108        free
109    );
110}

112// ==================== 할당 / 해제 ====================
113
114/// 물리 프레임 1개 할당
115///
116/// 비트맵을 64비트 단위로 스캔해서 첫 번째 free 프레임을 찾아 반환.
117/// 반환값: 해당 프레임의 물리 주소 (4KB 정렬)
118#[allow(dead_code)] // alloc_contiguous() 위주로 쓰지만 단일 프레임 할당용으로 공개
119pub fn alloc_frame() -> Option<u64> {
120    // BITMAP.iter_mut() 대신 raw 포인터를 사용.
121    // static mut에 대한 가변 참조(&mut)는 Rust 2024부터 경고가 됨.
122    // raw 포인터(*mut u64)는 참조 규칙 밖에 있어 경고 없이 접근 가능.
123    // Safety: 단일 코어, 인터럽트 없는 초기화 단계에서만 호출됨.
124    unsafe {
125        let bitmap = core::ptr::addr_of_mut!(BITMAP);
126        for word_idx in 0..BITMAP_WORDS {
127            let word_ptr = (*bitmap).as_mut_ptr().add(word_idx);
128            let word_val = word_ptr.read();
129            if word_val == 0 {
130                continue; // 이 64개 프레임은 전부 used
131            }
132            // trailing_zeros(): 최하위 set 비트 위치 = 가장 낮은 주소의 free 프레임
133            let bit = word_val.trailing_zeros() as usize;
134            word_ptr.write(word_val & !(1u64 << bit)); // used 마킹
135            FREE_FRAMES.fetch_sub(1, Ordering::Relaxed);
136
137            let frame_idx = word_idx * 64 + bit;
138            return Some(frame_idx as u64 * PAGE_SIZE);
139        }
140    }
141    None // 물리 메모리 고갈
142}

144/// 연속된 `count`개의 물리 프레임을 한 번에 할당
145///
146/// 연결 리스트 힙 할당자는 연속된 물리 메모리가 필요하므로 이 함수를 사용.
147/// 반환값: 첫 번째 프레임의 물리 주소 (해당 주소부터 count * 4KB가 연속)
148///
149/// 성능: O(MAX_FRAMES) — 최악의 경우 전체 비트맵 순회
150pub fn alloc_contiguous(count: usize) -> Option<u64> {
151    if count == 0 {
152        return None;
153    }
154
155    unsafe {
156        let mut run_start = 0usize; // 현재 연속 구간 시작
157        let mut run_len = 0usize;   // 현재 연속 구간 길이
158
159        for frame in 0..MAX_FRAMES {
160            if is_free(frame) {
161                if run_len == 0 {
162                    run_start = frame;
163                }
164                run_len += 1;
165
166                if run_len == count {
167                    // count개 연속 free 프레임 발견 → 전부 used로 마킹
168                    for i in run_start..run_start + count {
169                        set_used(i);
170                    }
171                    FREE_FRAMES.fetch_sub(count as u64, Ordering::Relaxed);
172                    return Some(run_start as u64 * PAGE_SIZE);
173                }
174            } else {
175                // 연속이 끊김 → 처음부터 다시
176                run_len = 0;
177            }
178        }
179    }
180    None // 연속된 count개 프레임 없음
181}

183/// 물리 프레임 해제 (반환)
184///
185/// `phys_addr`은 반드시 PAGE_SIZE 배수여야 함.
186#[allow(dead_code)] // 프로세스 종료 시 메모리 반환 때 사용
187pub fn free_frame(phys_addr: u64) {
188    let frame = (phys_addr / PAGE_SIZE) as usize;
189    if frame < MAX_FRAMES {
190        unsafe {
191            // 이미 free인데 또 free하는 double-free는 검사하지 않음 (성능 우선)
192            // 나중에 debug_assert!로 추가 가능
193            if !is_free(frame) {
194                set_free(frame);
195                FREE_FRAMES.fetch_add(1, Ordering::Relaxed);
196            }
197        }
198    }
199}

203pub fn free_frame_count() -> u64 {
204    FREE_FRAMES.load(Ordering::Relaxed)
205}

207#[allow(dead_code)]
208pub fn total_usable_frames() -> u64 {
209    TOTAL_USABLE.load(Ordering::Relaxed)
