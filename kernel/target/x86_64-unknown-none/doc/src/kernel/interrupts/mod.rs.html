1//! 인터럽트 서브시스템
2//!
3//! ## 초기화 순서 (반드시 이 순서를 지켜야 함)
4//!
5//! 1. `gdt::init()` — GDT + TSS 로드 (CS/SS 세그먼트 설정)
6//! 2. `idt::init()` — IDT 로드 (예외/IRQ 핸들러 등록)
7//! 3. `pic::init()` — 8259A PIC 리매핑 + IRQ 마스크 설정
8//! 4. `sti`         — 인터럽트 활성화 (IF 플래그 세트)
9//!
10//! GDT 이전에 IDT를 로드하면 CS 셀렉터가 맞지 않을 수 있고,
11//! IDT 이전에 STI하면 핸들러 없는 인터럽트가 트리플 폴트를 유발함.
12
13pub mod gdt;
14pub mod handlers;
15pub mod idt;
16pub mod pic;

18pub use handlers::TICK;

20/// SYSCALL/SYSRET MSR 초기화 — musl-static 등 `syscall` 인스트럭션 사용 바이너리 지원
21///
22/// `paging::init()`이 TSS.RSP0을 설정한 직후에 호출해야 함.
23/// `kern_rsp` = 커널 스택 최상단 (= TSS.RSP0 값).
24pub fn init_syscall(kern_rsp: u64) {
25    extern "C" { fn syscall_entry(); }
26    unsafe {
27        handlers::syscall_kern_rsp = kern_rsp;
28
29        // IA32_EFER (0xC000_0080): SCE 비트(0) 세트 → SYSCALL/SYSRET 활성화
30        let efer = rdmsr(0xC0000080);
31        wrmsr(0xC0000080, efer | 1);
32
33        // IA32_STAR (0xC000_0081):
34        //   [47:32] = 0x0008 → SYSCALL: CS=0x08(kernel code), SS=0x10(kernel data)
35        //   [63:48] = 0x0018 → SYSRET base (미사용 — 복귀는 IRETQ로)
36        wrmsr(0xC0000081, (0x0018u64 << 48) | (0x0008u64 << 32));
37
38        // IA32_LSTAR (0xC000_0082): syscall 진입점
39        wrmsr(0xC0000082, syscall_entry as u64);
40
41        // IA32_FMASK (0xC000_0084): syscall 진입 시 클리어할 RFLAGS 비트
42        // 0x200 = IF → syscall 처리 중 인터럽트 비활성
43        wrmsr(0xC0000084, 0x200);
44    }
45    crate::serial_println!(
46        "[syscall] LSTAR={:#x}  (syscall 인스트럭션 지원 ON)",
47        syscall_entry as usize
48    );
49}

51unsafe fn rdmsr(msr: u32) -> u64 {
52    let lo: u32;
53    let hi: u32;
54    core::arch::asm!(
55        "rdmsr",
56        in("ecx") msr,
57        out("eax") lo,
58        out("edx") hi,
59        options(nomem, nostack),
60    );
61    ((hi as u64) << 32) | lo as u64
62}
63
64unsafe fn wrmsr(msr: u32, val: u64) {
65    core::arch::asm!(
66        "wrmsr",
67        in("ecx") msr,
68        in("eax") val as u32,
69        in("edx") (val >> 32) as u32,
70        options(nomem, nostack),
71    );
72}

74/// 인터럽트 서브시스템 전체 초기화
75///
76/// `memory::init()` 이후에 호출해야 함 (시리얼 출력이 필요하기 때문).
77pub fn init() {
78    gdt::init();
79    idt::init();
80    pic::init();
81
82    // STI: RFLAGS.IF = 1 → CPU가 마스크되지 않은 인터럽트를 처리하기 시작
83    // 이 명령어 이후부터 타이머/키보드 인터럽트가 실제로 발생함
84    unsafe {
85        core::arch::asm!("sti", options(nomem, nostack));
86    }
87
88    crate::serial_println!("[interrupts] interrupts enabled (STI)");
89}
