1//! GDT (Global Descriptor Table) + TSS 설정
2//!
3//! ## GDT 레이아웃 (Milestone 3.6 기준)
4//!
5//! ```
6//! [0x00] null             — ABI 요구 (첫 엔트리 항상 null)
7//! [0x08] kernel code      — CS: L=1(64비트), DPL=0, execute/read
8//! [0x10] kernel data      — DS/SS: DPL=0, read/write
9//! [0x18] TSS low          — 16바이트 시스템 디스크립터 (두 슬롯)
10//! [0x20] TSS high         ┘
11//! [0x28] user code        — CS: L=1(64비트), DPL=3, execute/read
12//! [0x30] user data        — DS/SS: DPL=3, read/write
13//! ```
14//!
15//! ## 셀렉터 RPL(Request Privilege Level)
16//!
17//! IRETQ로 ring3로 진입할 때 CS/SS 셀렉터에 RPL=3을 OR해야 함.
18//! CPU가 IRETQ 시 셀렉터의 RPL을 새 CPL(Current Privilege Level)로 설정.
19//!
20//! 예: user code selector = 0x28, RPL=3 → 0x2B (push할 때 0x2B 사용)
21
22use core::{arch::asm, mem};
23
24// ── 셀렉터 상수 ────────────────────────────────────────────────────────────
25pub const KERNEL_CODE_SEL:      u16 = 0x08;
26pub const KERNEL_DATA_SEL:      u16 = 0x10;
27pub const TSS_SEL:              u16 = 0x18;
28/// ring3 코드 세그먼트 (RPL=3 OR 전 베이스값)
29pub const USER_CODE_SEL:        u16 = 0x28;
30/// ring3 데이터 세그먼트 (RPL=3 OR 전 베이스값)
31pub const USER_DATA_SEL:        u16 = 0x30;
32/// IRETQ 프레임에 넣을 user CS (RPL=3 세트)
33pub const USER_CS_RPL3:         u16 = USER_CODE_SEL | 3; // 0x2B
34/// IRETQ 프레임에 넣을 user SS (RPL=3 세트)
35pub const USER_SS_RPL3:         u16 = USER_DATA_SEL | 3; // 0x33
36
37// ── TSS ────────────────────────────────────────────────────────────────────
38//
39// x86_64 TSS는 104바이트 고정 크기.
40// 우리가 사용하는 필드:
41//   rsp[0] = ring3→ring0 전환 시 CPU가 로드할 커널 스택 포인터
42//            (int/exception이 ring3에서 발생하면 CPU가 자동으로 이 값을 RSP에 씀)
43#[repr(C, packed)]
44pub struct Tss {
45    _reserved0: u32,
46    pub rsp:        [u64; 3],   // RSP0-RSP2: 특권 레벨 전환용 스택
47    _reserved1: u64,
48    pub ist:        [u64; 7],   // IST1-IST7: 인터럽트 전용 스택 (현재 미사용)
49    _reserved2: u64,
50    _reserved3: u16,
51    pub iomap_base: u16,         // IOPB 오프셋: sizeof(Tss) → IOPB 없음
52}
53
54// ── GDT 테이블 ─────────────────────────────────────────────────────────────
55//
56// ## 세그먼트 디스크립터 비트 레이아웃 (8바이트)
57//
58// access byte(bits 47:40):
59//   bit 47 (P)   = 세그먼트 유효 여부 (1 = present)
60//   bit 46:45(DPL)= 특권 레벨 (00=ring0, 11=ring3)
61//   bit 44 (S)   = 0=시스템, 1=코드/데이터
62//   bit 43:40(Type)= 세그먼트 타입
63//     1010 = 코드, 실행/읽기
64//     0010 = 데이터, 읽기/쓰기
65//
66// flags nibble(bits 55:52): G | D/B | L | AVL
67//   L=1, D=0 → 64비트 코드 세그먼트
68//   G=1       → 리밋을 4KB 단위로 (64비트 모드에서는 무시)
69//
70// 커널 코드 (0x9A access, 0xA flags):
71//   access=0x9A: P=1, DPL=0, S=1, Type=0xA(코드 실행/읽기)
72//   flags=0xA:   G=1, D=0, L=1(64비트 코드)
73//   → 0x00af_9a00_0000_ffff
74//
75// 유저 코드 (0xFA access, 0xA flags):
76//   access=0xFA: P=1, DPL=3, S=1, Type=0xA
77//   → 0x00af_fa00_0000_ffff
78//
79// 커널/유저 데이터 (0x92/0xF2 access):
80//   access=0x92: P=1, DPL=0, S=1, Type=0x2(데이터 읽기/쓰기)
81//   access=0xF2: P=1, DPL=3, S=1, Type=0x2
82//   64비트 모드에서 데이터 세그먼트 base/limit/flags는 대부분 무시됨
83
84#[repr(C, align(8))]
85struct Gdt([u64; 7]);
86
87#[repr(C, packed)]
88struct GdtPointer { limit: u16, base: u64 }
89
90static mut TSS: Tss = Tss {
91    _reserved0: 0,
92    rsp:        [0; 3],
93    _reserved1: 0,
94    ist:        [0; 7],
95    _reserved2: 0,
96    _reserved3: 0,
97    iomap_base: mem::size_of::<Tss>() as u16,
98};
99
100static mut GDT: Gdt = Gdt([
101    0x0000_0000_0000_0000, // [0] null
102    0x00af_9a00_0000_ffff, // [1] kernel code  (DPL=0, L=1) → 0x08
103    0x00cf_9200_0000_ffff, // [2] kernel data  (DPL=0)      → 0x10
104    0x0000_0000_0000_0000, // [3] TSS low      (runtime)    → 0x18
105    0x0000_0000_0000_0000, // [4] TSS high     (runtime)    → 0x20
106    0x00af_fa00_0000_ffff, // [5] user code    (DPL=3, L=1) → 0x28
107    0x00cf_f200_0000_ffff, // [6] user data    (DPL=3)      → 0x30
108]);
109
110pub fn init() {
111    unsafe {
112        // ── TSS 디스크립터 빌드 (16바이트 시스템 디스크립터) ───────────────
113        //
114        // 하위 8바이트:
115        //   bits 0:15  = limit[15:0]
116        //   bits 16:39 = base[23:0]
117        //   bits 40:47 = access (0x89 = Present, Available 64-bit TSS)
118        //   bits 48:51 = limit[19:16]
119        //   bits 56:63 = base[31:24]
120        // 상위 8바이트:
121        //   bits 0:31  = base[63:32]
122        let tss_ptr  = &raw const TSS as u64;
123        let tss_size = (mem::size_of::<Tss>() - 1) as u64;
124
125        let tss_low =
126              (tss_size & 0xFFFF)
127            | ((tss_ptr & 0x00FF_FFFF) << 16)
128            | (0x89_u64 << 40)
129            | (((tss_size >> 16) & 0xF) << 48)
130            | (((tss_ptr >> 24) & 0xFF) << 56);
131
132        GDT.0[3] = tss_low;
133        GDT.0[4] = tss_ptr >> 32;
134
135        // ── GDTR 로드 ──────────────────────────────────────────────────────
136        let gdtr = GdtPointer {
137            limit: (mem::size_of::<Gdt>() - 1) as u16,
138            base:  GDT.0.as_ptr() as u64,
139        };
140        asm!("lgdt [{}]", in(reg) &gdtr, options(readonly, nostack, preserves_flags));
141
142        // ── CS 재로드 (원거리 반환 트릭) ───────────────────────────────────
143        // lgdt 이후 세그먼트 디스크립터 캐시를 갱신하려면 CS를 다시 로드해야 함.
144        // 64비트에서 CS는 MOV 불가 → 스택에 [새 CS, 복귀 RIP] push 후 LRETQ.
145        asm!(
146            "pushq {sel}",
147            "leaq 1f(%rip), %rax",
148            "pushq %rax",
149            "lretq",
150            "1:",
151            sel = in(reg) KERNEL_CODE_SEL as u64,
152            out("rax") _,
153            options(att_syntax),
154        );
155
156        // ── 데이터 세그먼트 재로드 ─────────────────────────────────────────
157        asm!(
158            "mov ds, ax",
159            "mov es, ax",
160            "mov ss, ax",
161            in("ax") KERNEL_DATA_SEL,
162            options(nostack, nomem, preserves_flags),
163        );
164        asm!(
165            "xor eax, eax",
166            "mov fs, ax",
167            "mov gs, ax",
168            out("eax") _,
169            options(nostack, nomem, preserves_flags),
170        );
171
172        // ── TSS 로드 ───────────────────────────────────────────────────────
173        asm!("ltr ax", in("ax") TSS_SEL, options(nostack, nomem));
174    }
175
176    crate::serial_println!(
177        "[gdt] GDT loaded: kernel(0x{:02x}/0x{:02x}) tss(0x{:02x}) user(0x{:02x}/0x{:02x})",
178        KERNEL_CODE_SEL, KERNEL_DATA_SEL, TSS_SEL, USER_CODE_SEL, USER_DATA_SEL
179    );
180}

182/// TSS.RSP0 설정 — ring3 → ring0 전환 시 CPU가 사용할 커널 스택 포인터
183///
184/// int/exception이 ring3에서 발생할 때 CPU는 자동으로:
185/// 1. RSP ← TSS.RSP0 (이 함수로 설정한 값)
186/// 2. 기존 SS/RSP/RFLAGS/CS/RIP를 새 스택에 push
187/// 3. 핸들러로 점프
188///
189/// paging::init()에서 전용 스택(USER_KERNEL_STACK)을 할당 후 여기에 설정.
190pub fn set_tss_rsp0(rsp0: u64) {
191    unsafe { TSS.rsp[0] = rsp0; }
192    crate::serial_println!("[gdt] TSS.RSP0 = {:#x}", rsp0);
193}
