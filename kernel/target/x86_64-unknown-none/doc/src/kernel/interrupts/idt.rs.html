1//! IDT (Interrupt Descriptor Table) 설정
2//!
3//! ## IDT 게이트 디스크립터 (16바이트)
4//!
5//! ```
6//! bits 0:15   = handler[15:0]
7//! bits 16:31  = 코드 세그먼트 셀렉터 (CS = 0x08)
8//! bits 32:34  = IST 인덱스 (0 = 전용 스택 없음)
9//! bits 35:39  = 예약 (0)
10//! bits 40:47  = 타입 + 속성
11//!                 0x8E = 인터럽트 게이트 (IF 클리어, 하드웨어 IRQ에 사용)
12//!                 0x8F = 트랩 게이트    (IF 유지,   CPU 예외에 사용)
13//! bits 48:63  = handler[31:16]
14//! bits 64:95  = handler[63:32]
15//! bits 96:127 = 예약 (0)
16//! ```
17//!
18//! ## 인터럽트 게이트 vs 트랩 게이트
19//!
20//! 인터럽트 게이트 (0x8E): 진입 시 RFLAGS.IF = 0 (인터럽트 비활성화)
21//!   → 하드웨어 IRQ: 핸들러 실행 중 같은 IRQ가 재진입하는 것을 방지
22//!
23//! 트랩 게이트 (0x8F): RFLAGS.IF 변경 없음
24//!   → CPU 예외: 예외 핸들러 안에서도 다른 인터럽트 허용 (디버깅 편의)
25
26use core::{arch::asm, mem};
27use super::gdt::KERNEL_CODE_SEL;
28
29const INT_GATE:       u8 = 0x8E; // 인터럽트 게이트 (진입 시 IF 클리어, DPL=0)
30const TRAP_GATE:      u8 = 0x8F; // 트랩 게이트 (IF 유지, DPL=0)
31/// 트랩 게이트 DPL=3: ring3의 `int 0x80` 소프트웨어 인터럽트 허용
32/// DPL이 3이어야 ring3 코드가 `int 0x80`을 발생시킬 수 있음.
33/// DPL=0이면 ring3에서 int 0x80 시 #GP 발생.
34const TRAP_GATE_DPL3: u8 = 0xEF; // P=1, DPL=3, Type=0xF (trap gate)
35
36/// IDT 게이트 디스크립터 (16바이트)
37#[derive(Clone, Copy)]
38#[repr(C, packed)]
39struct IdtEntry {
40    offset_low:  u16,
41    selector:    u16,
42    ist:         u8,  // bits[2:0] = IST 인덱스 (0 = 전용 스택 없음)
43    type_attr:   u8,
44    offset_mid:  u16,
45    offset_high: u32,
46    _reserved:   u32,
47}
48
49impl IdtEntry {
50    const fn absent() -> Self {
51        Self { offset_low: 0, selector: 0, ist: 0, type_attr: 0,
52               offset_mid: 0, offset_high: 0, _reserved: 0 }
53    }
54
55    fn set(&mut self, handler: u64, gate: u8, ist: u8) {
56        self.offset_low  = handler as u16;
57        self.offset_mid  = (handler >> 16) as u16;
58        self.offset_high = (handler >> 32) as u32;
59        self.selector    = KERNEL_CODE_SEL;
60        self.ist         = ist & 0x7;
61        self.type_attr   = gate;
62        self._reserved   = 0;
63    }
64}
65
66#[repr(C, align(16))]
67struct Idt([IdtEntry; 256]);
68
69#[repr(C, packed)]
70struct IdtPointer {
71    limit: u16,
72    base:  u64,
73}
74
75static mut IDT: Idt = Idt([IdtEntry::absent(); 256]);
76
77// handlers.rs의 global_asm에서 정의된 어셈블리 스텁 선언
78extern "C" {
79    fn isr0();  fn isr1();  fn isr2();  fn isr3();
80    fn isr4();  fn isr5();  fn isr6();  fn isr7();
81    fn isr8();  fn isr9();  fn isr10(); fn isr11();
82    fn isr12(); fn isr13(); fn isr14(); fn isr15();
83    fn isr16(); fn isr17(); fn isr18(); fn isr19();
84    fn isr20(); fn isr21(); fn isr22(); fn isr23();
85    fn isr24(); fn isr25(); fn isr26(); fn isr27();
86    fn isr28(); fn isr29(); fn isr30(); fn isr31();
87    fn isr32();  // 타이머 IRQ0 — 선점형 스케줄러
88    fn isr33();  // 키보드 IRQ1
89    fn isr44();  // 마우스 IRQ12 (PIC2, 벡터 0x2C)
90    fn isr64();  // 자발적 양보 (int 0x40, yield_now)
91    fn isr128(); // 소프트웨어 인터럽트 (syscall: int 0x80)
92}

94pub fn init() {
95    unsafe {
96        // ── CPU 예외 (벡터 0-31): 트랩 게이트 ────────────────────────────
97        // 트랩 게이트를 쓰는 이유: 예외 핸들러 안에서도 타이머 인터럽트 등
98        // 다른 인터럽트가 처리될 수 있어야 함 (특히 디버그 예외).
99        // IST = 0: 전용 스택 없음 (현재 커널 스택 그대로 사용).
100        let stubs: [unsafe extern "C" fn(); 32] = [
101            isr0,  isr1,  isr2,  isr3,  isr4,  isr5,  isr6,  isr7,
102            isr8,  isr9,  isr10, isr11, isr12, isr13, isr14, isr15,
103            isr16, isr17, isr18, isr19, isr20, isr21, isr22, isr23,
104            isr24, isr25, isr26, isr27, isr28, isr29, isr30, isr31,
105        ];
106        for (vec, &stub) in stubs.iter().enumerate() {
107            IDT.0[vec].set(stub as u64, TRAP_GATE, 0);
108        }
109
110        // ── 하드웨어 IRQ: 인터럽트 게이트 ────────────────────────────────
111        // 인터럽트 게이트: 핸들러 실행 중 IF=0 → 같은 IRQ 재진입 방지.
112        // PIC 리매핑 후: IRQ0 → 벡터 0x20, IRQ1 → 벡터 0x21.
113        IDT.0[0x20].set(isr32 as u64, INT_GATE, 0); // IRQ0  = 타이머
114        IDT.0[0x21].set(isr33 as u64, INT_GATE, 0); // IRQ1  = 키보드
115        IDT.0[0x2C].set(isr44 as u64, INT_GATE, 0); // IRQ12 = 마우스 (PIC2)
116
117        // ── 자발적 양보: yield_now() = int 0x40 (ALPHA M1) ───────────────
118        // INT_GATE, DPL=0: 커널 코드(ring0)만 발생 가능.
119        // 인터럽트 게이트이므로 IF=0 → voluntary_yield 실행 중 재진입 방지.
120        IDT.0[0x40].set(isr64 as u64, INT_GATE, 0);
121
122        // ── 소프트웨어 인터럽트: syscall (벡터 0x80) ─────────────────────
123        // TRAP_GATE_DPL3: ring3 코드가 `int 0x80`을 실행할 수 있게 DPL=3.
124        // 트랩 게이트를 쓰는 이유: syscall 처리 중 타이머 인터럽트 허용.
125        IDT.0[0x80].set(isr128 as u64, TRAP_GATE_DPL3, 0);
126
127        // ── IDTR 로드 ──────────────────────────────────────────────────────
128        let idtr = IdtPointer {
129            limit: (mem::size_of::<Idt>() - 1) as u16,
130            base:  IDT.0.as_ptr() as u64,
131        };
132        asm!("lidt [{}]", in(reg) &idtr, options(readonly, nostack, preserves_flags));
133    }
134
135    crate::serial_println!("[idt] IDT loaded (32 exceptions + IRQ0/IRQ1/IRQ12 + int 0x40/0x80)");
136}
