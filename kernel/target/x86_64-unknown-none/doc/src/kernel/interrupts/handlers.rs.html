1//! 인터럽트/예외 핸들러 + 어셈블리 스텁
2//!
3//! ## 인터럽트 진입 흐름
4//!
5//! CPU가 인터럽트를 감지하면:
6//! 1. 현재 실행을 중단
7//! 2. 스택에 복귀 정보 push: [RFLAGS, CS, RIP] (+ 에러 코드가 있는 예외는 error_code)
8//! 3. IDT에서 핸들러 주소를 읽어 점프
9//!
10//! 우리 스텁(stub)이 하는 일:
11//! 1. 에러 코드가 없는 예외: 더미 0 push → 벡터 번호 push
12//! 2. 에러 코드가 있는 예외: 벡터 번호만 push (에러 코드는 CPU가 이미 push)
13//! 3. IRQ: 모든 caller-saved 레지스터 저장 → Rust 핸들러 호출 → 복원
14//!
15//! ## 스택 레이아웃 (exception_common 진입 시점)
16//!
17//! ```
18//! [낮은 주소]  ← RSP (exception_common 진입 시)
19//!   vector
20//!   error_code   (없으면 더미 0)
21//!   RIP
22//!   CS
23//!   RFLAGS
24//! [높은 주소]
25//! ```
26

30/// PIT 타이머 틱 카운터 (원자적 — 향후 멀티코어 대비)
31pub static TICK: AtomicU64 = AtomicU64::new(0);

33/// 예외 진입 시 스택 레이아웃 (어셈블리 스텁이 구성)
34///
35/// `exception_common`이 레지스터를 push한 뒤 RSP가 이 구조체를 가리킴.
36/// `#[repr(C)]`로 필드 순서와 패딩을 보장.
37#[repr(C)]
38pub struct ExceptionFrame {
39    // ── 스텁이 push한 레지스터 (역순: 마지막 push = 낮은 주소) ──────────
40    pub r15: u64, pub r14: u64, pub r13: u64, pub r12: u64,
41    pub r11: u64, pub r10: u64, pub r9:  u64, pub r8:  u64,
42    pub rbp: u64, pub rdi: u64, pub rsi: u64, pub rdx: u64,
43    pub rcx: u64, pub rbx: u64, pub rax: u64,
44    // ── 스텁이 push한 메타정보 ───────────────────────────────────────────
45    pub vector:     u64, // 인터럽트 벡터 번호
46    pub error_code: u64, // CPU가 push한 에러 코드 (없으면 더미 0)
47    // ── CPU가 자동으로 push한 복귀 정보 ─────────────────────────────────
48    pub rip:    u64,
49    pub cs:     u64,
50    pub rflags: u64,
51    // NOTE: ring3→ring0 전환 시에만 RSP/SS가 추가로 push됨
52}

54/// CPU 예외 핸들러 (어셈블리 exception_common에서 호출)
55///
56/// 현재 구현: 예외 정보를 시리얼에 출력 후 무한 HLT.
57/// 복구 불가 예외는 항상 커널 패닉으로 처리.
58#[no_mangle]
59pub extern "C" fn exception_handler(frame: &ExceptionFrame) {
60    let vec = frame.vector;
61    let ec  = frame.error_code;
62
63    // Page Fault(#PF, 벡터 14)일 때 CR2 = 폴트 발생 가상 주소
64    let cr2: u64 = if vec == 14 {
65        let v: u64;
66        unsafe {
67            core::arch::asm!("mov {}, cr2",
68                out(reg) v,
69                options(nomem, nostack, preserves_flags));
70        }
71        v
72    } else { 0 };
73
74    crate::serial_println!("\n!!! CPU EXCEPTION !!!");
75    crate::serial_println!("  vec={} ({})", vec, exception_name(vec));
76    crate::serial_println!("  error_code = {:#x}", ec);
77    crate::serial_println!("  RIP    = {:#018x}", frame.rip);
78    crate::serial_println!("  CS     = {:#x}", frame.cs);
79    crate::serial_println!("  RFLAGS = {:#x}", frame.rflags);
80    if vec == 14 {
81        // 에러 코드 비트: P(존재), W(쓰기), U(유저), I(명령어 fetch)
82        crate::serial_println!("  CR2    = {:#018x}  (폴트 주소)", cr2);
83        crate::serial_println!("  PF flags: P={} W={} U={} I={}",
84            ec & 1, (ec >> 1) & 1, (ec >> 2) & 1, (ec >> 4) & 1);
85    }
86
87    loop {
88        unsafe {
89            core::arch::asm!("cli; hlt", options(nomem, nostack, preserves_flags));
90        }
91    }
92}

94/// 타이머 인터럽트 선점형 핸들러 (ALPHA M1)
95///
96/// `isr32` 어셈블리 스텁이 ALL 레지스터를 스택에 저장한 뒤 이 함수를 호출.
97///
98/// ## 인수
99/// `rsp`: ISR 스텁이 ALL 레지스터를 push한 직후의 스택 포인터.
100///        = 현재 프로세스의 선점형 컨텍스트 프레임 시작 주소.
101///
102/// ## 반환값
103/// 다음에 실행할 프로세스의 `preempt_rsp`.
104/// 스위치가 없으면 `rsp` 그대로 반환.
105/// 스위치가 있으면 다른 프로세스의 저장된 RSP를 반환.
106///
107/// ## 동작
108/// 1. TICK 증가 + PIC EOI
109/// 2. TIME_SLICE 틱마다 `scheduler::preempt(rsp)` 호출
110/// 3. 반환값은 `isr32` 스텁의 `mov rsp, rax`로 스택을 전환
111#[no_mangle]
112pub extern "C" fn timer_preempt(rsp: u64) -> u64 {
113    let tick = TICK.fetch_add(1, Ordering::Relaxed) + 1;
114    if tick % 18 == 0 {
115        crate::serial_println!("[timer] tick={} (~{}s)", tick, tick / 18);
116    }
117    // PIC Master에 EOI 전송 — 이 인터럽트 처리 완료 신호
118    // EOI 전에 다른 인터럽트를 막기 위해 핸들러 진입 시 IF=0 (INT_GATE)
119    pic::eoi_master();
120
121    // TIME_SLICE: Policy Engine이 실시간 조정. 기본 3틱(≈165ms).
122    let slice = crate::policy::TIME_SLICE.load(Ordering::Relaxed).max(1);
123    if tick % slice == 0 {
124        crate::process::scheduler::preempt(rsp)
125    } else {
126        rsp
127    }
128}

131///
132/// `yield_now()`가 `int 0x40`으로 구현됨.
133/// `isr64` 스텁이 ALL 레지스터를 스택에 저장한 뒤 이 함수를 호출.
134/// 타이머 선점과 완전히 동일한 ISR 프레임 형식 → 컨텍스트 전환 방식 통일.
135///
136/// 반환값: 다음 프로세스의 `preempt_rsp` (항상 즉시 스위치).
137#[no_mangle]
138pub extern "C" fn voluntary_yield(rsp: u64) -> u64 {
139    crate::process::scheduler::voluntary_preempt(rsp)
140}

142/// 키보드 인터럽트 핸들러 (IRQ1, 벡터 0x21)
143///
144/// 키보드 컨트롤러(i8042)는 IRQ1 발생 시 포트 0x60에 스캔 코드를 올림.
145/// 반드시 포트 0x60을 읽어야 인터럽트가 해제됨.
146///
147/// 스캔 코드 Set 1 (기본값):
148/// - 0x01-0x58: 키 누름 (make code)
149/// - 0x81-0xD8: 키 뗌   (break code = make | 0x80)
150#[no_mangle]
151pub extern "C" fn irq_handler_keyboard() {
152    let scancode: u8;
153    unsafe {
154        core::arch::asm!("in al, 0x60",
155            out("al") scancode,
156            options(nomem, nostack, preserves_flags));
157    }
158    crate::kbd::handle_scancode(scancode);
159    if scancode < 0x80 {
160        crate::process::scheduler::keyboard_boost();
161    }
162    pic::eoi_master();
163}

175// ── Syscall 핸들러 (int 0x80) ────────────────────────────────────────────────
176//
177// ring3 → ring0 전환 흐름:
178//   1. ring3 코드 `int 0x80` 실행
179//   2. CPU: TSS.RSP0으로 스택 전환, [SS, RSP, RFLAGS, CS, RIP] push
180//   3. IDT[0x80] 핸들러(isr128 스텁) 진입
181//   4. isr128: caller-saved 레지스터 저장 → syscall_handler() 호출
182//   5. syscall_handler(): 처리 후 반환 → isr128: 레지스터 복원 → IRETQ ring3 복귀
183//
184// 3회 호출 후 longjmp:
185//   paging::KERNEL_MAIN_RSP로 RSP 복원 + after_user_demo() 점프.
186//   longjmp 이후 isr128의 나머지 코드는 실행되지 않음.
187
188/// ring3 syscall 호출 카운터 (디버그용)
189pub static USER_SYSCALL_COUNT: AtomicU32 = AtomicU32::new(0);
190
191/// ring3 프로세스가 `sys_exit(code)`를 호출할 때 저장되는 종료 코드
192pub static USER_EXIT_CODE: AtomicI32 = AtomicI32::new(0);

194/// `syscall` 인스트럭션 진입 시 유저 RSP 임시 저장 (LSTAR 핸들러용)
195#[no_mangle]
196pub static mut syscall_user_rsp: u64 = 0;
197/// `syscall` 인스트럭션 핸들러가 사용할 커널 RSP (= TSS.RSP0)
198#[no_mangle]
199pub static mut syscall_kern_rsp: u64 = 0;

201/// int 0x80 syscall 디스패처 (ALPHA 13)
202///
203/// `isr128` 스텁이 9개 레지스터를 push한 직후 RSP를 이 함수에 인수로 전달.
204/// 반환값(i64)은 isr128이 저장된 RAX 슬롯([rsp+64])에 써 넣어 user-space로 전달.
205///
206/// ## 스택 레이아웃 (`frame_rsp` 기준)
207///
208/// ```text
209/// frame[0] = r11  frame[1] = r10(arg4)  frame[2] = r9(arg6)
210/// frame[3] = r8(arg5)   frame[4] = rdi(arg1)  frame[5] = rsi(arg2)
211/// frame[6] = rdx(arg3)  frame[7] = rcx        frame[8] = rax(syscall#)
212/// ```
213#[no_mangle]
214pub extern "C" fn syscall_dispatch(frame_rsp: u64) -> i64 {
215    USER_SYSCALL_COUNT.fetch_add(1, Ordering::Relaxed);
216    crate::syscall::dispatch(frame_rsp as *const u64)
217}

219/// 실행 대기 중인 패키지 인덱스 (ALPHA 16)
220///
221/// `-1` = 없음 (mushell 재시작), `>= 0` = `pkg::PACKAGES[idx]` 실행.
222/// `sys_execve`가 설정하고 `after_user_demo` Phase 2+가 소비.
223pub static PENDING_PKG_IDX: AtomicI32 = AtomicI32::new(-1);
224
225/// ring3 프로세스 종료 후 커널이 재개하는 지점 (sys_exit longjmp 목적지)
226///
227/// ## Phase 설계
228///
229/// - **Phase 0**: ALPHA 13 raw code 종료 → ALPHA 14 test ELF 실행
230/// - **Phase 1**: ALPHA 14 ELF 종료 → ALPHA 15 mushell 첫 실행
231/// - **Phase 2+**: mushell/패키지 종료 → PENDING_PKG_IDX 확인
232///   - `>= 0`: 해당 패키지 ELF 실행 (ALPHA 16 mukg run)
233///   - `-1`:   mushell 재시작 (패키지 종료 후 셸 복귀)
234///
235/// longjmp로 도달 — `extern "C"` + `#[no_mangle]` 필수.
236#[no_mangle]
237pub extern "C" fn after_user_demo() -> ! {
238    static PHASE: core::sync::atomic::AtomicU8 = core::sync::atomic::AtomicU8::new(0);
239    let phase = PHASE.fetch_add(1, Ordering::Relaxed);
240    let calls = USER_SYSCALL_COUNT.load(Ordering::Relaxed);
241    let code  = USER_EXIT_CODE.load(Ordering::Relaxed);
242
243    USER_SYSCALL_COUNT.store(0, Ordering::Relaxed);
244    USER_EXIT_CODE.store(0, Ordering::Relaxed);
245
246    match phase {
247        0 => {
248            // ALPHA 13 완료 → ALPHA 14 진입
249            crate::serial_println!(
250                "[ring3] exited: code={} ({} syscalls total)", code, calls
251            );
252            crate::serial_println!("--- ALPHA 13 complete ---\n");
253            crate::serial_println!("===========================================");
254            crate::serial_println!("  ALPHA 14: ELF Loader");
255            crate::serial_println!("  (실제 ELF64 바이너리를 ring3에서 실행)");
256            crate::serial_println!("===========================================");
257            crate::serial_println!("[elf] loading test ELF ({} bytes)...",
258                crate::ELF_TEST.len());
259            unsafe { crate::paging::enter_elf(crate::ELF_TEST); }
260        }
261        1 => {
262            // ALPHA 14 완료 → ALPHA 15 셸 첫 진입
263            crate::serial_println!(
264                "[ring3] ELF test exited: code={} ({} syscalls total)", code, calls
265            );
266            crate::serial_println!("--- ALPHA 14 complete ---\n");
267            crate::serial_println!("===========================================");
268            crate::serial_println!("  ALPHA 15 / 16: Shell + Package Manager");
269            crate::serial_println!("  (mukg list/install/run 으로 패키지 실행)");
270            crate::serial_println!("===========================================");
271            crate::serial_println!("[shell] loading mushell ({} bytes)...",
272                crate::MUSHELL_ELF.len());
273            crate::syscall::fd::init(); // BETA 4: fd 테이블 초기화
274            unsafe { crate::paging::enter_elf(crate::MUSHELL_ELF); }
275        }
276        _ => {
277            // Phase 2+: mushell 또는 패키지 종료
278            PHASE.store(2, Ordering::Relaxed);
279
280            let pkg_idx = PENDING_PKG_IDX.swap(-1, Ordering::Relaxed);
281            if pkg_idx >= 0 {
282                let pkg = &crate::pkg::PACKAGES[pkg_idx as usize];
283                crate::serial_println!("[mukg] running {} v{} ({} bytes)...",
284                    pkg.name, pkg.version, pkg.elf.len());
285                crate::syscall::fd::init(); // BETA 4
286                unsafe { crate::paging::enter_elf(pkg.elf); }
287            } else {
288                crate::syscall::fd::init(); // BETA 4
289                unsafe { crate::paging::enter_elf(crate::MUSHELL_ELF); }
290            }
291        }
292    }
293}

295fn exception_name(v: u64) -> &'static str {
296    match v {
297        0  => "#DE Divide Error",
298        1  => "#DB Debug",
299        2  => "NMI",
300        3  => "#BP Breakpoint",
301        4  => "#OF Overflow",
302        5  => "#BR Bound Range",
303        6  => "#UD Invalid Opcode",
304        7  => "#NM Device Not Available",
305        8  => "#DF Double Fault",
306        9  => "Coprocessor Segment Overrun",
307        10 => "#TS Invalid TSS",
308        11 => "#NP Segment Not Present",
309        12 => "#SS Stack-Segment Fault",
310        13 => "#GP General Protection",
311        14 => "#PF Page Fault",
312        16 => "#MF x87 FPU Error",
313        17 => "#AC Alignment Check",
314        18 => "#MC Machine Check",
315        19 => "#XF SIMD FPU Error",
316        _  => "Unknown",
317    }
318}

320// ══ 어셈블리 인터럽트 스텁 ═══════════════════════════════════════════════════
321//
322// ## 레지스터 저장 전략
323//
324// x86_64 System V ABI에서 "caller-saved" 레지스터:
325//   RAX, RCX, RDX, RSI, RDI, R8, R9, R10, R11
326// "callee-saved" 레지스터 (Rust 핸들러가 직접 보존):
327//   RBX, RBP, R12, R13, R14, R15
328//
329// 예외 핸들러(exception_common)는 모든 레지스터를 저장 (디버깅 정보용).
330// IRQ 핸들러는 caller-saved만 저장 (callee-saved는 Rust가 알아서 보존).
331//
332// ## 스택 정렬
333//
334// System V ABI: CALL 직전 RSP는 16바이트 정렬이어야 함.
335// CPU가 인터럽트 시 3개(rip, cs, rflags = 24바이트) push → RSP는 8 misaligned.
336// IRQ 스텁: 9개 caller-saved push(72바이트) → 합계 96바이트 → 16 정렬 ✓
337// exception 스텁: 추가 2개 push(vector+ec = 16바이트) + 15개 push(120바이트)
338//                 합계 160바이트 → 16 정렬 ✓ (에러코드 있는 예외도 동일)
339
340core::arch::global_asm!(
341    // ── exception_common: 모든 예외의 공통 진입점 ────────────────────────
342    // 진입 시 스택: [vector, error_code, rip, cs, rflags]
343    "exception_common:",
344    "push rax", "push rbx", "push rcx", "push rdx",
345    "push rsi", "push rdi", "push rbp",
346    "push r8",  "push r9",  "push r10", "push r11",
347    "push r12", "push r13", "push r14", "push r15",
348    "mov rdi, rsp",              // 첫 번째 인수: &ExceptionFrame
349    "call exception_handler",
350    "pop r15", "pop r14", "pop r13", "pop r12",
351    "pop r11", "pop r10", "pop r9",  "pop r8",
352    "pop rbp",  "pop rdi", "pop rsi", "pop rdx",
353    "pop rcx",  "pop rbx", "pop rax",
354    "add rsp, 16",               // vector + error_code 제거
355    "iretq",
356
357    // ── isr32: 타이머 IRQ (벡터 0x20) — 선점형 스케줄러 (ALPHA M1) ────────
358    //
359    // ## 선점형 컨텍스트 스위치 원리
360    //
361    // 1. CPU가 타이머 IRQ 감지 → ring0 인터럽트이므로 스택에 push:
362    //       [RFLAGS, CS, RIP]  (3개, 24바이트)
363    //    이후 IDT의 핸들러(isr32)로 점프.
364    //
365    // 2. isr32 스텁: ALL 범용 레지스터를 스택에 push (15개, 120바이트).
366    //    push 순서: rax, rcx, rdx, rsi, rdi, r8..r11, rbx, rbp, r12..r15
367    //    이후 RSP = 프레임 시작 (r15 위치).
368    //
369    // 3. `mov rdi, rsp` → timer_preempt(rsp) 호출.
370    //    반환값(rax) = 다음 프로세스의 preempt_rsp.
371    //
372    // 4. `mov rsp, rax` → 스택 전환 (다른 RSP면 다른 프로세스의 스택).
373    //
374    // 5. pop r15..rax → 새 프로세스의 레지스터 복원.
375    //
376    // 6. iretq → 새 프로세스의 [RIP, CS, RFLAGS] 팝 → 새 프로세스 실행.
377    //

527    //
528    // CPU가 `syscall` 실행 시:
529    //   RCX ← 복귀 RIP, R11 ← 유저 RFLAGS, RSP ← 그대로(유저 RSP!)
530    //   CS  ← STAR[47:32] (커널 코드), RFLAGS &= ~FMASK
531    //
532    // isr128(int 0x80)과 동일한 9-push 프레임을 구성해 syscall_dispatch 재사용.
533    // 복귀는 SYSRETQ 대신 IRETQ 사용 (GDT STAR[63:48] 정렬 불필요).
534    ".global syscall_entry",
535    "syscall_entry:",
536    // 유저 RSP 저장 + 커널 스택으로 전환 (RIP-relative: 커널 상위 절반 주소)
537    "mov [rip + syscall_user_rsp], rsp",
538    "mov rsp, [rip + syscall_kern_rsp]",
539    // isr128과 동일한 순서로 push → syscall_dispatch frame 재사용
540    "push rax",   // [rsp+0x40] = syscall nr
541    "push rcx",   // [rsp+0x38] = 복귀 RIP (dispatch에서 미사용)
542    "push rdx",   // [rsp+0x30] = arg3
543    "push rsi",   // [rsp+0x28] = arg2
544    "push rdi",   // [rsp+0x20] = arg1
545    "push r8",    // [rsp+0x18] = arg5
546    "push r9",    // [rsp+0x10] = arg6
547    "push r10",   // [rsp+0x08] = arg4
548    "push r11",   // [rsp+0x00] = 유저 RFLAGS (dispatch에서 미사용)
549    "mov rdi, rsp",
550    "call syscall_dispatch",
551    "mov [rsp + 0x40], rax",   // rax 슬롯에 반환값 덮어씀
552    // 복원
553    "pop r11",   // 유저 RFLAGS (IRETQ에 사용)
554    "pop r10",
555    "pop r9",
556    "pop r8",
557    "pop rdi",
558    "pop rsi",
559    "pop rdx",
560    "pop rcx",   // 복귀 RIP (IRETQ에 사용)
561    "pop rax",   // syscall 반환값
562    // r10을 user_rsp 로드에 임시 사용 (caller-saved이므로 무방)
563    "mov r10, [rip + syscall_user_rsp]",
564    // IRETQ 프레임 구성 (ring3 복귀)
565    "push 0x33",   // SS  = USER_SS_RPL3
566    "push r10",    // RSP = 유저 RSP
567    "push r11",    // RFLAGS
568    "push 0x2b",   // CS  = USER_CS_RPL3
569    "push rcx",    // RIP = 복귀 주소
570    "iretq",

572    // ── isr128: int 0x80 syscall (벡터 0x80) — ALPHA 13 ─────────────────
573    //
574    // ring3 → ring0 전환: CPU가 스택에 SS,RSP,RFLAGS,CS,RIP(5개=40B) push.
575    //
576    // 저장 순서 (낮은 주소 먼저):
577    //   push rax, rcx, rdx, rsi, rdi, r8, r9, r10, r11  (9개 = 72B)
578    //
579    // 스택 레이아웃 (9 push 직후, `call` 이전):
580    //   [rsp+0x00] r11   [rsp+0x08] r10(arg4)  [rsp+0x10] r9(arg6)
581    //   [rsp+0x18] r8(arg5)  [rsp+0x20] rdi(arg1)  [rsp+0x28] rsi(arg2)
582    //   [rsp+0x30] rdx(arg3)  [rsp+0x38] rcx  [rsp+0x40] rax(syscall#)
583    //
584    // 1. `mov rdi, rsp` → frame_rsp = 9-push 직후 RSP (call 전이라 return addr 없음)
585    // 2. `call syscall_dispatch(frame_rsp)` → RAX = syscall 반환값 (i64)
586    // 3. `mov [rsp+0x40], rax` → 저장된 RAX 슬롯을 반환값으로 덮어씀
587    //    → `pop rax` 시 user-space가 반환값을 읽음
588    //
589    // sys_exit: longjmp 발생 → IRETQ 미실행, 스텁 자연 종료.
590    ".global isr128",
591    "isr128:",
592    "push rax", "push rcx", "push rdx",
593    "push rsi",  "push rdi",
594    "push r8",  "push r9",  "push r10", "push r11",
595    "mov rdi, rsp",                 // frame_rsp → arg1
596    "call syscall_dispatch",        // RAX = 반환값 (i64)
597    "mov [rsp + 0x40], rax",        // 저장된 user RAX 슬롯 덮어쓰기
598    "pop r11", "pop r10", "pop r9",  "pop r8",
599    "pop rdi",  "pop rsi",
600    "pop rdx",  "pop rcx", "pop rax", // rax = syscall 반환값
601    "iretq",     // ring3 (사용자 코드)로 복귀
