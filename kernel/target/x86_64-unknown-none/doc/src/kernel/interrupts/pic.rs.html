1//! 8259A PIC (Programmable Interrupt Controller) 드라이버
2//!
3//! ## 하드웨어 구조
4//!
5//! PC에는 두 개의 8259A PIC가 계단식(cascade)으로 연결됨:
6//! - 마스터 PIC: IRQ0-7  → 슬레이브의 IRQ2에 연결
7//! - 슬레이브 PIC: IRQ8-15
8//!
9//! ```
10//! 마스터 PIC (0x20/0x21)       슬레이브 PIC (0xA0/0xA1)
11//! IRQ0 = PIT 타이머             IRQ8  = RTC
12//! IRQ1 = 키보드                 IRQ9  = (재지정 가능)
13//! IRQ2 = 슬레이브 캐스케이드    IRQ10 = (재지정 가능)
14//! IRQ3 = COM2                   IRQ11 = (재지정 가능)
15//! IRQ4 = COM1                   IRQ12 = PS/2 마우스
16//! IRQ5 = LPT2                   IRQ13 = FPU
17//! IRQ6 = 플로피                 IRQ14 = ATA 기본
18//! IRQ7 = LPT1                   IRQ15 = ATA 보조
19//! ```
20//!
21//! ## 왜 리매핑이 필요한가?
22//!
23//! 기본값: 마스터 IRQ0-7 → 인터럽트 벡터 0x08-0x0F
24//! 문제:   CPU 예외(#DE=0, #DF=8, #GP=13, #PF=14)와 충돌!
25//! 해결:   마스터 → 0x20-0x27, 슬레이브 → 0x28-0x2F 로 이동
26
27const PIC1_CMD:  u16 = 0x20;
28const PIC1_DATA: u16 = 0x21;
29const PIC2_CMD:  u16 = 0xA0;
30const PIC2_DATA: u16 = 0xA1;
31
32const PIC_EOI:   u8 = 0x20; // End-Of-Interrupt 명령
33
34// ICW = Initialization Command Word (초기화 명령어 순서)
35const ICW1_INIT: u8 = 0x11; // 엣지 트리거, 캐스케이드, ICW4 필요
36const ICW4_8086: u8 = 0x01; // 8086 모드 (8080 모드 아님)
37
38/// 마스터 PIC IRQ0-7 → 벡터 0x20-0x27
39pub const PIC1_OFFSET: u8 = 0x20;
40/// 슬레이브 PIC IRQ8-15 → 벡터 0x28-0x2F
41pub const PIC2_OFFSET: u8 = 0x28;
42
43fn outb(port: u16, val: u8) {
44    unsafe {
45        core::arch::asm!("out dx, al",
46            in("dx") port, in("al") val,
47            options(nomem, nostack, preserves_flags));
48    }
49}
50
51fn inb(port: u16) -> u8 {
52    let v: u8;
53    unsafe {
54        core::arch::asm!("in al, dx",
55            out("al") v, in("dx") port,
56            options(nomem, nostack, preserves_flags));
57    }
58    v
59}

61/// 짧은 I/O 지연 (포트 0x80 = POST 진단 포트)
62///
63/// 오래된 하드웨어에서 PIC 초기화 명령어 사이에 지연이 필요함.
64/// QEMU에서는 사실 불필요하지만, 실제 하드웨어 호환성을 위해 유지.
65fn io_wait() {
66    outb(0x80, 0);
67}
68
69/// 8259A PIC 초기화 및 IRQ 리매핑
70///
71/// 초기화 후 활성화되는 IRQ:
72/// - IRQ0 (타이머, 벡터 0x20)
73/// - IRQ1 (키보드, 벡터 0x21)
74/// 나머지는 모두 마스킹(비활성화).
75pub fn init() {
76    // 현재 마스크 저장 (필요하면 나중에 복원)
77    let _mask1 = inb(PIC1_DATA);
78    let _mask2 = inb(PIC2_DATA);
79
80    // ── ICW1: 초기화 시작 ─────────────────────────────────────────────
81    outb(PIC1_CMD,  ICW1_INIT); io_wait();
82    outb(PIC2_CMD,  ICW1_INIT); io_wait();
83
84    // ── ICW2: 벡터 오프셋 설정 ────────────────────────────────────────
85    outb(PIC1_DATA, PIC1_OFFSET); io_wait(); // 마스터: IRQ0-7 → 0x20-0x27
86    outb(PIC2_DATA, PIC2_OFFSET); io_wait(); // 슬레이브: IRQ8-15 → 0x28-0x2F
87
88    // ── ICW3: 캐스케이드 설정 ─────────────────────────────────────────
89    outb(PIC1_DATA, 0x04); io_wait(); // 마스터: IRQ2에 슬레이브 연결 (비트 2 = 1)
90    outb(PIC2_DATA, 0x02); io_wait(); // 슬레이브: 캐스케이드 ID = 2
91
92    // ── ICW4: 8086 모드 ───────────────────────────────────────────────
93    outb(PIC1_DATA, ICW4_8086); io_wait();
94    outb(PIC2_DATA, ICW4_8086); io_wait();
95
96    // ── 인터럽트 마스크 설정 ──────────────────────────────────────────
97    // 마스터 마스크: 0xF8 = 1111_1000b
98    //   bit0(IRQ0)=0 타이머, bit1(IRQ1)=0 키보드, bit2(IRQ2)=0 캐스케이드 활성화
99    //   IRQ2가 열려야 PIC2(슬레이브) IRQ8-15가 전달됨
100    // 슬레이브 마스크: 0xEF = 1110_1111b
101    //   bit4(IRQ12)=0 마우스 활성화, 나머지 차단
102    outb(PIC1_DATA, 0xF8);
103    outb(PIC2_DATA, 0xEF);
104
105    crate::serial_println!("[pic] 8259A initialized: IRQ0(timer) IRQ1(kbd) IRQ2(cascade) IRQ12(mouse)");
106}

108/// 마스터 PIC에 EOI 전송 (IRQ0-7 처리 후 호출)
109#[inline]
110pub fn eoi_master() {
111    outb(PIC1_CMD, PIC_EOI);
112}
113
114/// 슬레이브 PIC에 EOI 전송 (IRQ8-15 처리 후 호출)
115///
116/// 슬레이브 IRQ는 마스터 IRQ2를 통해 전달되므로,
117/// 슬레이브와 마스터 양쪽에 모두 EOI를 보내야 함.
118#[allow(dead_code)]
119#[inline]
120pub fn eoi_slave() {
121    outb(PIC2_CMD, PIC_EOI);
122    outb(PIC1_CMD, PIC_EOI);
123}
