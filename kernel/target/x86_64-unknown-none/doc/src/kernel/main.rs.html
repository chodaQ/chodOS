1//! 커널 진입점 (Kernel Entry Point)
2//!
3//! ## 부팅 순서 (Milestone ALPHA)
4//! 1. limine 부트로더 → _start() 호출
5//! 2. 시리얼 포트 초기화
6//! 3. 메모리 서브시스템 초기화 (프레임 할당자 + 힙)
7//! 4. 인터럽트 서브시스템 초기화 (GDT + IDT + PIC + STI)
8//! 5. 페이징 초기화 (커널 PML4 빌드, CR3 전환, TSS.RSP0 설정)
9//! 6. Policy Engine 초기화 (ALPHA M3)
10//! 7. IPC 데모 — 선점형 스케줄러 + 협력적 yield (ALPHA M1)
11//! 8. Zero-copy Capability IPC 데모 (ALPHA M2)
12//! 9. 선점형 스케줄러 데모 — yield 없는 태스크가 타이머에 의해 강제 전환
13//! 10. VFS 데모: tmpfs 마운트 → 파일/디렉토리 생성/읽기/열거 (M3.7)
14//! 11. ring3 데모: IRETQ 진입 → 유저 코드 `int 0x80` × 3 → longjmp 복귀 (M3.6)

64const PID_RECEIVER: process::Pid = 2;
65
66fn proc_sender() -> ! {
67    serial_println!("[sender pid=1] started");
68    let mut counter: u64 = 0;
69    loop {
70        if counter < 10 {
71            process::ipc::send_u64(PID_RECEIVER, counter);
72            counter += 1;
73        }
74        // yield_now() = int 0x40 → isr64 → voluntary_yield (ALPHA M1)
75        process::scheduler::yield_now();
76    }
77}
78
79fn proc_receiver() -> ! {
80    serial_println!("[receiver pid=2] started");
81    loop {
82        while let Some(msg) = process::ipc::recv() {
83            let value = process::ipc::msg_as_u64(&msg);
84            serial_println!("[receiver] <- pid={} | counter={}", msg.sender, value);
85        }
86        process::scheduler::yield_now();
87    }
88}

90// ==================== ALPHA M1: 선점형 스케줄러 데모 태스크 ====================
91//
92// 이 태스크들은 yield_now()를 호출하지 않는다.
93// 타이머 IRQ(TIME_SLICE=3틱 ≈ 165ms)가 강제로 컨텍스트 스위치를 일으킨다.
94// 타이머 ISR이 ALL 레지스터를 스택에 저장 → preempt_rsp 교환 → iretq로 복귀.
95
96fn preempt_task_a() -> ! {
97    serial_println!("[task_a] started — no yield, preempted by timer");
98    let mut i: u64 = 0;
99    loop {
100        i += 1;
101        // QEMU(에뮬레이션) 속도 기준 ~5백만 반복 ≈ 타임슬라이스 1-2회
102        if i % 5_000_000 == 0 {
103            serial_println!("[task_a] running... (loop #{})", i / 5_000_000);
104        }
105    }
106}
107
108fn preempt_task_b() -> ! {
109    serial_println!("[task_b] started — no yield, preempted by timer");
110    let mut i: u64 = 0;
111    loop {
112        i += 1;
113        if i % 5_000_000 == 0 {
114            serial_println!("[task_b] running... (loop #{})", i / 5_000_000);
115        }
116    }
117}

119// ==================== ALPHA 14: 내장 테스트 ELF 바이너리 ====================
120//
121// 손으로 조립한 최소 ELF64 실행 파일 (181 = 0xB5 bytes)
122//
123// ## 구조
124//
125// ```
126// Offset  Size  Content
127// ──────────────────────────────
128// 0x00    64    ELF64 헤더
129//               e_entry = 0x400078
130//               e_phoff = 0x40  (프로그램 헤더 시작)
131//               e_phnum = 1
132// 0x40    56    PT_LOAD 프로그램 헤더
133//               p_offset = 0     (파일 처음부터 매핑)
134//               p_vaddr  = 0x400000
135//               p_filesz = 0xB5  (181 bytes)
136//               p_memsz  = 0x1000 (1 페이지)
137//               p_align  = 0x1000
138// 0x78    61    코드 + 데이터 (vaddr 0x400078)
139// ──────────────────────────────
140// ```
141//
142// ## 코드 (vaddr 0x400078):
143//
144// ```asm
145// 0x400078:  EB 10          jmp +16   ; 메시지 건너뛰기
146// 0x40007A:  "Hello from ELF!\n"  ; 16 bytes (0x40007A..0x400089)
147// 0x40008A:  B8 01 00 00 00  mov eax, 1      ; SYS_write
148//            BF 01 00 00 00  mov edi, 1      ; fd = stdout
149//            48 8D 35 DF FF FF FF  lea rsi, [rip-0x21]   ; → 0x40007A (메시지)
150//              ; after lea: RIP = 0x40009B, disp = 0x40007A - 0x40009B = -0x21 ✓
151//            BA 10 00 00 00  mov edx, 16     ; length
152//            CD 80           int 0x80        ; → sys_write
153//            B8 27 00 00 00  mov eax, 39     ; SYS_getpid
154//            CD 80           int 0x80        ; → RAX = pid
155//            B8 3C 00 00 00  mov eax, 60     ; SYS_exit
156//            BF 00 00 00 00  mov edi, 0      ; exit code
157//            CD 80           int 0x80        ; → longjmp
158// ```
159// BETA 2-1: musl-static hello 바이너리로 교체 — Linux ABI 호환성 검증
160pub static ELF_TEST: &[u8] = include_bytes!(
161    concat!(env!("CARGO_MANIFEST_DIR"), "/../build/musl_hello.elf")

175#[no_mangle]
176pub extern "C" fn _start() -> ! {
177    assert!(BASE_REVISION.is_supported(), "limine revision not supported");
178
179    // ── 1. 시리얼 ────────────────────────────────────────────────────────
180    serial::init();
181    serial_println!("===========================================");
182    serial_println!("  MuKernel v0.1.0 - Milestone ALPHA");
183    serial_println!("  Preemptive Sched + ZeroCopy IPC + Policy");
184    serial_println!("===========================================");
185    if let Some(info) = BOOTLOADER_INFO.response() {
186        serial_println!("[boot] {} v{}", info.name(), info.version());
187    }
188
189    // ── 2. 메모리 ────────────────────────────────────────────────────────
190    let hhdm_offset = HHDM.response().expect("HHDM response missing").offset;
191    let memmap = MEMMAP.response().expect("memmap response missing").entries();
192    memory::init(hhdm_offset, memmap);
193
194    // ── 3. 인터럽트 (GDT→IDT→PIC→STI) ─────────────────────────────────
195    interrupts::init();
196
197    // ── 4. 페이징 ─────────────────────────────────────────────────────────
198    paging::init();
199
200    // ── 5. Policy Engine 초기화 (ALPHA M3) ───────────────────────────────
201    policy::init();
202
203    // ── 6. 선점형 스케줄러 + IPC 데모 ────────────────────────────────────
204    serial_println!("\n--- IPC Demo (선점형 스케줄러 적용) ---");
205    process::scheduler::init();
206
207    let sid = process::scheduler::alloc_pid(); // 1
208    process::scheduler::spawn(process::Process::new(sid, "sender", proc_sender));
209    let rid = process::scheduler::alloc_pid(); // 2
210    process::scheduler::spawn(process::Process::new(rid, "receiver", proc_receiver));
211
212    // yield_now()이 이제 int 0x40 → isr64 → voluntary_yield 경로를 탐
213    for _ in 0..30 {
214        process::scheduler::yield_now();
215    }
216
217    // IPC 데모 프로세스 종료 (Dead 표시 → 스케줄러가 건너뜀)
218    process::scheduler::kill_pid(sid);
219    process::scheduler::kill_pid(rid);
220    serial_println!("--- IPC Demo complete ---\n");
221
222    // ── 7. Zero-copy Capability IPC 데모 (ALPHA M2) ──────────────────────
223    serial_println!("===========================================");
224    serial_println!("  ALPHA M2: Zero-copy Capability IPC");
225    serial_println!("===========================================");
226
227    // 송신자: 공유 버퍼 할당 → 데이터 기록 → CapId 전송
228    {
229        use alloc::vec::Vec;
230        let sender_pid: process::Pid = 0; // kernel_main이 sender 역할
231        let receiver_pid: process::Pid = 99; // 가상 수신자 (데모용)
232
233        // 4KB 데이터를 공유 버퍼에 등록 (복사 없이 참조 전달)
234        let mut payload: Vec<u8> = Vec::with_capacity(4096);
235        for i in 0u8..=255 {
236            payload.extend_from_slice(&[i; 16]); // 256 × 16 = 4096 bytes
237        }
238        let data_size = payload.len();
239        let cap = process::ipc_cap::alloc_shared(sender_pid, payload);
240        serial_println!("[cap-ipc] alloc_shared: cap_id={}, size={} bytes", cap, data_size);
241
242        // append: 추가 데이터도 복사 없이 버퍼에 직접 추가
243        process::ipc_cap::append_shared(cap, b"[appended metadata]");
244        serial_println!("[cap-ipc] append_shared: +19 bytes");
245
246        // read: 수신자 입장 — cap_id로 데이터에 직접 접근 (복사 없음)
247        process::ipc_cap::read_shared(cap, |data| {
248            serial_println!("[cap-ipc] read_shared: {} bytes total", data.len());
249            serial_println!("[cap-ipc]   data[0]   = {:#04x}", data[0]);
250            serial_println!("[cap-ipc]   data[4095] = {:#04x}", data[4095]);
251            serial_println!("[cap-ipc]   tail = {:?}",
252                core::str::from_utf8(&data[data.len()-19..]).unwrap_or("?"));
253        });
254
255        // capability 소멸: 버퍼 해제 + 이후 접근 불가
256        let freed = process::ipc_cap::drop_shared(cap);
257        serial_println!("[cap-ipc] drop_shared: cap_id={} freed={}", cap, freed);
258        serial_println!("[cap-ipc] is_valid after drop: {}", process::ipc_cap::is_valid(cap));
259    }
260    serial_println!("--- Zero-copy IPC demo complete ---\n");
261
262    // ── 8. 선점형 스케줄러 데모 (ALPHA M1 핵심) ──────────────────────────
263    //
264    // task_a / task_b: yield_now() 호출 없이 무한 루프.
265    // 타이머 IRQ0이 ~165ms마다 강제로 컨텍스트 스위치를 일으킴.
266    // Policy Engine이 CPU 점유율을 실시간 추적 후 주기 리포트.
267    serial_println!("===========================================");
268    serial_println!("  ALPHA M1: 선점형 스케줄러 데모");
269    serial_println!("  (yield 없는 태스크가 타이머에 의해 강제 전환)");
270    serial_println!("===========================================");
271
272    let pa = process::scheduler::alloc_pid(); // 3
273    process::scheduler::spawn(process::Process::new(pa, "task_a", preempt_task_a));
274    let pb = process::scheduler::alloc_pid(); // 4
275    process::scheduler::spawn(process::Process::new(pb, "task_b", preempt_task_b));
276
277    // kernel_main은 HLT 대기 루프로 ~3초 대기.
278    // 이 동안 타이머 IRQ가 task_a / task_b / kernel_main을 라운드로빈으로 전환.
279    // HLT: 다음 인터럽트까지 CPU 슬립 → 타이머가 깨우면 틱 확인 후 재슬립.
280    let start_tick = interrupts::handlers::TICK.load(Ordering::Relaxed);
281    while interrupts::handlers::TICK.load(Ordering::Relaxed) - start_tick < 54 {
282        // ~3초 (54틱 × ~55ms/틱) 대기
283        unsafe { core::arch::asm!("hlt", options(nomem, nostack, preserves_flags)); }
284    }
285
286    // 태스크 종료 (Dead 표시)
287    process::scheduler::kill_pid(pa);
288    process::scheduler::kill_pid(pb);
289    serial_println!("[sched] task_a / task_b killed. Continuing...\n");
290
291    // ── 9. ext4 데모 (ALPHA 8) ────────────────────────────────────────────
292    serial_println!("===========================================");
293    serial_println!("  ALPHA 8: ext4 Disk Image");
294    serial_println!("===========================================");
295
296    // 커널 바이너리에 embed된 ext4 이미지 (빌드 시 Makefile이 생성)
297    static EXT4_IMAGE: &[u8] = include_bytes!(
298        concat!(env!("CARGO_MANIFEST_DIR"), "/../build/rootfs.ext4")
299    );
300
301    if vfs::mount_ext4(EXT4_IMAGE) {
302        // /etc/os-release 읽기
303        if let Some(data) = vfs::ext4_read_file("/etc/os-release") {
304            serial_println!("[ext4] /etc/os-release ({} bytes):", data.len());
305            for line in core::str::from_utf8(&data).unwrap_or("").lines() {
306                serial_println!("       {}", line);
307            }
308        }
309
310        // /etc/motd 읽기
311        if let Some(data) = vfs::ext4_read_file("/etc/motd") {
312            serial_println!("[ext4] /etc/motd: {:?}",
313                core::str::from_utf8(&data).unwrap_or("").trim_end());
314        }
315
316        // /var/log/boot.log 읽기
317        if let Some(data) = vfs::ext4_read_file("/var/log/boot.log") {
318            serial_println!("[ext4] /var/log/boot.log ({} bytes):", data.len());
319            for line in core::str::from_utf8(&data).unwrap_or("").lines() {
320                serial_println!("       {}", line);
321            }
322        }
323
324        // / 디렉토리 열거
325        serial_println!("[ext4] ls /:");
326        for entry in vfs::ext4_list_dir("/") {
327            serial_println!("       {}{}", entry.name,
328                if entry.is_dir { "/" } else { "" });
329        }
330
331        // /etc 디렉토리 열거
332        serial_println!("[ext4] ls /etc:");
333        for entry in vfs::ext4_list_dir("/etc") {
334            serial_println!("       {}{}", entry.name,
335                if entry.is_dir { "/" } else { "" });
336        }
337    }
338    serial_println!("--- ext4 demo complete ---\n");
339
340    // ── ALPHA 9: Capability Handle Table ─────────────────────────────────────
341    serial_println!("===========================================");
342    serial_println!("  ALPHA 9: Capability Handle Table");
343    serial_println!("  (Linux fd <-> Handle <-> Capability<T>)");
344    serial_println!("===========================================");
345    {
346        use process::handle::{Capability, Rights};
347
348        // 1. 읽기 전용 버퍼 Capability 등록
349        let buf_ro: alloc::vec::Vec<u8> = b"hello from capability".to_vec();
350        let h_ro = process::scheduler::insert_capability(
351            Capability::new(buf_ro, Rights::READ),
352        );
353        serial_println!("[cap9] insert fd={} rights={:?}", h_ro.id, h_ro.rights);
354
355        // 2. 읽기+쓰기 Capability 등록
356        let buf_rw: alloc::vec::Vec<u8> = b"writable buffer".to_vec();
357        let h_rw = process::scheduler::insert_capability(
358            Capability::new(buf_rw, Rights::READ | Rights::WRITE),
359        );
360        serial_println!("[cap9] insert fd={} rights={:?}", h_rw.id, h_rw.rights);
361
362        // 3. 읽기 조회 (성공)
363        if let Some(data) = process::scheduler::get_capability::<alloc::vec::Vec<u8>>(
364            h_ro.id, Rights::READ,
365        ) {
366            serial_println!("[cap9] read fd={}: {:?}",
367                h_ro.id, core::str::from_utf8(&data).unwrap_or("?"));
368        }
369
370        // 4. 읽기 전용 핸들에 쓰기 시도 (권한 거부)
371        let denied = process::scheduler::get_capability::<alloc::vec::Vec<u8>>(
372            h_ro.id, Rights::WRITE,
373        ).is_none();
374        serial_println!("[cap9] write on read-only fd={}: {}",
375            h_ro.id, if denied { "denied ✓" } else { "BUG: should be denied!" });
376
377        // 5. 타입 불일치 조회 (거부)
378        let type_mismatch = process::scheduler::get_capability::<u64>(
379            h_ro.id, Rights::READ,
380        ).is_none();
381        serial_println!("[cap9] wrong-type get fd={}: {}",
382            h_ro.id, if type_mismatch { "denied ✓" } else { "BUG!" });
383
384        // 6. 핸들 목록 출력
385        for (id, rights) in process::scheduler::list_handles() {
386            serial_println!("[cap9]   open fd={} rights={:?}", id, rights);
387        }
388        serial_println!("[cap9] open handles: {}", process::scheduler::handle_count());
389
390        // 7. 핸들 닫기
391        process::scheduler::close_handle(h_ro.id);
392        process::scheduler::close_handle(h_rw.id);
393        serial_println!("[cap9] after close: {} handles remain",
394            process::scheduler::handle_count());
395    }
396    serial_println!("--- ALPHA 9 demo complete ---\n");
397
398    // ── ALPHA 10: VirtIO Block 드라이버 ──────────────────────────────────────
399    serial_println!("===========================================");
400    serial_println!("  ALPHA 10: VirtIO Block Driver");
401    serial_println!("===========================================");
402    if let Some(mut blk) = virtio::VirtioBlk::init() {
403        // 섹터 0 읽기 (MBR / 부트 섹터)
404        let mut sector0 = [0u8; 512];
405        if blk.read_sector(0, &mut sector0) {
406            serial_println!("[virtio-blk] sector 0 read OK ({} bytes)", sector0.len());
407            // 처음 64바이트를 16진수로 출력
408            for row in 0..4 {
409                let off = row * 16;
410                serial_println!(
411                    "[virtio-blk]   {:04x}: {:02x} {:02x} {:02x} {:02x}  {:02x} {:02x} {:02x} {:02x}  {:02x} {:02x} {:02x} {:02x}  {:02x} {:02x} {:02x} {:02x}",
412                    off,
413                    sector0[off+0],  sector0[off+1],  sector0[off+2],  sector0[off+3],
414                    sector0[off+4],  sector0[off+5],  sector0[off+6],  sector0[off+7],
415                    sector0[off+8],  sector0[off+9],  sector0[off+10], sector0[off+11],
416                    sector0[off+12], sector0[off+13], sector0[off+14], sector0[off+15],
417                );
418            }
419            // ASCII 헤더 출력 (텍스트 포함 시)
420            let text = core::str::from_utf8(&sector0[..64]).unwrap_or("");
421            let printable: alloc::string::String = text.chars()
422                .map(|c| if c.is_ascii_graphic() || c == ' ' { c } else { '.' })
423                .collect();
424            serial_println!("[virtio-blk]   ascii: {}", &printable[..printable.len().min(48)]);
425        } else {
426            serial_println!("[virtio-blk] sector 0 read FAILED");
427        }
428
429        // 섹터 1 쓰기 후 다시 읽기 (왕복 검증)
430        let mut write_buf = [0u8; 512];
431        let magic = b"MuKernel VirtIO Block OK";
432        write_buf[..magic.len()].copy_from_slice(magic);
433        write_buf[510] = 0xAA;
434        write_buf[511] = 0x55;
435        if blk.write_sector(1, &write_buf) {
436            let mut read_buf = [0u8; 512];
437            if blk.read_sector(1, &mut read_buf) && read_buf[..magic.len()] == *magic {
438                serial_println!("[virtio-blk] write→read roundtrip OK: {:?}",
439                    core::str::from_utf8(&read_buf[..magic.len()]).unwrap_or("?"));
440            } else {
441                serial_println!("[virtio-blk] roundtrip MISMATCH");
442            }
443        }
444    } else {
445        serial_println!("[virtio-blk] device not found — QEMU disk 없음");
446        serial_println!("  (Makefile의 DISK_IMG 타겟과 run 타겟에 -device virtio-blk-pci 필요)");
447    }
448    serial_println!("--- ALPHA 10 demo complete ---\n");
449
450    // ── ALPHA 11: VirtIO Net 드라이버 ────────────────────────────────────────
451    serial_println!("===========================================");
452    serial_println!("  ALPHA 11: VirtIO Net Driver");
453    serial_println!("  (ARP request to QEMU SLIRP gateway)");
454    serial_println!("===========================================");
455
456    if let Some(mut nic) = virtio::VirtioNet::init() {
457        let mac = nic.mac;
458        serial_println!(
459            "[virtio-net] MAC = {:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
460            mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]
461        );
462
463        // ARP request: who has 10.0.2.2? tell 10.0.2.15
464        let src_ip: [u8; 4] = [10, 0, 2, 15];
465        let dst_ip: [u8; 4] = [10, 0, 2,  2];
466        let arp_frame = virtio::net::build_arp_request(mac, src_ip, dst_ip);
467
468        serial_println!(
469            "[virtio-net] Sending ARP request: who has {}.{}.{}.{}? tell {}.{}.{}.{}",
470            dst_ip[0], dst_ip[1], dst_ip[2], dst_ip[3],
471            src_ip[0], src_ip[1], src_ip[2], src_ip[3],
472        );
473
474        if nic.send(&arp_frame) {
475            serial_println!("[virtio-net] ARP frame sent OK");
476        } else {
477            serial_println!("[virtio-net] ARP send timeout");
478        }
479
480        // RX 폴링: ~1초 안에 ARP reply 기다림
481        let poll_start = interrupts::handlers::TICK.load(Ordering::Relaxed);
482        let mut got_reply = false;
483        while interrupts::handlers::TICK.load(Ordering::Relaxed) - poll_start < 18 {
484            if let Some(frame) = nic.try_recv() {
485                let etype = virtio::net::ethertype(&frame);
486                serial_println!("[virtio-net] RX {} bytes, ethertype=0x{:04x}", frame.len(), etype);
487
488                if let Some((reply_mac, reply_ip)) = virtio::net::parse_arp_reply(&frame) {
489                    serial_println!(
490                        "[virtio-net] ARP reply: {}.{}.{}.{} is at {:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
491                        reply_ip[0], reply_ip[1], reply_ip[2], reply_ip[3],
492                        reply_mac[0], reply_mac[1], reply_mac[2], reply_mac[3], reply_mac[4], reply_mac[5],
493                    );
494                    got_reply = true;
495                    break;
496                }
497            }
498            unsafe { core::arch::asm!("pause", options(nomem, nostack)); }
499        }
500
501        if !got_reply {
502            serial_println!("[virtio-net] no ARP reply received (timeout ~1s)");
503        }
504    } else {
505        serial_println!("[virtio-net] device not found — QEMU NIC 없음");
506        serial_println!("  (Makefile run 타겟에 -netdev user + -device virtio-net-pci 필요)");
507    }
508    serial_println!("--- ALPHA 11 demo complete ---\n");
509
510    // ── ALPHA 12: TCP/IP 네트워크 스택 ───────────────────────────────────────
511    serial_println!("===========================================");
512    serial_println!("  ALPHA 12: TCP/IP Network Stack");
513    serial_println!("  (ARP table + IPv4 + ICMP ping + UDP)");
514    serial_println!("===========================================");
515
516    if let Some(nic) = virtio::VirtioNet::init() {
517        let guest_ip = [10u8, 0, 2, 15];
518        let gateway  = [10u8, 0, 2,  2];
519        let dns      = [10u8, 0, 2,  3];
520
521        let mut stack = net::NetworkStack::new(nic, guest_ip);
522        serial_println!(
523            "[net] stack up: IP={}.{}.{}.{}  MAC={:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
524            guest_ip[0], guest_ip[1], guest_ip[2], guest_ip[3],
525            stack.mac[0], stack.mac[1], stack.mac[2],
526            stack.mac[3], stack.mac[4], stack.mac[5],
527        );
528
529        // 1. ARP resolve gateway
530        serial_println!("[net] ARP resolve {}.{}.{}.{}...", gateway[0], gateway[1], gateway[2], gateway[3]);
531        if let Some(gw_mac) = stack.arp_resolve(gateway) {
532            serial_println!(
533                "[net] gateway MAC = {:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
534                gw_mac[0], gw_mac[1], gw_mac[2], gw_mac[3], gw_mac[4], gw_mac[5],
535            );
536        } else {
537            serial_println!("[net] ARP resolve timeout");
538        }
539
540        // 2. ICMP ping gateway (3회)
541        for i in 0..3u16 {
542            let t0 = interrupts::handlers::TICK.load(Ordering::Relaxed);
543            match stack.ping(gateway) {
544                Some(seq) => {
545                    let rtt = interrupts::handlers::TICK.load(Ordering::Relaxed) - t0;
546                    serial_println!(
547                        "[net] ping {}.{}.{}.{}: reply seq={} rtt=~{}ms",
548                        gateway[0], gateway[1], gateway[2], gateway[3],
549                        seq, rtt * 55
550                    );
551                }
552                None => serial_println!("[net] ping seq={} timeout", i),
553            }
554        }
555
556        // 3. UDP 송신 (DNS query to 10.0.2.3:53 — "mukernel\x00" A record lookup)
557        //
558        // 최소 DNS query 패킷 (www.example.com A record 요청 형식)
559        // SLIRP가 응답하지 않아도 TX 성공이 목표.
560        let dns_query: &[u8] = &[
561            0x00, 0x01,  // Transaction ID
562            0x01, 0x00,  // Flags: standard query
563            0x00, 0x01,  // QDCOUNT = 1
564            0x00, 0x00,  // ANCOUNT = 0
565            0x00, 0x00,  // NSCOUNT = 0
566            0x00, 0x00,  // ARCOUNT = 0
567            // QNAME: "mukernel\x00"
568            0x08, b'm', b'u', b'k', b'e', b'r', b'n', b'e', b'l',
569            0x00,        // root label
570            0x00, 0x01,  // QTYPE = A
571            0x00, 0x01,  // QCLASS = IN
572        ];
573        let sent = stack.udp_send(dns, 53, 1024, dns_query);
574        serial_println!("[net] UDP DNS query to {}.{}.{}.{}:53 — tx={}",
575            dns[0], dns[1], dns[2], dns[3], if sent { "OK" } else { "FAIL" });
576
577        // 4. RX poll ~500ms: SLIRP DNS 응답 기다리기
578        let poll_end = interrupts::handlers::TICK.load(Ordering::Relaxed) + 9;
579        while interrupts::handlers::TICK.load(Ordering::Relaxed) < poll_end {
580            if let Some(pkt) = stack.poll() {
581                match pkt {
582                    net::Packet::Udp { src, src_port, dst_port, data } => {
583                        serial_println!(
584                            "[net] UDP RX from {}.{}.{}.{}:{} → port {} ({} bytes)",
585                            src[0], src[1], src[2], src[3], src_port, dst_port, data.len()
586                        );
587                    }
588                    net::Packet::IcmpEchoReply { src, id, seq } => {
589                        serial_println!(
590                            "[net] ICMP reply from {}.{}.{}.{} id={:#x} seq={}",
591                            src[0], src[1], src[2], src[3], id, seq
592                        );
593                    }
594                }
595            }
596        }
597    } else {
598        serial_println!("[net] VirtioNet not available");
599    }
600    serial_println!("--- ALPHA 12 demo complete ---\n");
601
602    // ── ALPHA 17: GUI / Window System ─────────────────────────────────────────
603    serial_println!("===========================================");
604    serial_println!("  ALPHA 17: GUI / Window System");
605    serial_println!("  (Limine Framebuffer + 8x8 Font + MuWM)");
606    serial_println!("===========================================");
607
608    if let Some(fb_resp) = FB_REQ.response() {
609        let fbs = fb_resp.framebuffers();
610        if let Some(fb) = fbs.first() {
611            let addr   = fb.address() as *mut u8 as usize;
612            let width  = fb.width  as u32;
613            let height = fb.height as u32;
614            let pitch  = fb.pitch  as u32;
615            let r_sh   = fb.red_mask_shift;
616            let g_sh   = fb.green_mask_shift;
617            let b_sh   = fb.blue_mask_shift;
618            serial_println!("[fb] {}x{} pitch={} bpp={} r_sh={} g_sh={} b_sh={}",
619                width, height, pitch, fb.bpp, r_sh, g_sh, b_sh);
620            fb::init(addr, width, height, pitch, r_sh, g_sh, b_sh);
621            wm::render_desktop();
622            serial_println!("[fb] desktop rendered — {} windows + taskbar", 3);
623
624            // PS/2 마우스 초기화 + 초기 커서 렌더링
625            mouse::init();
626            let (mx, my) = (
627                mouse::MOUSE_X.load(core::sync::atomic::Ordering::Relaxed),
628                mouse::MOUSE_Y.load(core::sync::atomic::Ordering::Relaxed),
629            );
630            fb::draw_cursor(mx, my);
631        } else {
632            serial_println!("[fb] no framebuffer in response");
633        }
634    } else {
635        serial_println!("[fb] FramebufferRequest not responded");
636        serial_println!("     GUI requires display: use 'make run-gui'");
637    }
638    serial_println!("--- ALPHA 17 demo complete ---\n");
639
640    // ── 10. VFS 데모 (Milestone 3.7) ─────────────────────────────────────
641    serial_println!("===========================================");
642    serial_println!("  Milestone 3.7: VFS + tmpfs");
643    serial_println!("===========================================");
644
645    vfs::init();
646    vfs::mkdir("/etc").expect("mkdir /etc");
647    vfs::mkdir("/var").expect("mkdir /var");
648    vfs::mkdir("/var/log").expect("mkdir /var/log");
649    vfs::create_file("/etc/hostname").expect("create /etc/hostname");
650    vfs::write_file("/etc/hostname", b"mukernel\n");
651    vfs::create_file("/etc/version").expect("create /etc/version");
652    vfs::write_file("/etc/version", b"MuKernel v0.1.0-alpha\n");
653    vfs::create_file("/var/log/kernel.log").expect("create /var/log/kernel.log");
654    vfs::write_file("/var/log/kernel.log", b"[boot] kernel started\n");
655    vfs::append_file("/var/log/kernel.log", b"[alpha] preemptive scheduler active\n");
656    vfs::append_file("/var/log/kernel.log", b"[alpha] zero-copy IPC ready\n");
657
658    let hostname = vfs::read_file("/etc/hostname").expect("read hostname");
659    serial_println!("[vfs] /etc/hostname = {:?}",
660        core::str::from_utf8(&hostname).unwrap_or("").trim_end());
661
662    let log = vfs::read_file("/var/log/kernel.log").expect("read kernel.log");
663    serial_println!("[vfs] /var/log/kernel.log ({} bytes):", log.len());
664    for line in core::str::from_utf8(&log).unwrap_or("").lines() {
665        serial_println!("       {}", line);
666    }
667
668    serial_println!("[vfs] ls /:");
669    for entry in vfs::list_dir("/") {
670        serial_println!("       {}{}", entry.name, if entry.is_dir { "/" } else { "" });
671    }
672    serial_println!("--- VFS demo complete ---\n");
673
674    // ── ALPHA 13 → 14 연속 데모 ──────────────────────────────────────────────
675    //
676    // ALPHA 13: raw 머신 코드를 ring3에서 실행 → syscall table 검증
677    // ALPHA 14: 실제 ELF64 바이너리를 ring3에서 실행 → ELF 로더 검증
678    //
679    // 두 데모는 sys_exit longjmp → after_user_demo → enter_elf 체인으로 연결됨.
680    serial_println!("===========================================");
681    serial_println!("  ALPHA 13: Linux Compat Tier 1");
682    serial_println!("  (write / getpid / mmap / exit via int 0x80)");
683    serial_println!("  → after_user_demo → ALPHA 14 ELF Loader");
684    serial_println!("===========================================");
685
686    // ── ring3 머신 코드 레이아웃 ─────────────────────────────────────────────
687    //
688    // [0x00] EB 12         jmp short → [0x14]  (skip 18-byte message)
689    // [0x02..0x13]         "Hello from ring3!\n"  (18 bytes)
690    // [0x14] code:
691    //   B8 01 00 00 00     mov eax, 1       (SYS_write)
692    //   BF 01 00 00 00     mov edi, 1       (fd = stdout)
693    //   48 8D 35 DD FF FF FF  lea rsi, [rip-0x23]  → [0x02] = 메시지
694    //     (RIP 계산: 인스트럭션 뒤 RIP = 0x10000+0x25, 메시지 = 0x10000+0x02
695    //      disp32 = 0x02 - 0x25 = -0x23 = 0xFFFFFFDD ✓)
696    //   BA 12 00 00 00     mov edx, 18      (length)
697    //   CD 80              int 0x80         → sys_write → "Hello from ring3!\n"
698    //
699    //   B8 27 00 00 00     mov eax, 39      (SYS_getpid)
700    //   CD 80              int 0x80         → RAX = current PID
701    //
702    //   B8 09 00 00 00     mov eax, 9       (SYS_mmap)
703    //   BF 00 00 00 00     mov edi, 0       (addr hint = NULL)
704    //   BE 00 10 00 00     mov esi, 0x1000  (len = 4096)
705    //   BA 03 00 00 00     mov edx, 3       (prot = PROT_READ|PROT_WRITE)
706    //   41 BA 22 00 00 00  mov r10d, 0x22   (flags = MAP_PRIVATE|MAP_ANONYMOUS)
707    //   45 31 C0           xor r8d, r8d     (fd = 0)  ← need -1: use 4D31C0+dec
708    //   49 83 C8 FF        or r8, -1        (fd = -1)
709    //   4D 31 C9           xor r9, r9       (off = 0)
710    //   CD 80              int 0x80         → RAX = mapped address
711    //
712    //   B8 3C 00 00 00     mov eax, 60      (SYS_exit)
713    //   BF 00 00 00 00     mov edi, 0       (exit code = 0)
714    //   CD 80              int 0x80         → longjmp → after_user_demo
715
716    let user_code: &[u8] = &[
717        // [0x00] jmp to [0x14]
718        0xEB, 0x12,
719        // [0x02] "Hello from ring3!\n" (18 bytes)
720        b'H', b'e', b'l', b'l', b'o', b' ',
721        b'f', b'r', b'o', b'm', b' ',
722        b'r', b'i', b'n', b'g', b'3', b'!', b'\n',
723        // [0x14] --- code ---
724        // sys_write(1, &msg, 18)
725        0xB8, 0x01, 0x00, 0x00, 0x00,        // mov eax, 1
726        0xBF, 0x01, 0x00, 0x00, 0x00,        // mov edi, 1
727        0x48, 0x8D, 0x35, 0xDD, 0xFF, 0xFF, 0xFF, // lea rsi, [rip-0x23]
728        0xBA, 0x12, 0x00, 0x00, 0x00,        // mov edx, 18
729        0xCD, 0x80,                           // int 0x80
730        // sys_getpid()
731        0xB8, 0x27, 0x00, 0x00, 0x00,        // mov eax, 39
732        0xCD, 0x80,                           // int 0x80  (RAX = pid)
733        // sys_mmap(0, 0x1000, PROT_RW, MAP_ANON|MAP_PRIVATE, -1, 0)
734        0xB8, 0x09, 0x00, 0x00, 0x00,        // mov eax, 9
735        0xBF, 0x00, 0x00, 0x00, 0x00,        // mov edi, 0
736        0xBE, 0x00, 0x10, 0x00, 0x00,        // mov esi, 0x1000
737        0xBA, 0x03, 0x00, 0x00, 0x00,        // mov edx, 3
738        0x41, 0xBA, 0x22, 0x00, 0x00, 0x00,  // mov r10d, 0x22
739        0x49, 0x83, 0xC8, 0xFF,              // or r8, -1  (fd = -1)
740        0x4D, 0x31, 0xC9,                    // xor r9, r9
741        0xCD, 0x80,                           // int 0x80  (RAX = mapped ptr)
742        // sys_exit(0)
743        0xB8, 0x3C, 0x00, 0x00, 0x00,        // mov eax, 60
744        0xBF, 0x00, 0x00, 0x00, 0x00,        // mov edi, 0
745        0xCD, 0x80,                           // int 0x80  (noreturn)
746    ];
747
748    serial_println!("[alpha13] entering ring3 ({} bytes of user code)", user_code.len());
749    unsafe { paging::enter_user_demo(user_code); }
750}

754#[panic_handler]
755fn panic(info: &core::panic::PanicInfo) -> ! {
756    serial::init();
757    serial_println!("\n!!! KERNEL PANIC !!!");
758    serial_println!("{}", info);
759    loop {
760        unsafe { core::arch::asm!("cli; hlt", options(nomem, nostack, preserves_flags)); }
761    }
762}

764#[alloc_error_handler]
765fn alloc_error(layout: core::alloc::Layout) -> ! {
766    panic!("OOM: size={} align={}", layout.size(), layout.align());
