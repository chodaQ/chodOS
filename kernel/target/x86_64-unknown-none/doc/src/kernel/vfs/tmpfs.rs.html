1//! tmpfs — 메모리 기반 파일시스템 구현
2//!
3//! ## 구조
4//!
5//! ```
6//! VfsNode::Dir(TmpfsDir)
7//!   └── children: BTreeMap<String, NodeRef>
8//!         ├── "etc" → VfsNode::Dir(TmpfsDir)
9//!         │     └── "hosts" → VfsNode::File(Vec<u8>)
10//!         └── "hello.txt" → VfsNode::File(Vec<u8>)
11//! ```
12//!
13//! ## 특징
14//!
15//! - 모든 데이터는 커널 힙(`alloc`)에 저장
16//! - 시스템 리셋(재부팅)시 소멸 — 비휘발성 저장소 없음
17//! - `BTreeMap`: 삽입 순서 독립적, 이름 순 정렬 열거 보장
18//! - 중첩 깊이 제한 없음 (힙 크기에만 의존)
19
20use alloc::{
21    collections::BTreeMap,
22    string::String,
23    sync::Arc,
24    vec::Vec,
25};
26use spin::Mutex;

32// TmpfsFile은 별도 구조체가 필요 없음:
33// VfsNode::File(Vec<u8>) 자체가 파일 데이터를 보유.

35// ── 디렉토리 노드 ────────────────────────────────────────────────────────────

37/// tmpfs 디렉토리.
38///
39/// `children`: 항목 이름 → 자식 `NodeRef` 매핑.
40/// BTreeMap이므로 항상 이름 순(오름차순)으로 열거됨.
41pub struct TmpfsDir {
42    pub children: BTreeMap<String, NodeRef>,
43}
44
45impl TmpfsDir {
46    pub fn new() -> Self {
47        Self { children: BTreeMap::new() }
48    }
49
50    /// 이름으로 자식 노드 검색.
51    ///
52    /// 경로 컴포넌트 탐색에 사용 (예: `"etc"` 검색 → 재귀적으로 다음 컴포넌트 탐색).
53    pub fn lookup(&self, name: &str) -> Option<NodeRef> {
54        self.children.get(name).cloned()
55    }
56
57    /// 빈 파일 노드를 생성하고 등록.
58    ///
59    /// 이미 같은 이름이 있으면 기존 노드를 그대로 반환(덮어쓰지 않음).
60    /// 이렇게 하면 `create_file` 후 바로 `write_file`해도 같은 노드를 수정.
61    pub fn create_file(&mut self, name: &str) -> NodeRef {
62        self.children
63            .entry(name.into())
64            .or_insert_with(|| Arc::new(Mutex::new(VfsNode::File(Vec::new()))))
65            .clone()
66    }
67
68    /// 빈 디렉토리 노드를 생성하고 등록.
69    ///
70    /// 이미 같은 이름이 있으면 기존 노드 반환.
71    pub fn create_dir(&mut self, name: &str) -> NodeRef {
72        self.children
73            .entry(name.into())
74            .or_insert_with(|| Arc::new(Mutex::new(VfsNode::Dir(TmpfsDir::new()))))
75            .clone()
76    }
77
78    /// 디렉토리 내용을 `(이름, is_dir, size)` 목록으로 반환.
79    ///
80    /// BTreeMap 이므로 이름 오름차순 정렬이 보장됨.
81    /// `size`: 파일이면 바이트 수, 디렉토리면 0.
82    pub fn list(&self) -> Vec<(String, bool, usize)> {
83        self.children
84            .iter()
85            .map(|(name, node_ref)| {
86                let node = node_ref.lock();
87                let (is_dir, size) = match &*node {
88                    VfsNode::File(data) => (false, data.len()),
89                    VfsNode::Dir(_)     => (true,  0),
90                };
91                (name.clone(), is_dir, size)
92            })
93            .collect()
94    }
95}
96
97// ── 팩토리 ──────────────────────────────────────────────────────────────────

99/// 새 빈 tmpfs 루트 디렉토리(`/`)를 생성.
100///
101/// `vfs::init()`이 이것을 글로벌 마운트 포인트로 설치.
102pub fn new_root() -> NodeRef {
103    Arc::new(Mutex::new(VfsNode::Dir(TmpfsDir::new())))
104}
