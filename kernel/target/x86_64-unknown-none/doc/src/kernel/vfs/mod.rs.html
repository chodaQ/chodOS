1//! VFS (Virtual File System) — 가상 파일시스템 계층
2//!
3//! ## 설계 개요
4//!
5//! ```
6//!  커널 코드 / 시스템콜
7//!       │
8//!       ▼
9//!  ┌─────────────────────────────────┐
10//!  │         VFS 인터페이스          │  ← 이 모듈 (경로 파싱, 마운트 테이블)
11//!  └─────────────────────────────────┘
12//!       │
13//!       ▼
14//!  ┌─────────────────────────────────┐
15//!  │   tmpfs 구현체 (vfs/tmpfs.rs)   │  ← 메모리 기반 파일시스템
16//!  └─────────────────────────────────┘
17//! ```
18//!
19//! ## 핵심 타입
20//!
21//! - `NodeRef`: `Arc<Mutex<VfsNode>>` — VFS 노드의 공유 가변 참조
22//! - `VfsNode`: File(Vec<u8>) 또는 Dir(BTreeMap) 열거형
23//!
24//! ## 경로 규칙
25//!
26//! - 항상 `/`로 시작하는 절대 경로
27//! - 예: `/etc/config`, `/home/user/data.txt`
28//! - 루트는 `/`
29//!
30//! ## 현재 제한
31//!
32//! - 마운트 포인트 1개 (`/` = tmpfs)
33//! - 동기 인터페이스 (비동기 I/O 없음)
34//! - 파일 권한/소유자 없음 (Milestone 4에서 추가 예정)
35

50/// VFS 노드의 공유 가변 참조.
51///
52/// `Arc`: 여러 경로가 같은 노드를 참조할 수 있음 (하드링크 기반).
53/// `Mutex`: 멀티코어/인터럽트 안전 — 노드 수정 시 잠금 필요.
54pub type NodeRef = Arc<Mutex<VfsNode>>;
55
56/// VFS 노드 — 파일 또는 디렉토리
57pub enum VfsNode {
58    /// 일반 파일: 바이트 시퀀스
59    File(Vec<u8>),
60    /// 디렉토리: 이름 → 자식 노드 매핑
61    Dir(TmpfsDir),
62}
63
64/// 디렉토리 열거(`list_dir`) 결과의 한 항목
65pub struct DirEntry {
66    /// 파일명 (경로 제외)
67    pub name: String,
68    /// true = 디렉토리, false = 파일
69    pub is_dir: bool,
70    /// 파일 크기 (바이트), 디렉토리는 0
71    pub size: usize,
72}

74// ── 글로벌 루트 ─────────────────────────────────────────────────────────────

76/// 루트 파일시스템 마운트 포인트 (`/`) — tmpfs
77static ROOT: Mutex<Option<NodeRef>> = Mutex::new(None);
78
79/// ext4-view의 `Ext4`는 내부에서 `Rc`를 사용해 `!Send`.
80/// 단일 코어 커널이므로 실제 데이터 경쟁은 없음 — Mutex 보호로 충분.
81struct SendableExt4(ext4fs::Ext4Fs);
82unsafe impl Send for SendableExt4 {}
83
84/// ext4 이미지 마운트 (ALPHA 8)
85static EXT4: Mutex<Option<SendableExt4>> = Mutex::new(None);
86
87// ── 초기화 ──────────────────────────────────────────────────────────────────
88
89/// VFS 초기화 — tmpfs를 루트(`/`)에 마운트
90pub fn init() {
91    let root = tmpfs::new_root();
92    *ROOT.lock() = Some(root);
93    crate::serial_println!("[vfs] tmpfs mounted at /");
94}
95
96/// ext4 이미지를 마운트 (ALPHA 8).
97///
98/// `image`: 커널에 embed된 ext4 파티션 이미지 (`include_bytes!`로 로드).
99/// 성공 시 `ext4_read_file` / `ext4_list_dir`로 접근 가능.
100pub fn mount_ext4(image: &'static [u8]) -> bool {
101    match ext4fs::Ext4Fs::new(image) {
102        Ok(fs) => {
103            *EXT4.lock() = Some(SendableExt4(fs));
104            crate::serial_println!("[vfs] ext4 mounted ({} bytes image)", image.len());
105            true
106        }
107        Err(e) => {
108            crate::serial_println!("[vfs] ext4 mount failed: {:?}", e);
109            false
110        }
111    }
112}
113
114/// ext4에서 파일을 읽어 반환.
115pub fn ext4_read_file(path: &str) -> Option<Vec<u8>> {
116    EXT4.lock().as_ref().map(|s| s.0.read_file(path))?
117}
118
119/// ext4 디렉토리 항목 열거.
120pub fn ext4_list_dir(path: &str) -> Vec<DirEntry> {
121    EXT4.lock().as_ref()
122        .map(|s| s.0.list_dir(path))
123        .unwrap_or_default()
124}
125
126/// ext4 경로 존재 여부 확인.
127pub fn ext4_exists(path: &str) -> bool {
128    EXT4.lock().as_ref().map(|s| s.0.exists(path)).unwrap_or(false)
129}

131// ── 경로 탐색 ───────────────────────────────────────────────────────────────
132
133/// 절대 경로로 VFS 노드를 검색.
134///
135/// 경로 예시: `/`, `/etc`, `/etc/hosts`
136///
137/// 반환: 찾으면 `Some(NodeRef)`, 없으면 `None`
138pub fn lookup(path: &str) -> Option<NodeRef> {
139    let root = ROOT.lock().clone()?;
140
141    // 루트 자체 요청
142    if path == "/" {
143        return Some(root);
144    }
145
146    // `/a/b/c` → ["a", "b", "c"] (빈 항목 제거: 선행 / 처리)
147    let parts: Vec<&str> = path
148        .trim_start_matches('/')
149        .split('/')
150        .filter(|s| !s.is_empty())
151        .collect();
152
153    let mut current = root;
154    for part in parts {
155        let next = {
156            let node = current.lock();
157            match &*node {
158                VfsNode::Dir(dir) => dir.lookup(part),
159                VfsNode::File(_) => return None, // 파일 아래에는 경로가 없음
160            }
161        };
162        current = next?;
163    }
164    Some(current)
165}

167/// 부모 디렉토리 경로와 파일명을 분리.
168///
169/// 예: `/etc/hosts` → (`/etc`, `hosts`)
170///     `/foo`       → (`/`,   `foo`)
171fn split_parent(path: &str) -> Option<(String, String)> {
172    let path = path.trim_end_matches('/');
173    let slash = path.rfind('/')?;
174
175    let parent = if slash == 0 { "/".to_string() } else { path[..slash].to_string() };
176    let name = path[slash + 1..].to_string();
177
178    if name.is_empty() { return None; }
179    Some((parent, name))
180}

182// ── 파일 및 디렉토리 생성 ────────────────────────────────────────────────────
183
184/// 경로에 빈 파일을 생성하고 `NodeRef`를 반환.
185///
186/// 부모 디렉토리가 없으면 `None`.
187/// 이미 같은 이름이 존재하면 기존 노드를 반환(덮어쓰기 않음).
188pub fn create_file(path: &str) -> Option<NodeRef> {
189    let (parent_path, name) = split_parent(path)?;
190    let parent = lookup(&parent_path)?;
191    let mut parent_node = parent.lock();
192    match &mut *parent_node {
193        VfsNode::Dir(dir) => Some(dir.create_file(&name)),
194        VfsNode::File(_) => None,
195    }
196}
197
198/// 경로에 디렉토리를 생성하고 `NodeRef`를 반환.
199pub fn mkdir(path: &str) -> Option<NodeRef> {
200    let (parent_path, name) = split_parent(path)?;
201    let parent = lookup(&parent_path)?;
202    let mut parent_node = parent.lock();
203    match &mut *parent_node {
204        VfsNode::Dir(dir) => Some(dir.create_dir(&name)),
205        VfsNode::File(_) => None,
206    }
207}

209// ── 읽기 / 쓰기 ─────────────────────────────────────────────────────────────
210
211/// 파일 전체 내용을 `Vec<u8>`으로 읽어 반환.
212///
213/// 경로가 없거나 파일이 아니면 `None`.
214pub fn read_file(path: &str) -> Option<Vec<u8>> {
215    let node_ref = lookup(path)?;
216    let node = node_ref.lock();
217    match &*node {
218        VfsNode::File(data) => Some(data.clone()),
219        VfsNode::Dir(_) => None,
220    }
221}
222
223/// 파일에 데이터를 (덮어)쓰기.
224///
225/// 경로가 없거나 디렉토리이면 false.
226pub fn write_file(path: &str, data: &[u8]) -> bool {
227    let node_ref = match lookup(path) {
228        Some(n) => n,
229        None => return false,
230    };
231    let mut node = node_ref.lock();
232    match &mut *node {
233        VfsNode::File(buf) => {
234            buf.clear();
235            buf.extend_from_slice(data);
236            true
237        }
238        VfsNode::Dir(_) => false,
239    }
240}
241
242/// 파일에 데이터를 이어 쓰기(append).
243pub fn append_file(path: &str, data: &[u8]) -> bool {
244    let node_ref = match lookup(path) {
245        Some(n) => n,
246        None => return false,
247    };
248    let mut node = node_ref.lock();
249    match &mut *node {
250        VfsNode::File(buf) => {
251            buf.extend_from_slice(data);
252            true
253        }
254        VfsNode::Dir(_) => false,
255    }
256}

258// ── 디렉토리 열거 ────────────────────────────────────────────────────────────
259
260/// 디렉토리 내용을 `DirEntry` 목록으로 반환 (이름 순 정렬).
261///
262/// 경로가 없거나 파일이면 빈 Vec.
263pub fn list_dir(path: &str) -> Vec<DirEntry> {
264    let node_ref = match lookup(path) {
265        Some(n) => n,
266        None => return Vec::new(),
267    };
268    let node = node_ref.lock();
269    match &*node {
270        VfsNode::Dir(dir) => dir
271            .list()
272            .into_iter()
273            .map(|(name, is_dir, size)| DirEntry { name, is_dir, size })
274            .collect(),
275        VfsNode::File(_) => Vec::new(),
276    }
