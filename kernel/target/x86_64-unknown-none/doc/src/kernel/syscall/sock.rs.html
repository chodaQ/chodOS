1//! BETA 3: socket syscall 스텁
2//!
3//! MuKernel은 커널 내부에 virtio-net + TCP/IP 스택이 있지만
4//! 유저스페이스 소켓 API는 아직 연결되지 않았음.
5//! 이 파일은 musl/glibc 프로그램이 네트워크 syscall을 호출할 때
6//! 크래시하지 않도록 의미 있는 errno를 반환한다.
7
8// ── 소켓 errno ────────────────────────────────────────────────────────────────
9
10const EAFNOSUPPORT: i64  = -97;   // Address family not supported
11const EPROTONOSUPPORT: i64 = -93; // Protocol not supported
12const ECONNREFUSED: i64  = -111;  // Connection refused
13const ENOTCONN: i64      = -107;  // Transport endpoint not connected
14const EOPNOTSUPP: i64    = -95;   // Operation not supported
15
16// ── 가짜 소켓 fd 풀 ──────────────────────────────────────────────────────────
17
18use core::sync::atomic::{AtomicI32, Ordering};
19static SOCK_NEXT_FD: AtomicI32 = AtomicI32::new(20); // fd 20번 이상 사용
20
21pub fn is_sock_fd(fd: i32) -> bool {
22    fd >= 20
23}
24
25// ── socket 구현 ──────────────────────────────────────────────────────────────

27/// 41: socket(domain, type, protocol) → 가짜 소켓 fd
28///
29/// AF_INET(2) / AF_INET6(10) / AF_UNIX(1) 모두 허용.
30/// 추후 유저스페이스 TCP/UDP 연결 시 실제 구현으로 교체.
31pub fn sys_socket(domain: u64, sock_type: u64, _protocol: u64) -> i64 {
32    // AF_UNIX = 1, AF_INET = 2, AF_INET6 = 10 허용
33    match domain {
34        1 | 2 | 10 => {}
35        _ => return EAFNOSUPPORT,
36    }
37    // SOCK_STREAM = 1, SOCK_DGRAM = 2, SOCK_RAW = 3 (SOCK_NONBLOCK = 0x800 마스크 제거)
38    let base_type = sock_type & !0x800;
39    match base_type {
40        1 | 2 | 3 => {}
41        _ => return EPROTONOSUPPORT,
42    }
43    let fd = SOCK_NEXT_FD.fetch_add(1, Ordering::Relaxed);
44    crate::serial_println!("[sock] socket(domain={}, type={}) → fd {}", domain, sock_type, fd);
45    fd as i64
46}

48/// 42: connect(sockfd, addr*, addrlen) → ECONNREFUSED
49/// 유저스페이스 TCP 미구현 — ECONNREFUSED 반환
50pub fn sys_connect(sockfd: u64, _addr: u64, _addrlen: u64) -> i64 {
51    crate::serial_println!("[sock] connect(fd={}) → ECONNREFUSED (no userspace TCP)", sockfd);
52    ECONNREFUSED
53}

66/// 51: getsockname(sockfd, addr*, addrlen*) → 0, 주소 0으로 초기화
67pub fn sys_getsockname(_sockfd: u64, addr: u64, addrlen: u64) -> i64 {
68    if addr != 0 && addrlen != 0 {
69        let len = unsafe { *(addrlen as *const u32) } as usize;
70        unsafe { core::ptr::write_bytes(addr as *mut u8, 0, len.min(128)); }
71    }
72    0
73}

105/// 55: getsockopt(sockfd, level, optname, optval*, optlen*) → 0
106pub fn sys_getsockopt(_sockfd: u64, level: u64, optname: u64,
107                      optval: u64, optlen: u64) -> i64 {
108    // SO_ERROR(4), SO_TYPE(3) 등 → 0 반환
109    if optval != 0 && optlen != 0 {
110        let len = unsafe { *(optlen as *const u32) } as usize;
111        unsafe { core::ptr::write_bytes(optval as *mut u8, 0, len.min(128)); }
112    }
113    let _ = (level, optname);
114    0
