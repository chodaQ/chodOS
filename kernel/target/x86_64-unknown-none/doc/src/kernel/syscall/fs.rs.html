8static mut LAST_OPEN_BUF: [u8; 256] = [0u8; 256];
9static     LAST_OPEN_LEN: AtomicUsize = AtomicUsize::new(0);
10static     LAST_OPEN_DIR: AtomicBool  = AtomicBool::new(false);
11
12pub(crate) fn set_last_open(path: &str, is_dir: bool) {
13    let bytes = path.as_bytes();
14    let n = bytes.len().min(255);
15    unsafe { LAST_OPEN_BUF[..n].copy_from_slice(&bytes[..n]); }
16    LAST_OPEN_LEN.store(n, Ordering::Relaxed);
17    LAST_OPEN_DIR.store(is_dir, Ordering::Relaxed);
18}
19
20fn last_open_path() -> &'static str {
21    let n = LAST_OPEN_LEN.load(Ordering::Relaxed);
22    if n == 0 { return "/"; }
23    core::str::from_utf8(unsafe { &LAST_OPEN_BUF[..n] }).unwrap_or("/")
24}

53pub fn fill_stat_pub(stat_ptr: u64, size: i64, is_dir: bool) { fill_stat(stat_ptr, size, is_dir); }
54fn fill_stat(stat_ptr: u64, size: i64, is_dir: bool) {
55    let s = unsafe { &mut *(stat_ptr as *mut Stat) };
56    *s = Stat {
57        st_dev: 1, st_ino: 1, st_nlink: 1,
58        st_mode: if is_dir { S_IFDIR | DEFAULT_DIR_PERM }
59                 else       { S_IFREG | DEFAULT_FILE_PERM },
60        st_uid: 0, st_gid: 0, _pad0: 0, st_rdev: 0,
61        st_size: size,
62        st_blksize: 4096,
63        st_blocks: (size + 511) / 512,
64        st_atime: 0, st_atime_ns: 0,
65        st_mtime: 0, st_mtime_ns: 0,
66        st_ctime: 0, st_ctime_ns: 0,
67        _unused: [0; 3],
68    };
69}

83/// 4: stat(path*, statbuf*)
84pub fn sys_stat(path_vaddr: u64, stat_vaddr: u64) -> i64 {
85    let path = match unsafe { read_cstr(path_vaddr) } { Some(p) => p, None => return EINVAL };
86    // 디렉토리 판정
87    if path == "/" || path.ends_with('/') {
88        fill_stat(stat_vaddr, 0, true);
89        return 0;
90    }
91    // ext4 파일 확인
92    if let Some(data) = crate::vfs::ext4_read_file(path) {
93        fill_stat(stat_vaddr, data.len() as i64, false);
94        return 0;
95    }
96    // tmpfs 파일 확인
97    if let Some(data) = crate::vfs::read_file(path) {
98        fill_stat(stat_vaddr, data.len() as i64, false);
99        return 0;
100    }
101    // 디렉토리 목록이 존재하면 디렉토리
102    let entries = crate::vfs::list_dir(path);
103    if !entries.is_empty() {
104        fill_stat(stat_vaddr, 0, true);
105        return 0;
106    }
107    ENOENT
108}

110/// 5: fstat(fd, statbuf*)
111pub fn sys_fstat(fd: u64, stat_vaddr: u64) -> i64 {
112    let s = unsafe { &mut *(stat_vaddr as *mut Stat) };
113    *s = unsafe { core::mem::zeroed() };
114    match fd {
115        0 | 1 | 2 | FD_DEV_NULL => {
116            s.st_mode = S_IFCHR | 0o666;
117            s.st_nlink = 1;
118            0
119        }
120        3 => {
121            if LAST_OPEN_DIR.load(Ordering::Relaxed) {
122                fill_stat(stat_vaddr, 0, true);
123            } else {
124                fill_stat(stat_vaddr, 0, false);
125            }
126            0
127        }
128        _ => EBADF,
129    }
130}

158/// 19: readv(fd, iov*, iovcnt)
159pub fn sys_readv(fd: u64, iov_vaddr: u64, iovcnt: u64) -> i64 {
160    if fd != 0 { return EBADF; }
161    let mut total = 0i64;
162    for i in 0..iovcnt as usize {
163        let iov = unsafe { &*((iov_vaddr as *const Iovec).add(i)) };
164        if iov.iov_len == 0 { continue; }
165        let byte = crate::kbd::read_key_blocking();
166        unsafe { *(iov.iov_base as *mut u8) = byte; }
167        total += 1;
168        break; // 한 번에 1바이트만
169    }
170    total
171}

173/// 20: writev(fd, iov*, iovcnt) — stdout/stderr 출력
174pub fn sys_writev(fd: u64, iov_vaddr: u64, iovcnt: u64) -> i64 {
175    if fd != 1 && fd != 2 { return EBADF; }
176    let mut total = 0i64;
177    for i in 0..iovcnt as usize {
178        let iov = unsafe { &*((iov_vaddr as *const Iovec).add(i)) };
179        let bytes = unsafe {
180            core::slice::from_raw_parts(iov.iov_base as *const u8, iov.iov_len as usize)
181        };
182        for &b in bytes { crate::serial::write_byte(b); }
183        crate::term::write_bytes(bytes);
184        total += iov.iov_len as i64;
185    }
186    total
187}

189/// 21: access(path*, mode) → 존재하면 0, 없으면 ENOENT
190pub fn sys_access(path_vaddr: u64, _mode: u64) -> i64 {
191    let path = match unsafe { read_cstr(path_vaddr) } { Some(p) => p, None => return EINVAL };
192    if path == "/" { return 0; }
193    if crate::vfs::ext4_read_file(path).is_some() { return 0; }
194    if crate::vfs::read_file(path).is_some() { return 0; }
195    if !crate::vfs::list_dir(path).is_empty() { return 0; }
196    ENOENT
197}

210/// 72: fcntl(fd, cmd, arg) — F_GETFL/F_SETFL만 처리
211pub fn sys_fcntl(fd: u64, cmd: u64, _arg: u64) -> i64 {
212    const F_GETFL: u64 = 3;
213    const F_SETFL: u64 = 4;
214    const F_GETFD: u64 = 1;
215    const F_SETFD: u64 = 2;
216    match (fd, cmd) {
217        (_, F_GETFD)               => 0,       // FD_CLOEXEC 없음
218        (_, F_SETFD)               => 0,
219        (0, F_GETFL)               => 0,       // O_RDONLY
220        (1 | 2, F_GETFL)           => 1,       // O_WRONLY
221        (3, F_GETFL)               => 0,
222        (_, F_SETFL)               => 0,
223        _                          => EINVAL,
224    }
225}

291/// 257: openat(dirfd, path*, flags, mode)
292///      AT_FDCWD(-100)이면 절대 경로처럼 처리
293pub fn sys_openat(_dirfd: i64, path_vaddr: u64, flags: u64, _mode: u64) -> i64 {
294    // BETA 4: openat도 HandleTable 경유
295    let path = match unsafe { read_cstr(path_vaddr) } { Some(p) => p, None => return EINVAL };
296    if let Some(data) = crate::vfs::ext4_read_file(path) {
297        return crate::syscall::fd::open_file(path, data.to_vec(), false) as i64;
298    }
299    if let Some(data) = crate::vfs::read_file(path) {
300        let w = flags & 0x01 != 0 || flags & 0x02 != 0;
301        return crate::syscall::fd::open_file(path, data.to_vec(), w) as i64;
302    }
303    if flags & 0x40 != 0 {
304        let _ = crate::vfs::create_file(path);
305        return crate::syscall::fd::open_file(path, alloc::vec![], true) as i64;
306    }
307    ENOENT
308}

334/// linux_dirent64 구조체 (가변 길이 이름 포함)
335/// d_reclen은 전체 크기이고 8바이트 정렬.
336#[repr(C, packed)]
337struct Dirent64Hdr {
338    d_ino:    u64,
339    d_off:    i64,
340    d_reclen: u16,
341    d_type:   u8,
342    // d_name: [u8] 가변 길이 — hdr 바로 뒤에 위치
343}
344
345const DT_UNKNOWN: u8 = 0;
346const DT_REG:     u8 = 8;
347const DT_DIR:     u8 = 4;
348
349/// 217: getdents64(fd, buf*, count) → 읽은 바이트 수
350///
351/// fd=3으로 열린 마지막 경로의 VFS 목록을 반환.
352/// 실제 fd→path 테이블이 없으므로 마지막 open 경로를 전역으로 추적.
353pub fn sys_getdents64(fd: u64, buf_vaddr: u64, count: u64) -> i64 {
354    if fd == 0 || fd == 1 || fd == 2 { return EBADF; }
355    if buf_vaddr == 0 { return EFAULT; }
356    if count < 32 { return EINVAL; }
357
358    // BETA 4: HandleTable의 DirResource에서 경로 조회
359    // 없으면 기존 LAST_OPEN_BUF 폴백
360    let dir_path_owned;
361    let path: &str = if let Some(p) = crate::syscall::fd::dir_path(fd as u32) {
362        dir_path_owned = p;
363        &dir_path_owned
364    } else if LAST_OPEN_DIR.load(Ordering::Relaxed) {
365        last_open_path()
366    } else {
367        return 0;
368    };
369
370    // VFS 목록 수집 (tmpfs + ext4)
371    let mut entries: alloc::vec::Vec<(alloc::string::String, bool)> =
372        crate::vfs::list_dir(path)
373            .into_iter()
374            .map(|e| (e.name, e.is_dir))
375            .collect();
376    // ext4도 합침
377    for e in crate::vfs::ext4_list_dir(path) {
378        if !entries.iter().any(|(n, _)| n == &e.name) {
379            entries.push((e.name, e.is_dir));
380        }
381    }
382
383    let buf = buf_vaddr as *mut u8;
384    let max = count as usize;
385    let mut pos = 0usize;
386    let mut ino: u64 = 2;
387
388    for (name, is_dir) in &entries {
389        let name_bytes = name.as_bytes();
390        let name_len = name_bytes.len();
391        // reclen = sizeof(Dirent64Hdr) + name_len + 1(NUL), 8-byte aligned
392        let raw_len = core::mem::size_of::<Dirent64Hdr>() + name_len + 1;
393        let rec_len = (raw_len + 7) & !7;
394
395        if pos + rec_len > max { break; }
396
397        let hdr = unsafe { &mut *(buf.add(pos) as *mut Dirent64Hdr) };
398        hdr.d_ino    = ino;
399        hdr.d_off    = (pos + rec_len) as i64;
400        hdr.d_reclen = rec_len as u16;
401        hdr.d_type   = if *is_dir { DT_DIR } else { DT_REG };
402
403        // 이름 복사 (hdr 바로 뒤)
404        let name_dst = unsafe { buf.add(pos + core::mem::size_of::<Dirent64Hdr>()) };
405        unsafe {
406            core::ptr::copy_nonoverlapping(name_bytes.as_ptr(), name_dst, name_len);
407            *name_dst.add(name_len) = 0; // NUL
408            // 패딩 바이트 0으로 초기화
409            if rec_len > raw_len {
410                core::ptr::write_bytes(name_dst.add(name_len + 1), 0, rec_len - raw_len);
411            }
412        }
413
414        pos += rec_len;
415        ino += 1;
416    }
417
418    pos as i64
419}
