157pub(crate) unsafe fn read_cstr(vaddr: u64) -> Option<&'static str> {
158    if vaddr == 0 { return None; }
159    let ptr = vaddr as *const u8;
160    let mut len = 0usize;
161    while *ptr.add(len) != 0 && len < 4096 { len += 1; }
162    core::str::from_utf8(core::slice::from_raw_parts(ptr, len)).ok()
163}

165pub(crate) fn sys_exit_impl(code: u64) -> ! {
166    use core::sync::atomic::Ordering;
167    crate::interrupts::handlers::USER_EXIT_CODE.store(code as i32, Ordering::Relaxed);
168    crate::serial_println!("[syscall] exit({}) — longjmp to kernel", code);
169    unsafe {
170        core::arch::asm!(
171            "mov cr3, {kcr3}",
172            "mov rsp, {krsp}",
173            "jmp {resume}",
174            kcr3   = in(reg) crate::paging::KERNEL_CR3,
175            krsp   = in(reg) crate::paging::KERNEL_MAIN_RSP,
176            resume = sym crate::interrupts::handlers::after_user_demo,
177            options(noreturn, nostack),
178        );
179    }
180}

188        let a2 = *frame.add(5); // rsi
189        let a3 = *frame.add(6); // rdx
190        let a4 = *frame.add(1); // r10
191        let a5 = *frame.add(3); // r8
192        let a6 = *frame.add(2); // r9
193        (nr, a1, a2, a3, a4, a5, a6)

196    crate::serial_println!(
197        "[syscall] nr={:<3} ({})  a1={:#x} a2={:#x} a3={:#x}",

338fn sys_write(fd: u64, buf: u64, count: u64) -> i64 {
339    if count == 0 { return 0; }
340    // pipe write fd
341    if pipe::is_write_fd(fd as i32) {
342        return pipe::pipe_write(buf as *const u8, count as usize);
343    }
344    match fd {
345        1 | 2 | FD_DEV_NULL => {
346            if fd != FD_DEV_NULL {
347                let bytes = unsafe { core::slice::from_raw_parts(buf as *const u8, count as usize) };
348                for &b in bytes { crate::serial::write_byte(b); }
349                crate::term::write_bytes(bytes);
350            }
351            count as i64
352        }
353        _ => EBADF,
354    }
355}

357fn sys_read(fd: u64, buf: u64, count: u64) -> i64 {
358    if count == 0 { return 0; }
359    // pipe read fd (블로킹)
360    if pipe::is_read_fd(fd as i32) {
361        loop {
362            let n = pipe::pipe_read(buf as *mut u8, count as usize);
363            if n > 0 { return n; }
364            unsafe { core::arch::asm!("hlt", options(nomem, nostack, preserves_flags)); }
365        }
366    }
367    match fd {
368        0 => {
369            let byte = crate::kbd::read_key_blocking();
370            if byte == 3 { proc::deliver_sigint(); }
371            unsafe { *(buf as *mut u8) = byte; }
372            1
373        }
374        FD_DEV_NULL => 0,
375        // BETA 4: HandleTable에서 FileResource 조회
376        f if f >= 3 => fd::read(f as u32, buf as *mut u8, count as usize),
377        _ => EAGAIN,
378    }
379}

381fn sys_open(path_vaddr: u64, flags: u64) -> i64 {
382    let path = match unsafe { read_cstr(path_vaddr) } { Some(p) => p, None => return EINVAL };
383
384    // /dev/null, /dev/zero → 특수 fd (테이블 밖)
385    if path == "/dev/null" || path == "/dev/zero" {
386        return FD_DEV_NULL as i64;
387    }
388
389    // 디렉토리 열기
390    let o_dir = flags & 0x10000 != 0;
391    if o_dir || path.ends_with('/') || path == "/" {
392        return fd::open_dir(path) as i64;
393    }
394    if !crate::vfs::list_dir(path).is_empty() {
395        return fd::open_dir(path) as i64;
396    }
397
398    // ext4 파일
399    if let Some(data) = crate::vfs::ext4_read_file(path) {
400        let v = data.to_vec();
401        return fd::open_file(path, v, false) as i64;
402    }
403    // tmpfs 파일
404    if let Some(data) = crate::vfs::read_file(path) {
405        let v = data.to_vec();
406        let writable = flags & 0x01 != 0 || flags & 0x02 != 0; // O_WRONLY|O_RDWR
407        return fd::open_file(path, v, writable) as i64;
408    }
409    // O_CREAT(0x40): 빈 파일 생성
410    if flags & 0x40 != 0 {
411        let _ = crate::vfs::create_file(path);
412        return fd::open_file(path, alloc::vec![], true) as i64;
413    }
414    ENOENT
415}

417fn sys_close(fd: u64) -> i64 {
418    match fd {
419        0 | 1 | 2 => 0,      // stdio는 닫지 않음
420        FD_DEV_NULL => 0,
421        f if pipe::is_read_fd(f as i32)  => 0, // pipe 닫기는 간단 무시
422        f if pipe::is_write_fd(f as i32) => 0,
423        f => { fd::close(f as u32); 0 }
424    }
425}

429/// fstat v4 — stdio/dev/null은 기존 처리, fd≥3은 HandleTable 조회
430fn sys_fstat_v4(raw_fd: u64, stat_vaddr: u64) -> i64 {
431    match raw_fd {
432        0 | 1 | 2 | FD_DEV_NULL => fs::sys_fstat(raw_fd, stat_vaddr),
433        f => {
434            if let Some((size, is_dir)) = fd::fstat(f as u32) {
435                fs::fill_stat_pub(stat_vaddr, size, is_dir);
436                0
437            } else {
438                EBADF
439            }
440        }
441    }
442}

453/// pread64 v4 — fd≥3은 FileResource.pread() 사용 (파일 위치 불변)
454fn sys_pread64_v4(raw_fd: u64, buf_vaddr: u64, count: u64, offset: u64) -> i64 {
455    if raw_fd >= 3 {
456        fd::pread(raw_fd as u32, buf_vaddr as *mut u8, count as usize, offset as usize)
457    } else {
458        fs::sys_pread64(raw_fd, buf_vaddr, count, offset)
459    }
460}

462fn sys_mmap(addr: u64, len: u64, _prot: u64, flags: u64, fd: i64, _off: u64) -> i64 {
463    const MAP_ANONYMOUS: u64 = 0x20;
464    if flags & MAP_ANONYMOUS == 0 || fd != -1 { return EINVAL; }
465    let pages = (len as usize + 4095) / 4096;
466    let layout = alloc::alloc::Layout::from_size_align(pages * 4096, 4096)
467        .unwrap_or_else(|_| panic!("bad mmap layout"));
468    let ptr = unsafe { alloc::alloc::alloc_zeroed(layout) };
469    if ptr.is_null() { ENOMEM } else { let _ = addr; ptr as i64 }
470}

472fn sys_execve(path_vaddr: u64, argv_vaddr: u64) -> i64 {
473    use core::sync::atomic::Ordering;
474    let path = match unsafe { read_cstr(path_vaddr) } { Some(p) => p, None => return EINVAL };
475    let name = path.rsplit('/').next().unwrap_or(path);
476    let idx = match crate::pkg::find(name) {
477        Some(i) => i,
478        None => { crate::serial_println!("[mukg] not found: {:?}", name); return ENOENT; }
479    };
480    let args = if argv_vaddr != 0 {
481        unsafe { read_cstr(argv_vaddr) }.unwrap_or("").as_bytes()
482    } else { b"" };
483    crate::pkg::set_args(args);
484    crate::interrupts::handlers::PENDING_PKG_IDX.store(idx as i32, Ordering::Relaxed);
485    crate::serial_println!("[mukg] exec {:?}", name);
486    sys_exit_impl(0)
487}

489fn sys_pkg_list(buf_vaddr: u64, max_len: u64) -> i64 {
490    let max = max_len as usize;
491    let out = buf_vaddr as *mut u8;
492    let mut pos = 0usize;
493    for pkg in crate::pkg::PACKAGES {
494        for s in &[pkg.name.as_bytes(), b" ", pkg.version.as_bytes(),
495                   b" - ", pkg.desc.as_bytes(), b"\n"] {
496            let copy = s.len().min(max.saturating_sub(pos));
497            if copy == 0 { break; }
498            unsafe { core::ptr::copy_nonoverlapping(s.as_ptr(), out.add(pos), copy); }
499            pos += copy;
500        }
501        if pos >= max { break; }
502    }
503    pos as i64
504}
505
506fn sys_pkg_getargs(buf_vaddr: u64, max_len: u64) -> i64 {
507    let args = crate::pkg::get_args();
508    let copy = args.len().min(max_len as usize);
509    unsafe { core::ptr::copy_nonoverlapping(args.as_ptr(), buf_vaddr as *mut u8, copy); }
510    copy as i64
511}

513fn sys_ext4_read(path_vaddr: u64, buf_vaddr: u64, max_len: u64) -> i64 {
514    let path = match unsafe { read_cstr(path_vaddr) } { Some(p) => p, None => return EINVAL };
515    match crate::vfs::ext4_read_file(path) {
516        Some(data) => {
517            let copy = data.len().min(max_len as usize);
518            unsafe { core::ptr::copy_nonoverlapping(data.as_ptr(), buf_vaddr as *mut u8, copy); }
519            copy as i64
520        }
521        None => ENOENT,
522    }
523}
