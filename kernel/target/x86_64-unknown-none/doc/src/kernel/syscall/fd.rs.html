1//! BETA 4: Capability Handle Table ↔ 실제 fd 경로 연결
2//!
3//! ## 설계
4//!
5//! ```text
6//! sys_open()  →  FileResource / DirResource
7//!                     ↓
8//!              HandleTable::insert()  →  fd (u32)
9//!                     ↓
10//! sys_read/write/lseek/close(fd)
11//!     →  HandleTable::get::<FileResource>(fd)
12//!     →  데이터 읽기 / 위치 이동 / Arc 참조 해제
13//! ```
14//!
15//! ## 전역 싱글턴
16//!
17//! 커널은 한 번에 프로세스 하나만 실행하므로, 전역 `FD_TABLE` 하나로 충분.
18//! `init()` 은 새 프로세스 시작 시 테이블을 초기화한다.
19//!
20//! ## SAFETY
21//!
22//! `FdTableCell`은 `UnsafeCell`로 내부 가변성을 제공.
23//! 싱글코어 + 시스템콜 핸들러 비재진입 → 실제 경쟁 없음.
24
25use alloc::{string::String, vec::Vec};
26use alloc::sync::Arc;
27use core::cell::UnsafeCell;
28use core::sync::atomic::{AtomicUsize, Ordering};
29use crate::process::handle::{Capability, HandleTable, Rights};

33/// 열린 파일 — 내용을 open 시점에 캐시, 독립적인 읽기 위치 보유
34pub struct FileResource {
35    pub path:  String,
36    pub data:  Vec<u8>,
37    pos:       AtomicUsize,
38    pub write: bool, // 쓰기 권한 (tmpfs 파일만 true)
39}
40
41impl FileResource {
42    pub fn new(path: String, data: Vec<u8>, writable: bool) -> Self {
43        FileResource { path, data, pos: AtomicUsize::new(0), write: writable }
44    }
45
46    /// 현재 위치에서 최대 `count` 바이트 읽기 → 실제 읽은 바이트 수
47    pub fn read(&self, buf: *mut u8, count: usize) -> usize {
48        let pos   = self.pos.load(Ordering::Relaxed);
49        let avail = self.data.len().saturating_sub(pos);
50        let n     = count.min(avail);
51        if n > 0 {
52            unsafe { core::ptr::copy_nonoverlapping(self.data[pos..].as_ptr(), buf, n); }
53            self.pos.fetch_add(n, Ordering::Relaxed);
54        }
55        n
56    }
57
58    /// 특정 오프셋에서 읽기 (파일 위치 변경 없음)
59    pub fn pread(&self, buf: *mut u8, count: usize, offset: usize) -> usize {
60        let avail = self.data.len().saturating_sub(offset);
61        let n     = count.min(avail);
62        if n > 0 {
63            unsafe { core::ptr::copy_nonoverlapping(self.data[offset..].as_ptr(), buf, n); }
64        }
65        n
66    }
67
68    /// 파일 위치 이동 (SEEK_SET=0, SEEK_CUR=1, SEEK_END=2)
69    pub fn seek(&self, offset: i64, whence: u64) -> i64 {
70        let len = self.data.len() as i64;
71        let cur = self.pos.load(Ordering::Relaxed) as i64;
72        let new_pos = match whence {
73            0 => offset,
74            1 => cur + offset,
75            2 => len + offset,
76            _ => return -22, // EINVAL
77        };
78        if new_pos < 0 { return -22; }
79        let np = new_pos as usize;
80        self.pos.store(np, Ordering::Relaxed);
81        new_pos
82    }
83
84    pub fn pos(&self)  -> usize { self.pos.load(Ordering::Relaxed) }
85    pub fn size(&self) -> usize { self.data.len() }
86}

94// ── 전역 FD 테이블 ────────────────────────────────────────────────────────────
95
96struct FdTableCell(UnsafeCell<Option<HandleTable>>);
97
98// SAFETY: 싱글코어 커널 + 시스템콜 비재진입 → 안전
99unsafe impl Sync for FdTableCell {}
100
101static FD_TABLE: FdTableCell = FdTableCell(UnsafeCell::new(None));
102
103/// 테이블을 빌려 `f`를 실행.
104/// 테이블이 아직 없으면 자동 초기화 (fd 0/1/2는 HandleTable 밖에서 처리).
105fn with_table<R>(f: impl FnOnce(&mut HandleTable) -> R) -> R {
106    unsafe {
107        let ptr = FD_TABLE.0.get();
108        if (*ptr).is_none() {
109            *ptr = Some(HandleTable::new());
110        }
111        f((*ptr).as_mut().unwrap())
112    }
113}
114
115/// 새 프로세스 시작 전 fd 테이블 초기화 (이전 프로세스의 fd 전부 해제)
116pub fn init() {
117    unsafe {
118        let ptr = FD_TABLE.0.get();
119        *ptr = Some(HandleTable::new());
120    }
121    crate::serial_println!("[fd] table reset");
122}

126/// 파일 오픈 → fd 반환
127pub fn open_file(path: &str, data: Vec<u8>, writable: bool) -> u32 {
128    let rights = if writable { Rights::READ | Rights::WRITE } else { Rights::READ };
129    with_table(|t| {
130        let cap = Capability::new(FileResource::new(path.into(), data, writable), rights);
131        let h   = t.insert(cap);
132        crate::serial_println!("[fd] open_file {:?} → fd {}", path, h.id);
133        h.id
134    })
135}
136
137/// 디렉토리 오픈 → fd 반환
138pub fn open_dir(path: &str) -> u32 {
139    with_table(|t| {
140        let cap = Capability::new(DirResource { path: path.into() }, Rights::READ);
141        let h   = t.insert(cap);
142        crate::serial_println!("[fd] open_dir {:?} → fd {}", path, h.id);
143        h.id
144    })
145}

147/// 파일 읽기 → 실제 읽은 바이트 수 (EBADF = -9)
148pub fn read(fd: u32, buf: *mut u8, count: usize) -> i64 {
149    with_table(|t| {
150        if let Some(f) = t.get::<FileResource>(fd, Rights::READ) {
151            f.read(buf, count) as i64
152        } else {
153            -9 // EBADF
154        }
155    })
156}
157
158/// 오프셋 지정 읽기 (파일 위치 불변)
159pub fn pread(fd: u32, buf: *mut u8, count: usize, offset: usize) -> i64 {
160    with_table(|t| {
161        if let Some(f) = t.get::<FileResource>(fd, Rights::READ) {
162            f.pread(buf, count, offset) as i64
163        } else {
164            -9 // EBADF
165        }
166    })
167}

169/// 파일 위치 이동 → 새 위치 (EBADF = -9)
170pub fn seek(fd: u32, offset: i64, whence: u64) -> i64 {
171    with_table(|t| {
172        if let Some(f) = t.get::<FileResource>(fd, Rights::NONE) {
173            f.seek(offset, whence)
174        } else {
175            -9 // EBADF
176        }
177    })
178}

180/// fd 닫기 (Arc 참조 해제) → 성공 여부
181pub fn close(fd: u32) -> bool {
182    let ok = with_table(|t| t.close(fd));
183    if ok { crate::serial_println!("[fd] close fd {}", fd); }
184    ok
185}
186
187/// fstat 정보: (파일 크기, is_dir)
188pub fn fstat(fd: u32) -> Option<(i64, bool)> {
189    with_table(|t| {
190        if let Some(f) = t.get::<FileResource>(fd, Rights::NONE) {
191            Some((f.size() as i64, false))
192        } else if t.get::<DirResource>(fd, Rights::NONE).is_some() {
193            Some((0i64, true))
194        } else {
195            None
196        }
197    })
198}

200/// getdents64 에서 디렉토리 경로 조회
201pub fn dir_path(fd: u32) -> Option<String> {
202    with_table(|t| {
203        t.get::<DirResource>(fd, Rights::NONE).map(|d| d.path.clone())
204    })
205}
206
207/// 현재 열린 핸들 목록 (디버깅)
208pub fn list() -> alloc::vec::Vec<(u32, Rights)> {
209    with_table(|t| t.list())
210}
211
212/// fd가 열려 있는지 확인 (파일 또는 디렉토리)
213pub fn is_open(fd: u32) -> bool {
214    with_table(|t| {
215        t.get::<FileResource>(fd, Rights::NONE).is_some()
216            || t.get::<DirResource>(fd, Rights::NONE).is_some()
217    })
