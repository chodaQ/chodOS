1//! VirtIO Net 드라이버 (ALPHA 11)
2//!
3//! ## 큐 구조
4//!
5//! ```
6//! Queue 0 (receiveq):  driver가 빈 버퍼 pre-post → device가 수신 패킷으로 채움
7//! Queue 1 (transmitq): driver가 패킷 넣음       → device가 전송하고 used ring에 완료 표시
8//! ```
9//!
10//! ## TX 디스크립터 체인
11//!
12//! ```text
13//! Desc[0]: virtio_net_hdr (10 bytes, READ | NEXT)
14//! Desc[1]: Ethernet frame (READ)
15//! ```
16//!
17//! ## RX 디스크립터 (pre-posted)
18//!
19//! ```text
20//! Desc[i]: 1524-byte buffer (hdr + max-frame, WRITE)
21//! ```
22//!
23//! ## QEMU 네트워크 (user-mode SLIRP)
24//!
25//! ```
26//! Guest IP : 10.0.2.15
27//! Gateway  : 10.0.2.2
28//! DNS      : 10.0.2.3
29//! ```
30

37// ── 상수 ────────────────────────────────────────────────────────────────────
38
39const VIRTIO_VENDOR:     u16 = 0x1AF4;
40const VIRTIO_NET_DEVICE: u16 = 0x1000;
41
42const R_GUEST_FEATURES: u16 = 0x04;
43const R_STATUS:         u16 = 0x12;
44const R_MAC:            u16 = 0x14; // 6바이트 MAC
45
46const STATUS_ACK:       u8 = 1;
47const STATUS_DRIVER:    u8 = 2;
48const STATUS_DRIVER_OK: u8 = 4;
49const STATUS_FAILED:    u8 = 128;
50
51const VIRTQ_DESC_F_NEXT:  u16 = 1;
52const VIRTQ_DESC_F_WRITE: u16 = 2;
53
54/// VirtIO Net 헤더 크기 (legacy, MRG_RXBUF 없음)
55const NET_HDR_LEN: usize = 10;
56/// RX 버퍼 크기 = 헤더 + 최대 Ethernet 프레임
57const RX_BUF_SIZE: usize = NET_HDR_LEN + 1514;
58/// pre-post할 RX 버퍼 수
59const RX_BUF_COUNT: usize = 8;
60
61// ── VirtIO Net 헤더 ──────────────────────────────────────────────────────────
62
63#[repr(C)]
64struct NetHdr {
65    flags:       u8,
66    gso_type:    u8,
67    hdr_len:     u16,
68    gso_size:    u16,
69    csum_start:  u16,
70    csum_offset: u16,
71}

73// ── 드라이버 ─────────────────────────────────────────────────────────────────
74
75pub struct VirtioNet {
76    io_base:    u16,
77    rxq:        Queue,
78    txq:        Queue,
79    pub mac:    [u8; 6],
80    rx_bufs:    *mut u8,
81    rx_layout:  Layout,
82    rx_posted:  u16, // avail_idx when we finished pre-posting
83}
84
85unsafe impl Send for VirtioNet {}
86unsafe impl Sync for VirtioNet {}
87
88impl VirtioNet {
89    pub fn init() -> Option<Self> {
90        let (bus, dev) = pci::find_device(VIRTIO_VENDOR, VIRTIO_NET_DEVICE)?;
91        crate::serial_println!("[virtio-net] found at PCI {:02x}:{:02x}", bus, dev);
92
93        pci::enable_io_and_busmaster(bus, dev);
94        let io_base = pci::bar_io_base(bus, dev, 0)?;
95        crate::serial_println!("[virtio-net] BAR0 I/O = 0x{:04x}", io_base);
96
97        unsafe {
98            // 리셋 → ACK|DRIVER
99            outb(io_base + R_STATUS, 0);
100            outb(io_base + R_STATUS, STATUS_ACK | STATUS_DRIVER);
101            outl(io_base + R_GUEST_FEATURES, 0); // feature 없음 (최소 동작)
102
103            // MAC 주소 읽기
104            let mut mac = [0u8; 6];
105            for i in 0..6usize {
106                mac[i] = pci::inb(io_base + R_MAC + i as u16);
107            }
108            crate::serial_println!(
109                "[virtio-net] MAC = {:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",
110                mac[0], mac[1], mac[2], mac[3], mac[4], mac[5]
111            );
112
113            // 큐 0 (RX) 초기화
114            let rxq = Queue::init(io_base, 0)?;
115            crate::serial_println!("[virtio-net] rxq size={}", rxq.qsize);
116
117            // 큐 1 (TX) 초기화
118            let txq = Queue::init(io_base, 1)?;
119            crate::serial_println!("[virtio-net] txq size={}", txq.qsize);
120
121            // DRIVER_OK
122            outb(io_base + R_STATUS, STATUS_ACK | STATUS_DRIVER | STATUS_DRIVER_OK);
123
124            // RX 버퍼 힙 할당
125            let rx_layout = Layout::from_size_align(RX_BUF_COUNT * RX_BUF_SIZE, 64).unwrap();
126            let rx_bufs   = alloc_zeroed(rx_layout);
127            if rx_bufs.is_null() {
128                crate::serial_println!("[virtio-net] RX buf alloc failed");
129                outb(io_base + R_STATUS, STATUS_FAILED);
130                return None;
131            }
132
133            let mut net = VirtioNet {
134                io_base, rxq, txq, mac, rx_bufs, rx_layout, rx_posted: 0,
135            };
136
137            // RX 버퍼 pre-post
138            net.post_rx_all();
139
140            Some(net)
141        }
142    }
143
144    // ── RX ───────────────────────────────────────────────────────────────────
145
146    /// RX 버퍼 전체를 큐에 등록
147    fn post_rx_all(&mut self) {
148        let hhdm = crate::memory::hhdm_offset();
149        unsafe {
150            for i in 0..RX_BUF_COUNT {
151                let virt = self.rx_bufs as u64 + (i * RX_BUF_SIZE) as u64;
152                let phys = virt - hhdm;
153                let slot = self.rxq.ai as usize % self.rxq.qsize;
154                self.rxq.write_desc(i, phys, RX_BUF_SIZE as u32, VIRTQ_DESC_F_WRITE, 0);
155                self.rxq.avail_put(slot, i as u16);
156                self.rxq.ai = self.rxq.ai.wrapping_add(1);
157            }
158            self.rxq.avail_commit(self.rxq.ai);
159            self.rx_posted = self.rxq.ai;
160        }
161        // 큐 0 알림
162        unsafe { pci::outw(self.io_base + 0x10, 0); }
163        crate::serial_println!("[virtio-net] pre-posted {} RX buffers", RX_BUF_COUNT);
164    }
165
166    /// 수신된 패킷이 있으면 Ethernet 프레임(헤더 이후)을 반환.
167    pub fn try_recv(&mut self) -> Option<Vec<u8>> {
168        unsafe {
169            fence(Ordering::SeqCst);
170            let used_idx = self.rxq.used_idx();
171            if used_idx == self.rxq.lu { return None; }
172
173            let slot   = self.rxq.lu as usize % self.rxq.qsize;
174            let buf_id = self.rxq.used_id(slot) as usize;
175            let total  = self.rxq.used_len(slot) as usize;
176
177            self.rxq.lu = self.rxq.lu.wrapping_add(1);
178
179            // 헤더 이후가 실제 Ethernet 프레임
180            let pkt_len = total.saturating_sub(NET_HDR_LEN);
181            let pkt_ptr = self.rx_bufs.add(buf_id * RX_BUF_SIZE + NET_HDR_LEN);
182
183            let mut pkt = Vec::with_capacity(pkt_len);
184            for i in 0..pkt_len {
185                pkt.push(*pkt_ptr.add(i));
186            }
187
188            // 버퍼 재사용: 해당 desc를 avail ring에 다시 등록
189            let virt  = self.rx_bufs as u64 + (buf_id * RX_BUF_SIZE) as u64;
190            let phys  = virt - crate::memory::hhdm_offset();
191            let aslot = self.rxq.ai as usize % self.rxq.qsize;
192            self.rxq.write_desc(buf_id, phys, RX_BUF_SIZE as u32, VIRTQ_DESC_F_WRITE, 0);
193            self.rxq.avail_put(aslot, buf_id as u16);
194            self.rxq.ai = self.rxq.ai.wrapping_add(1);
195            fence(Ordering::SeqCst);
196            self.rxq.avail_commit(self.rxq.ai);
197            pci::outw(self.io_base + 0x10, 0); // 큐 0 알림
198
199            Some(pkt)
200        }
201    }
202
203    // ── TX ───────────────────────────────────────────────────────────────────
204
205    /// Ethernet 프레임 전송 (헤더는 드라이버가 prepend).
206    ///
207    /// 2-descriptor 체인: [NetHdr | NEXT] → [frame]
208    /// 완료까지 폴링 (최대 ~500ms).
209    pub fn send(&mut self, frame: &[u8]) -> bool {
210        if frame.is_empty() { return false; }
211        unsafe {
212            let hhdm = crate::memory::hhdm_offset();
213
214            let hdr  = NetHdr { flags: 0, gso_type: 0, hdr_len: 0, gso_size: 0,
215                                 csum_start: 0, csum_offset: 0 };
216            let hdr_phys   = (&hdr as *const NetHdr as u64).wrapping_sub(hhdm);
217            let frame_phys = (frame.as_ptr() as u64).wrapping_sub(hhdm);
218
219            // Desc[d0]: 헤더
220            let d0 = self.txq.nd as usize;
221            self.txq.nd = (self.txq.nd + 1) % self.txq.qsize as u16;
222            // Desc[d1]: 프레임
223            let d1 = self.txq.nd as usize;
224            self.txq.nd = (self.txq.nd + 1) % self.txq.qsize as u16;
225
226            self.txq.write_desc(d0, hdr_phys, NET_HDR_LEN as u32, VIRTQ_DESC_F_NEXT, d1 as u16);
227            self.txq.write_desc(d1, frame_phys, frame.len() as u32, 0, 0);
228
229            let slot = self.txq.ai as usize % self.txq.qsize;
230            self.txq.avail_put(slot, d0 as u16);
231            self.txq.ai = self.txq.ai.wrapping_add(1);
232            fence(Ordering::SeqCst);
233            self.txq.avail_commit(self.txq.ai);
234
235            // 큐 1 알림
236            pci::outw(self.io_base + 0x10, 1);
237
238            // 완료 폴링 (TX used ring 진행 확인)
239            let deadline = crate::interrupts::handlers::TICK.load(Ordering::Relaxed) + 9;
240            loop {
241                fence(Ordering::SeqCst);
242                if self.txq.used_idx() != self.txq.lu {
243                    self.txq.lu = self.txq.lu.wrapping_add(1);
244                    return true;
245                }
246                if crate::interrupts::handlers::TICK.load(Ordering::Relaxed) > deadline {
247                    return false;
248                }
249                core::arch::asm!("pause", options(nomem, nostack));
250            }
251        }
252    }
253}

255impl Drop for VirtioNet {
256    fn drop(&mut self) {
257        unsafe {
258            outb(self.io_base + R_STATUS, 0);
259            if !self.rx_bufs.is_null() {
260                dealloc(self.rx_bufs, self.rx_layout);
261            }
262        }
263    }
264}

268/// ARP request 프레임 (42 bytes) 빌드
269///
270/// ```
271/// who has `dst_ip`? tell `src_ip` (src_mac)
272/// ```
273pub fn build_arp_request(
274    src_mac: [u8; 6],
275    src_ip:  [u8; 4],
276    dst_ip:  [u8; 4],
277) -> [u8; 42] {
278    let mut f = [0u8; 42];
279    // Ethernet header
280    f[0..6].copy_from_slice(&[0xFF; 6]);          // dst: broadcast
281    f[6..12].copy_from_slice(&src_mac);            // src
282    f[12] = 0x08; f[13] = 0x06;                   // EtherType: ARP
283    // ARP payload
284    f[14] = 0x00; f[15] = 0x01;                   // hw_type: Ethernet
285    f[16] = 0x08; f[17] = 0x00;                   // proto: IPv4
286    f[18] = 0x06;                                  // hw_size
287    f[19] = 0x04;                                  // proto_size
288    f[20] = 0x00; f[21] = 0x01;                   // op: request
289    f[22..28].copy_from_slice(&src_mac);           // sender MAC
290    f[28..32].copy_from_slice(&src_ip);            // sender IP
291    // target MAC = 0 (unknown)
292    f[38..42].copy_from_slice(&dst_ip);            // target IP
293    f
294}

296/// Ethernet 프레임 EtherType 추출
297pub fn ethertype(frame: &[u8]) -> u16 {
298    if frame.len() < 14 { return 0; }
299    ((frame[12] as u16) << 8) | frame[13] as u16
300}
301
302/// ARP 응답에서 발신자 MAC + IP 추출
303pub fn parse_arp_reply(frame: &[u8]) -> Option<([u8; 6], [u8; 4])> {
304    if frame.len() < 42 { return None; }
305    if ethertype(frame) != 0x0806 { return None; }
306    let op = ((frame[20] as u16) << 8) | frame[21] as u16;
307    if op != 2 { return None; } // op=2: ARP reply
308    let mut mac = [0u8; 6]; mac.copy_from_slice(&frame[22..28]);
309    let mut ip  = [0u8; 4]; ip.copy_from_slice(&frame[28..32]);
310    Some((mac, ip))
311}
