1//! GUI 터미널 버퍼 — MuShell 창 (window index 1) 전용
2//!
3//! `sys_write(fd=1/2, ...)` 호출 시 이 버퍼에 텍스트가 쌓이고,
4//! `repaint()`가 호출되면 창 1의 콘텐츠 영역에 렌더링된다.
5//!
6//! ## 처리하는 제어 문자
7//! - `\n` (0x0A): 줄 바꿈 + 스크롤
8//! - `\r` (0x0D): 커서를 줄 처음으로
9//! - `\x08` (0x08): 커서 왼쪽 이동 (backspace)
10//! - `\x1B` (ESC): ANSI 시퀀스 skip (단순 무시)
11
12use core::sync::atomic::{AtomicU8, AtomicUsize, Ordering};
13
14// ── 그리드 크기 ─────────────────────────────────────────────────────────────
15//
16// 창 1: w=290, h=200
17// 콘텐츠: x+5, y+21, w≈280, h≈174
18// 8×8 폰트, 줄 간격 10px → 34 cols × 17 rows
19
20const COLS: usize = 34;
21const ROWS: usize = 17;
22const TOTAL: usize = COLS * ROWS;
23
24// ── 정적 버퍼 ────────────────────────────────────────────────────────────────
25
26static TERM_BUF: [AtomicU8;  TOTAL] = [const { AtomicU8::new(b' ') }; TOTAL];
27static TERM_COL: AtomicUsize = AtomicUsize::new(0);
28static TERM_ROW: AtomicUsize = AtomicUsize::new(0);
29
30// ESC 시퀀스 skip 상태
31static ESC_STATE: AtomicU8 = AtomicU8::new(0); // 0=normal,1=ESC,2=CSI
32
33// ── 버퍼 접근 ────────────────────────────────────────────────────────────────
34
35#[inline]
36fn cell(r: usize, c: usize) -> &'static AtomicU8 {
37    &TERM_BUF[r * COLS + c]
38}
39
40// ── 스크롤 ───────────────────────────────────────────────────────────────────
41
42fn scroll_up() {
43    for r in 0..ROWS - 1 {
44        for c in 0..COLS {
45            let b = cell(r + 1, c).load(Ordering::Relaxed);
46            cell(r, c).store(b, Ordering::Relaxed);
47        }
48    }
49    for c in 0..COLS {
50        cell(ROWS - 1, c).store(b' ', Ordering::Relaxed);
51    }
52    TERM_ROW.store(ROWS - 1, Ordering::Relaxed);
53}
54
55// ── 바이트 쓰기 ──────────────────────────────────────────────────────────────
56
57pub fn write_byte(b: u8) {
58    let esc = ESC_STATE.load(Ordering::Relaxed);
59    match esc {
60        1 => {
61            // ESC received — next byte determines sequence type
62            if b == b'[' {
63                ESC_STATE.store(2, Ordering::Relaxed); // CSI
64            } else {
65                ESC_STATE.store(0, Ordering::Relaxed); // unknown, reset
66            }
67            return;
68        }
69        2 => {
70            // CSI sequence — skip until letter (A-Z, a-z)
71            if b.is_ascii_alphabetic() {
72                ESC_STATE.store(0, Ordering::Relaxed);
73            }
74            return;
75        }
76        _ => {}
77    }
78
79    match b {
80        0x1B => {
81            ESC_STATE.store(1, Ordering::Relaxed);
82        }
83        b'\r' => {
84            TERM_COL.store(0, Ordering::Relaxed);
85        }
86        b'\n' => {
87            TERM_COL.store(0, Ordering::Relaxed);
88            let r = TERM_ROW.load(Ordering::Relaxed);
89            if r + 1 < ROWS {
90                TERM_ROW.store(r + 1, Ordering::Relaxed);
91            } else {
92                scroll_up();
93            }
94        }
95        0x08 => {
96            // 커서 왼쪽 이동 (erase-left 시퀀스 "\x08 \x08" 에서 첫/세 번째)
97            let col = TERM_COL.load(Ordering::Relaxed);
98            if col > 0 {
99                TERM_COL.store(col - 1, Ordering::Relaxed);
100            }
101        }
102        0x07 => {} // BEL 무시
103        b' '..=b'~' => {
104            let r = TERM_ROW.load(Ordering::Relaxed);
105            let c = TERM_COL.load(Ordering::Relaxed);
106            if r < ROWS && c < COLS {
107                cell(r, c).store(b, Ordering::Relaxed);
108                let nc = c + 1;
109                if nc >= COLS {
110                    // 줄 넘김
111                    TERM_COL.store(0, Ordering::Relaxed);
112                    let nr = r + 1;
113                    if nr < ROWS {
114                        TERM_ROW.store(nr, Ordering::Relaxed);
115                    } else {
116                        scroll_up();
117                    }
118                } else {
119                    TERM_COL.store(nc, Ordering::Relaxed);
120                }
121            }
122        }
123        _ => {} // 기타 제어 문자 무시
124    }
125}
126
127/// 여러 바이트를 쓰고 한 번에 repaint
128pub fn write_bytes(bytes: &[u8]) {
129    for &b in bytes {
130        write_byte(b);
131    }
132    repaint();
133}

137/// MuShell 창의 콘텐츠 영역에 현재 버퍼를 그린다.
138/// `cx, cy`: 콘텐츠 영역 시작 좌표 (픽셀)
139pub fn render_at(cx: u32, cy: u32) {
140    let bg = crate::fb::rgb(18, 18, 32);
141    let fg = crate::fb::rgb(220, 220, 220);
142    let cur_fg = crate::fb::rgb(0, 255, 128); // 커서 색
143
144    let cur_row = TERM_ROW.load(Ordering::Relaxed);
145    let cur_col = TERM_COL.load(Ordering::Relaxed);
146
147    // 콘텐츠 영역 배경 클리어
148    crate::fb::fill_rect(cx, cy, (COLS as u32) * 8, (ROWS as u32) * 10, bg);
149
150    for r in 0..ROWS {
151        for c in 0..COLS {
152            let ch = cell(r, c).load(Ordering::Relaxed);
153            let px = cx + (c as u32) * 8;
154            let py = cy + (r as u32) * 10;
155
156            // 현재 커서 위치 표시 (블록)
157            if r == cur_row && c == cur_col {
158                crate::fb::draw_char(px, py, if ch == b' ' { b'_' } else { ch }, cur_fg, bg);
159            } else if ch != b' ' {
160                crate::fb::draw_char(px, py, ch, fg, bg);
161            }
162        }
163    }
164}
165
166/// wm.rs가 shell 창 위치를 줄 때 쓰는 버전 (wm에서 호출)
167pub fn repaint() {
168    if let Some((cx, cy)) = crate::wm::shell_content_origin() {
169        render_at(cx, cy);
170    }
