1//! PCI 설정 공간 접근 (Config Mechanism #1)
2//!
3//! CONFIG_ADDRESS (0xCF8): 버스/디바이스/함수/오프셋 선택
4//! CONFIG_DATA    (0xCFC): 선택된 레지스터 읽기/쓰기
5//!
6//! 포트 I/O 헬퍼(outl/inl/outw/inw/outb/inb)도 여기서 제공.
7
8const ADDR_PORT: u16 = 0xCF8;
9const DATA_PORT: u16 = 0xCFC;
10
11fn mk_addr(bus: u8, dev: u8, func: u8, off: u8) -> u32 {
12    (1u32 << 31)
13        | ((bus  as u32) << 16)
14        | ((dev  as u32) << 11)
15        | ((func as u32) <<  8)
16        | ((off  as u32) & 0xFC)
17}
18
19// ── PCI 설정 읽기/쓰기 ──────────────────────────────────────────────────────
20
21pub fn read32(bus: u8, dev: u8, func: u8, off: u8) -> u32 {
22    unsafe { outl(ADDR_PORT, mk_addr(bus, dev, func, off)); inl(DATA_PORT) }
23}
24
25pub fn read16(bus: u8, dev: u8, func: u8, off: u8) -> u16 {
26    (read32(bus, dev, func, off & !3) >> ((off & 2) * 8)) as u16
27}
28
29pub fn write32(bus: u8, dev: u8, func: u8, off: u8, val: u32) {
30    unsafe { outl(ADDR_PORT, mk_addr(bus, dev, func, off)); outl(DATA_PORT, val); }
31}
32
33pub fn write16(bus: u8, dev: u8, func: u8, off: u8, val: u16) {
34    let old = read32(bus, dev, func, off & !3);
35    let shift = (off & 2) * 8;
36    let new = (old & !(0xFFFF << shift)) | ((val as u32) << shift);
37    write32(bus, dev, func, off & !3, new);
38}

40// ── 디바이스 탐색 ─────────────────────────────────────────────────────────────

42/// PCI 설정 공간에서 vendor/device ID로 디바이스 검색.
43/// 반환: (bus, dev) 또는 None
44pub fn find_device(vendor: u16, device: u16) -> Option<(u8, u8)> {
45    for bus in 0u8..=255 {
46        for dev in 0u8..32 {
47            let v = read32(bus, dev, 0, 0x00);
48            if v == 0xFFFF_FFFF { continue; }
49            let vid = (v & 0xFFFF) as u16;
50            let did = (v >> 16) as u16;
51            if vid == vendor && did == device {
52                return Some((bus, dev));
53            }
54        }
55    }
56    None
57}

59/// BAR(idx) I/O 포트 베이스 주소 (bit0=1이면 I/O 공간).
60/// 하위 2비트를 마스크해서 반환.
61pub fn bar_io_base(bus: u8, dev: u8, bar_idx: u8) -> Option<u16> {
62    let bar = read32(bus, dev, 0, 0x10 + bar_idx * 4);
63    if bar & 1 == 1 { Some((bar & !3) as u16) } else { None }
64}

66/// Command 레지스터에 I/O Space Enable + Bus Master Enable 설정.
67pub fn enable_io_and_busmaster(bus: u8, dev: u8) {
68    let cmd = read16(bus, dev, 0, 0x04);
69    write16(bus, dev, 0, 0x04, cmd | 0x0005); // bit0=I/O, bit2=BusMaster
70}

74#[inline]
75pub unsafe fn outl(port: u16, val: u32) {
76    core::arch::asm!("out dx, eax", in("dx") port, in("eax") val, options(nomem, nostack));
77}
78
79#[inline]
80pub unsafe fn inl(port: u16) -> u32 {
81    let v: u32;
82    core::arch::asm!("in eax, dx", out("eax") v, in("dx") port, options(nomem, nostack));
83    v
84}
85
86#[inline]
87pub unsafe fn outw(port: u16, val: u16) {
88    core::arch::asm!("out dx, ax", in("dx") port, in("ax") val, options(nomem, nostack));
89}
90
91#[inline]
92pub unsafe fn inw(port: u16) -> u16 {
93    let v: u16;
94    core::arch::asm!("in ax, dx", out("ax") v, in("dx") port, options(nomem, nostack));
95    v
96}
97
98#[inline]
99pub unsafe fn outb(port: u16, val: u8) {
100    core::arch::asm!("out dx, al", in("dx") port, in("al") val, options(nomem, nostack));
101}
102
103#[inline]
104pub unsafe fn inb(port: u16) -> u8 {
105    let v: u8;
106    core::arch::asm!("in al, dx", out("al") v, in("dx") port, options(nomem, nostack));
107    v
