1//! 프로세스 서브시스템
2//!
3//! ## 구조
4//! - `mod.rs`       — Process PCB, Message, ProcessState 타입 정의
5//! - `context.rs`   — CPU 레지스터 스냅샷 (참조용, 현재는 직접 미사용)
6//! - `scheduler.rs` — 선점형 라운드로빈 스케줄러 (ALPHA M1)
7//! - `ipc.rs`       — 복사 기반 소형 메시지 IPC
8//! - `ipc_cap.rs`   — Zero-copy capability IPC (ALPHA M2)
9//!
10//! ## ALPHA M1 이후 컨텍스트 스위치 방식
11//!
12//! 기존(협력형): `switch_context_asm`으로 callee-saved 레지스터만 교환.
13//! 새(선점형):   타이머 ISR의 어셈블리 스텁이 ALL 레지스터를 스택에 저장.
14//!               Rust의 `preempt(rsp) -> rsp` 함수가 스택 포인터만 교환.
15//!               `iretq`가 새 프로세스의 스택에서 RIP/CS/RFLAGS를 꺼내 복귀.
16//!
17//! ## 초기 스택 프레임 레이아웃 (새 프로세스, ring0 커널 스레드)
18//!
19//! 타이머 ISR(ring0 인터럽트)이 스택에 쌓는 구조를 미리 흉내냄.
20//! CPU가 ring0 인터럽트 시 push하는 항목: RFLAGS, CS, RIP (3개, 24바이트)
21//! ISR 스텁이 추가로 push: rax, rcx, rdx, rsi, rdi, r8-r11, rbx, rbp, r12-r15 (15개)
22//!
23//! ```text
24//! [stack_top - 8]   = RFLAGS = 0x202     ← 인터럽트 활성화(IF=1)
25//! [stack_top - 16]  = CS     = 0x08      ← 커널 코드 세그먼트
26//! [stack_top - 24]  = RIP    = entry_fn  ← iretq가 여기로 점프
27//! [stack_top - 32]  = rax    = 0
28//! [stack_top - 40]  = rcx    = 0
29//! [stack_top - 48]  = rdx    = 0
30//! [stack_top - 56]  = rsi    = 0
31//! [stack_top - 64]  = rdi    = 0
32//! [stack_top - 72]  = r8     = 0
33//! [stack_top - 80]  = r9     = 0
34//! [stack_top - 88]  = r10    = 0
35//! [stack_top - 96]  = r11    = 0
36//! [stack_top - 104] = rbx    = 0
37//! [stack_top - 112] = rbp    = 0
38//! [stack_top - 120] = r12    = 0
39//! [stack_top - 128] = r13    = 0
40//! [stack_top - 136] = r14    = 0
41//! [stack_top - 144] = r15    = 0         ← preempt_rsp 가 여기를 가리킴
42//! ```
43//!
44//! `iretq` 직전 RSP = preempt_rsp + 120 → RIP/CS/RFLAGS 차례로 팝 → entry_fn 실행 시작.
45

57/// 프로세스 ID
58pub type Pid = u64;
59
60/// 커널 스택 크기: 64KB
61const KERNEL_STACK_SIZE: usize = 64 * 1024;
62
63/// ISR 스텁이 저장하는 레지스터 수 (15개 범용 레지스터)
64const ISR_REG_COUNT: usize = 15;
65/// ring0 iretq 프레임 크기 (RIP + CS + RFLAGS = 3 × 8바이트)
66const IRETQ_FRAME_SIZE: usize = 3;
67/// 초기 선점형 컨텍스트 프레임 총 크기 (바이트)
68///
69/// = (ISR_REG_COUNT + IRETQ_FRAME_SIZE) × 8
70pub const PREEMPT_FRAME_BYTES: usize = (ISR_REG_COUNT + IRETQ_FRAME_SIZE) * 8; // 144
71
72/// 프로세스 실행 상태
73#[derive(Debug, Clone, Copy, PartialEq, Eq)]
74pub enum ProcessState {
75    Ready,
76    Running,
77    Blocked,
78    Dead,
79}
80
81/// 스케줄링 우선순위 (ALPHA 5)
82///
83/// I/O바운드 → High (자주 선택), CPU바운드 → Low (드물게 선택).
84/// Linux CFS의 interactive task detection과 동일한 방향.
85#[repr(u8)]
86#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
87pub enum Priority {
88    Idle   = 0, // kernel idle — hlt 대기, 스케줄 최하위
89    Low    = 1, // CPU바운드 — 드물게 선택
90    Normal = 2, // 기본값
91    High   = 3, // I/O바운드, 키보드 포그라운드 — 자주 선택
92}
93
94impl Priority {
95    pub fn from_u8(v: u8) -> Self {
96        match v {
97            0 => Priority::Idle,
98            1 => Priority::Low,
99            3 => Priority::High,
100            _ => Priority::Normal,
101        }
102    }
103    pub fn name(self) -> &'static str {
104        match self {
105            Priority::Idle   => "Idle",
106            Priority::Low    => "Low",
107            Priority::Normal => "Normal",
108            Priority::High   => "High",
109        }
110    }
111}

122/// 프로세스 제어 블록 (PCB)
123pub struct Process {
124    pub pid: Pid,
125    pub name: &'static str,
126    pub state: ProcessState,
127    /// 스케줄링 우선순위 (ALPHA 5)
128    pub priority: Priority,
129    /// 선점형 컨텍스트 스위치에 사용되는 저장된 스택 포인터.
130    pub preempt_rsp: u64,
131    /// 스택 메모리 (힙에 할당, 드롭 시 자동 해제)
132    pub kernel_stack: Vec<u8>,
133    /// IPC 수신 큐
134    pub message_queue: VecDeque<Message>,
135
136    // ── ALPHA 4: 스케줄링 행동 분류 카운터 ───────────────────────────────
137    /// 타이머에 의해 강제 선점된 횟수 → CPU바운드 지표
138    pub forced_preempts: u64,
139    /// 자발적으로 CPU를 양보한 횟수 (yield_now) → I/O바운드 지표
140    pub voluntary_yields: u64,
141    /// 키보드 부스트 잔여 틱 (0이면 부스트 없음, ALPHA 6)
142    pub boost_ticks: u64,
143    /// Ready 상태로 진입한 틱 (aging 대기 시간 측정, ALPHA 7)
144    pub ready_since_tick: u64,
145    // ── ALPHA 9: Capability Handle Table ─────────────────────────────────────
146    /// Linux fd ↔ Handle ↔ Capability<T> 연결 테이블
147    pub handle_table: handle::HandleTable,
148}
149
150impl Process {
151    /// 새 커널 스레드 프로세스 생성.
152    ///
153    /// 초기 스택에 "타이머 ISR이 중단한 것처럼" 보이는 가짜 프레임을 구성한다.
154    /// 이 프로세스가 처음 스케줄되면:
155    ///   1. ISR 스텁이 `mov rsp, preempt_rsp` → 이 스택으로 전환
156    ///   2. `pop r15..rax` → rax = entry_fn, 나머지 0으로 복원
157    ///   3. `iretq` → RIP=process_start(트램폴린), CS=0x08, RFLAGS=0x202
158    ///   4. process_start 트램폴린: 'T' 출력 후 `jmp rax` → entry_fn 실행 시작
159    pub fn new(pid: Pid, name: &'static str, entry_fn: fn() -> !) -> Self {
160        // process_start: handlers.rs global_asm에 정의된 어셈블리 트램폴린
161        extern "C" { fn process_start(); }
162
163        let mut stack = Vec::with_capacity(KERNEL_STACK_SIZE);
164        stack.resize(KERNEL_STACK_SIZE, 0u8);
165        let stack_top = stack.as_ptr() as u64 + KERNEL_STACK_SIZE as u64;
166
167        // 스택의 상단에서 PREEMPT_FRAME_BYTES 만큼 내려와서 프레임 시작
168        let frame_start = stack_top - PREEMPT_FRAME_BYTES as u64;
169        let frame = frame_start as *mut u64;
170
171        unsafe {
172            // 인덱스 0..13 = r15..rcx → 0
173            for i in 0..ISR_REG_COUNT - 1 {
174                *frame.add(i) = 0;
175            }
176            // 인덱스 5 = rbx: stack_top 저장 (process_start 트램폴린이 `mov rsp, rbx`로 RSP 복원)
177            // iretq가 ring-change 팝(5개)을 수행하는 경우 RSP=0이 될 수 있으므로
178            // rbx에 stack_top을 미리 적어 두어 process_start에서 명시적으로 RSP를 교정.
179            *frame.add(5) = stack_top;
180            // 인덱스 14 = rax: entry_fn 주소 (트램폴린이 `jmp rax`로 진입)
181            *frame.add(14) = entry_fn as u64;
182            // 인덱스 15 = RIP: process_start 트램폴린 (iretq가 여기로 점프)
183            *frame.add(15) = process_start as u64;
184            // 인덱스 16 = CS: 커널 코드 세그먼트 (DPL=0)
185            *frame.add(16) = KERNEL_CODE_SEL as u64;
186            // 인덱스 17 = RFLAGS: 인터럽트 활성(IF=1), 다른 비트는 기본값
187            *frame.add(17) = 0x202;
188        }
189
190        Process {
191            pid,
192            name,
193            state: ProcessState::Ready,
194            priority: Priority::Normal,
195            preempt_rsp: frame_start,
196            kernel_stack: stack,
197            message_queue: VecDeque::new(),
198            forced_preempts: 0,
199            voluntary_yields: 0,
200            boost_ticks: 0,
201            ready_since_tick: 0,
202            handle_table: handle::HandleTable::new(),
203        }
204    }
205
206    pub fn new_kernel_main() -> Self {
207        Process {
208            pid: 0,
209            name: "kernel_main",
210            state: ProcessState::Running,
211            priority: Priority::Normal, // Policy Engine이 hlt 루프 진입 후 Idle로 강등
212            preempt_rsp: 0,
213            kernel_stack: Vec::new(),
214            message_queue: VecDeque::new(),
215            forced_preempts: 0,
216            voluntary_yields: 0,
217            boost_ticks: 0,
218            ready_since_tick: 0,
219            handle_table: handle::HandleTable::new(),
220        }
221    }
