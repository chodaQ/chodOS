1//! Zero-copy Capability IPC (ALPHA M2)
2//!
3//! ## 기존 IPC의 문제
4//!
5//! `ipc.rs`의 Message는 최대 64바이트만 전달 가능하며 데이터를 복사함.
6//! 대용량 데이터(파일 내용, 네트워크 패킷 등)를 전달하려면 복사 비용이 큼.
7//!
8//! ## 해결책: Capability 토큰 기반 공유 버퍼
9//!
10//! ```text
11//! 송신자                         수신자
12//!   │                              │
13//!   ├─ alloc_shared(data) ──────►  │        (SHARED_BUFFERS에 버퍼 등록)
14//!   │   └► cap_id (u64)            │
15//!   │                              │
16//!   ├─ send_cap(to, cap_id) ─────► │        (기존 IPC로 cap_id 전달)
17//!   │                              │
18//!   │                    recv_cap()├─►       (cap_id 수신)
19//!   │                              │
20//!   │                read_shared(cap_id)─►   (버퍼에 직접 접근, 복사 없음)
21//!   │                              │
22//!   │                drop_shared(cap_id)─►   (버퍼 해제 + 권한 소멸)
23//! ```
24//!
25//! ## Zero-copy 보장
26//!
27//! 데이터는 `SHARED_BUFFERS`의 `Vec<u8>` 하나에만 존재함.
28//! 송신자 → 수신자 경로에서 데이터 복사가 **전혀** 없음.
29//! 전달되는 것은 64비트 정수 `CapId` (토큰) 뿐.
30//!
31//! ## Capability vs 포인터
32//!
33//! 단순 포인터와 달리 CapId는:
34//! - 소유권: 명시적으로 `drop_shared`하기 전까지는 버퍼가 살아있음
35//! - 추적: 누가 어떤 버퍼를 소유하는지 SHARED_BUFFERS 테이블이 기록
36//! - 향후: 다중 수신자, 접근 권한(RO/RW) 확장 가능
37//!
38//! ## 현재 제한
39//!
40//! - 커널 힙 기반 (가상 주소 공유, 물리 페이지 매핑은 아님)
41//! - 단일 코어 전용 (spin::Mutex로 보호되지만 interrupt-context 주의)
42//! - 최대 동시 공유 버퍼: `MAX_SHARED` 개 (고정 테이블)
43
44use alloc::collections::BTreeMap;
45use alloc::vec::Vec;
46use core::sync::atomic::{AtomicU64, Ordering};
47use spin::Mutex;
48
49use super::Pid;
50
51/// 공유 버퍼 식별자 (Capability 토큰)
52pub type CapId = u64;
53
54/// 전역 고유 ID 카운터
55static NEXT_CAP_ID: AtomicU64 = AtomicU64::new(1);
56
57/// 개별 공유 버퍼 메타데이터
58pub struct SharedBuffer {
59    /// 이 버퍼의 고유 ID
60    pub id: CapId,
61    /// 실제 데이터 (힙에 할당)
62    pub data: Vec<u8>,
63    /// 버퍼를 생성한 프로세스
64    pub owner: Pid,
65}
66
67/// 전역 공유 버퍼 테이블
68///
69/// `BTreeMap<CapId, SharedBuffer>`: cap_id → 버퍼 매핑.
70/// `Mutex`: interrupt context에서 접근하지 않으므로 spin lock 사용.
71static SHARED_BUFFERS: Mutex<BTreeMap<CapId, SharedBuffer>> =
72    Mutex::new(BTreeMap::new());
73
74// ── 공개 API ─────────────────────────────────────────────────────────────────
75
76/// 공유 버퍼 할당 — 데이터를 등록하고 CapId 반환.
77///
78/// 반환된 CapId를 `send_cap()`으로 수신자에게 전달하면
79/// 수신자가 `read_shared(cap_id)`로 데이터에 직접 접근할 수 있음.
80pub fn alloc_shared(owner: Pid, data: Vec<u8>) -> CapId {
81    let id = NEXT_CAP_ID.fetch_add(1, Ordering::Relaxed);
82    SHARED_BUFFERS.lock().insert(id, SharedBuffer { id, data, owner });
83    id
84}
85
86/// 공유 버퍼에 데이터를 복사 없이 추가 (append).
87///
88/// 버퍼가 존재하지 않으면 false 반환.
89pub fn append_shared(id: CapId, extra: &[u8]) -> bool {
90    let mut table = SHARED_BUFFERS.lock();
91    if let Some(buf) = table.get_mut(&id) {
92        buf.data.extend_from_slice(extra);
93        true
94    } else {
95        false
96    }
97}
98
99/// Capability 전달 — 기존 IPC 채널을 통해 cap_id를 수신자에게 전송.
100///
101/// `cap_id`를 8바이트 데이터로 포장해서 기존 메시지 큐에 넣음.
102/// 데이터 자체는 `SHARED_BUFFERS`에 그대로 있음 — 복사 없음.
103pub fn send_cap(from: Pid, to: Pid, cap_id: CapId) -> bool {
104    let _ = from; // 현재는 scheduler::current_pid()로 대체됨
105    // cap_id를 리틀엔디안 8바이트로 직렬화해서 기존 IPC 채널로 전달
106    super::ipc::send(to, &cap_id.to_le_bytes())
107}
108
109/// 현재 프로세스의 IPC 큐에서 CapId 수신.
110///
111/// 반환: Some(cap_id) 또는 None (큐가 비어있거나 타입 불일치)
112pub fn recv_cap() -> Option<CapId> {
113    use super::ipc::recv;
114    let msg = recv()?;
115    if msg.len != 8 {
116        return None;
117    }
118    let mut bytes = [0u8; 8];
119    bytes.copy_from_slice(&msg.data[0..8]);
120    Some(CapId::from_le_bytes(bytes))
121}
122
123/// 공유 버퍼에 콜백 방식으로 접근 (zero-copy 읽기).
124///
125/// `f(data: &[u8])`: 버퍼가 잠긴 동안 `data`를 읽을 수 있는 클로저.
126/// 클로저 실행 중 데이터가 이동하거나 복사되지 않음.
127///
128/// 반환: `Some(f의 반환값)` 또는 `None` (CapId 없음)
129pub fn read_shared<R>(id: CapId, f: impl FnOnce(&[u8]) -> R) -> Option<R> {
130    let table = SHARED_BUFFERS.lock();
131    let buf = table.get(&id)?;
132    Some(f(&buf.data))
133}
134
135/// 공유 버퍼 해제 — CapId 소멸, 메모리 반환.
136///
137/// 이후 이 CapId를 사용하는 모든 접근은 None 반환.
138pub fn drop_shared(id: CapId) -> bool {
139    SHARED_BUFFERS.lock().remove(&id).is_some()
140}
141
142/// 현재 등록된 공유 버퍼 수 (디버깅용)
143pub fn buffer_count() -> usize {
144    SHARED_BUFFERS.lock().len()
145}
146
147/// 특정 CapId가 유효한지 확인
148pub fn is_valid(id: CapId) -> bool {
149    SHARED_BUFFERS.lock().contains_key(&id)
150}
