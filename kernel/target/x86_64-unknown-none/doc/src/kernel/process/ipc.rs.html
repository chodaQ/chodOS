1//! IPC — 프로세스 간 메시지 전달 (Inter-Process Communication)
2//!
3//! ## 설계 원칙 (마이크로커널 관점)
4//!
5//! 마이크로커널에서 모든 프로세스 간 통신은 메시지로 이루어짐.
6//! 파일 읽기, 네트워크 요청, 드라이버 호출 모두 결국 메시지 전송.
7//!
8//! ## 현재 구현 (Milestone 3 — 단순 메시지 큐)
9//!
10//! 각 프로세스는 FIFO 메시지 큐를 가짐 (Process.message_queue).
11//! - send(to, data): to의 큐 끝에 추가
12//! - recv():         현재 프로세스의 큐 앞에서 꺼냄
13//!
14//! 비동기: send는 즉시 반환 (큐에 넣고 끝).
15//!         recv는 큐가 비어있으면 None 반환 (blocking 없음).
16//!
17//! ## 향후 계획 (Milestone 4+)
18//!
19//! - Capability 기반 채널: 권한 없으면 send 불가
20//! - Blocking recv: 메시지가 올 때까지 Blocked 상태로 대기
21//! - 응답 메시지(reply): call() = send + blocking recv
22//! - 크기 제한 큐: 가득 차면 send 실패 또는 sender를 block
23
24use super::scheduler;
25use super::{Message, Pid};
26
27/// 메시지 전송
28///
29/// `to`: 수신할 프로세스의 PID.
30/// `data`: 전송할 바이트 슬라이스 (최대 64바이트; 초과분은 버림).
31///
32/// # 반환값
33/// - `true`: 전송 성공 (수신자 큐에 메시지 추가됨)
34/// - `false`: 해당 PID의 프로세스 없음
35///
36/// # 비고
37/// 비동기 전송: 수신자가 아직 recv()를 호출하지 않아도 즉시 반환.
38/// 수신자 큐에 공간이 있는 한 실패하지 않음 (큐 크기 제한 없음).
39pub fn send(to: Pid, data: &[u8]) -> bool {
40    let sender = scheduler::current_pid();
41
42    // 메시지 구성 (64바이트 고정 크기 페이로드로 복사)
43    let mut msg = Message {
44        sender,
45        len: data.len().min(64),
46        data: [0u8; 64],
47    };
48    msg.data[..msg.len].copy_from_slice(&data[..msg.len]);
49
50    // 스케줄러를 통해 수신자 큐에 push
51    scheduler::send_msg(to, msg)
52}
53
54/// 메시지 수신 (현재 프로세스의 큐에서 꺼냄)
55///
56/// # 반환값
57/// - `Some(Message)`: 수신된 메시지
58/// - `None`:          큐가 비어있음 (메시지 없음)
59///
60/// # 비고
61/// 논블로킹: 큐가 비면 바로 None 반환.
62/// 블로킹이 필요하면 호출자가 직접 루프+yield 패턴을 써야 함:
63///
64/// ```rust
65/// loop {
66///     if let Some(msg) = ipc::recv() {
67///         // 처리
68///         break;
69///     }
70///     scheduler::yield_now(); // 메시지 올 때까지 다른 프로세스에 양보
71/// }
72/// ```
73pub fn recv() -> Option<Message> {
74    scheduler::recv_msg()
75}
76
77/// u64 값을 IPC 메시지로 전송하는 헬퍼
78pub fn send_u64(to: Pid, value: u64) -> bool {
79    send(to, &value.to_le_bytes())
80}
81
82/// IPC 메시지에서 u64 값 추출 헬퍼
83pub fn msg_as_u64(msg: &Message) -> u64 {
84    let mut bytes = [0u8; 8];
85    let len = msg.len.min(8);
86    bytes[..len].copy_from_slice(&msg.data[..len]);
87    u64::from_le_bytes(bytes)
88}
