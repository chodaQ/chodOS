1//! 페이징 서브시스템 (Paging Subsystem)
2//!
3//! ## x86_64 4단계 페이지 테이블 구조
4//!
5//! 가상 주소 48비트 분해:
6//! ```
7//! [63:48] 부호 확장 (반드시 비트47의 복사본)
8//! [47:39] PML4 인덱스 (9비트, 0-511)
9//! [38:30] PDPT  인덱스 (9비트, 0-511)
10//! [29:21] PD    인덱스 (9비트, 0-511)
11//! [20:12] PT    인덱스 (9비트, 0-511)
12//! [11:0]  페이지 내 오프셋 (12비트 = 4KB)
13//! ```
14//!
15//! 각 테이블: 512 × 8바이트 = 4096바이트 (= 1 프레임)
16//! PML4 → PDPT → PD → PT → 4KB 물리 프레임
17//!
18//! ## 주소 공간 분할
19//!
20//! ```
21//! 0x0000_0000_0000_0000  ← 유저 공간 시작
22//! ...
23//! 0x0000_7FFF_FFFF_FFFF  ← 유저 공간 끝 (PML4[0..255])
24//! [canonical hole]
25//! 0xFFFF_8000_0000_0000  ← 커널 공간 시작 (PML4[256..511])
26//! ...                       HHDM, 커널 코드/데이터
27//! 0xFFFF_FFFF_FFFF_FFFF
28//! ```
29//!
30//! ## Milestone 3.6 구현 범위
31//!
32//! - 커널 PML4: limine PML4의 상위 절반(인덱스 256-511) 복사 후 CR3 전환
33//! - 유저 PML4: 커널 절반 공유 + 유저 코드/스택 매핑
34//! - ring3 진입: IRETQ로 CPL 0 → CPL 3 전환
35//! - Syscall:    `int 0x80` → ring3 → ring0 → IRETQ ring3 복귀
36//! - Longjmp:    3회 syscall 후 커널 메인으로 복귀
37
38use core::arch::asm;
39use core::sync::atomic::{AtomicU32, Ordering};
40use crate::memory::{frame, hhdm_offset};
41use crate::interrupts::gdt;

43// ── 페이지 테이블 엔트리 플래그 ──────────────────────────────────────────────
44const PTE_PRESENT:  u64 = 1 << 0; // 페이지가 물리 메모리에 존재
45const PTE_WRITABLE: u64 = 1 << 1; // 쓰기 허용
46const PTE_USER:     u64 = 1 << 2; // ring3 접근 허용 (U/S 비트)
47// bit 7 = PS(Page Size): PD에서 1이면 2MB 거대 페이지 (현재 미사용)
48
49// ── 유저 프로세스 가상 주소 레이아웃 ─────────────────────────────────────────
50/// 유저 코드가 매핑되는 가상 주소 (64KB)
51const USER_CODE_VADDR: u64 = 0x0000_0000_1_0000;
52/// 유저 스택 최상단 주소 (1MB, 스택은 아래 방향으로 성장)
53const USER_STACK_TOP:  u64 = 0x0000_0000_10_0000;
54/// 유저 스택에 할당할 4KB 프레임 수 (= 16KB 스택)
55const USER_STACK_PAGES: usize = 4;
56
57// ── 전역 상태 ─────────────────────────────────────────────────────────────
58/// 커널 전용 페이지 테이블의 CR3 값 (유저 모드 복귀 시 CR3 복원에 사용)
59pub static mut KERNEL_CR3: u64 = 0;
60
61/// enter_user_demo() 진입 직전에 저장한 커널 메인 스택 포인터
62///
63/// syscall 핸들러가 3회 호출 후 이 값으로 RSP를 복원 (longjmp).
64pub static mut KERNEL_MAIN_RSP: u64 = 0;
65
66/// ring3 → ring0 인터럽트/예외 진입 시 CPU가 사용할 전용 스택
67///
68/// CPU는 ring3에서 인터럽트 발생 시 TSS.RSP0에 설정된 주소로 스택을 전환함.
69/// 이 버퍼의 최상단(+8192)이 TSS.RSP0에 설정됨.
70static mut USER_KERNEL_STACK: [u8; 8192] = [0; 8192];
71
72// ── 페이지 테이블 타입 ────────────────────────────────────────────────────
73/// x86_64 페이지 테이블 (PML4 / PDPT / PD / PT 모두 동일 구조)
74///
75/// `#[repr(C, align(4096))]`: 4KB 경계 정렬 필수 (CR3 로드 요구사항).
76/// 할당 후 반드시 0으로 초기화해야 함 (비어있음 = Not Present = 0).
77#[repr(C, align(4096))]
78struct PageTable([u64; 512]);
79
80/// 물리 프레임을 할당하고 0으로 초기화한 PageTable 포인터 반환
81///
82/// 반환값: 가상 주소(HHDM) 기반 포인터 (물리 주소 + HHDM 오프셋)
83fn alloc_table() -> *mut PageTable {
84    let phys = frame::alloc_frame().expect("OOM: page table frame");
85    let virt = phys + hhdm_offset();
86    unsafe {
87        // 새 페이지 테이블은 반드시 0으로 채워야 함
88        // (0 = Present 비트 클리어 = 해당 엔트리 미사용)
89        core::ptr::write_bytes(virt as *mut u8, 0, 4096);
90        virt as *mut PageTable
91    }
92}
93
94/// 물리 주소 → 가상 주소(HHDM) 기반 PageTable 가변 참조
95unsafe fn table_at(phys: u64) -> &'static mut PageTable {
96    &mut *((phys + hhdm_offset()) as *mut PageTable)
97}

99/// 가상 주소를 4KB 물리 페이지에 매핑
100///
101/// PML4 → PDPT → PD → PT 계층을 순서대로 탐색.
102/// 중간 테이블이 없으면 새 프레임을 할당해 생성.
103///
104/// # 인수
105/// - `pml4`: 루트 페이지 테이블 (가상 주소 포인터)
106/// - `vaddr`: 매핑할 가상 주소 (4KB 정렬이어야 함)
107/// - `paddr`: 매핑할 물리 주소 (4KB 정렬이어야 함)
108/// - `flags`: `PTE_WRITABLE`, `PTE_USER` 등 (PTE_PRESENT 자동 추가)
109pub unsafe fn map_4k(pml4: *mut PageTable, vaddr: u64, paddr: u64, flags: u64) {
110    // 가상 주소를 각 레벨의 9비트 인덱스로 분해
111    let i4 = ((vaddr >> 39) & 0x1FF) as usize; // PML4 인덱스
112    let i3 = ((vaddr >> 30) & 0x1FF) as usize; // PDPT 인덱스
113    let i2 = ((vaddr >> 21) & 0x1FF) as usize; // PD   인덱스
114    let i1 = ((vaddr >> 12) & 0x1FF) as usize; // PT   인덱스
115
116    // 중간 테이블 엔트리에는 PTE_USER를 전파해야 함.
117    // U/S 비트는 계층 전체에 AND로 적용되기 때문에
118    // 상위 테이블 엔트리에 U/S=0이 있으면 하위 레벨이 U/S=1이어도 supervisor만 접근.
119    let mid_flags = PTE_PRESENT | PTE_WRITABLE | PTE_USER;
120
121    let pdpt = get_or_create(&mut (*pml4).0[i4], mid_flags);
122    let pd   = get_or_create(&mut (*pdpt).0[i3], mid_flags);
123    let pt   = get_or_create(&mut (*pd).0[i2],   mid_flags);
124
125    // 최종 4KB 페이지 매핑 (PTE_PRESENT 자동 추가)
126    (*pt).0[i1] = (paddr & !0xFFF) | flags | PTE_PRESENT;
127}
128
129/// 페이지 테이블 엔트리가 없으면 새 테이블 할당 후 엔트리에 기록
130unsafe fn get_or_create(entry: &mut u64, flags: u64) -> *mut PageTable {
131    if *entry & PTE_PRESENT == 0 {
132        let new_tbl = alloc_table();
133        // 물리 주소 = 가상 주소 - HHDM 오프셋
134        let phys = new_tbl as u64 - hhdm_offset();
135        *entry = phys | flags;
136    }
137    // 엔트리 상위 52비트 = 물리 주소, 하위 12비트 = 플래그
138    table_at(*entry & !0xFFF)
139}

143/// 페이징 서브시스템 초기화
144///
145/// ## 동작
146/// 1. 현재 CR3(limine의 PML4)에서 커널 절반(인덱스 256-511) 복사
147/// 2. 새 커널 PML4로 CR3 전환
148/// 3. TSS.RSP0 = USER_KERNEL_STACK 최상단
149///    (ring3 → ring0 전환 시 CPU가 이 주소로 스택을 자동 교체)
150///
151/// ## 왜 새 PML4를 만드는가?
152/// limine의 PML4를 그대로 쓰면 우리가 유저 공간(인덱스 0-255)을
153/// 제어할 수 없음. 새 PML4를 만들어야 프로세스별 유저 매핑을 독립적으로 관리 가능.
154pub fn init() {
155    unsafe {
156        // 현재 CR3 읽기 (limine이 설정한 PML4의 물리 주소)
157        let old_cr3: u64;
158        asm!("mov {}, cr3", out(reg) old_cr3,
159            options(nomem, nostack, preserves_flags));
160
161        // 새 커널 PML4 할당 (유저 절반 0-255는 0으로 비워둠)
162        let new_pml4 = alloc_table();
163        let new_cr3  = new_pml4 as u64 - hhdm_offset();
164
165        // limine PML4의 커널 절반(256-511)을 새 PML4에 복사.
166        // 이 범위에 HHDM, 커널 코드/데이터, 스택 매핑이 있음.
167        let old_pml4 = table_at(old_cr3 & !0xFFF);
168        for i in 256..512usize {
169            (*new_pml4).0[i] = old_pml4.0[i];
170        }
171
172        // CR3 전환 (= TLB 전체 플러시)
173        asm!("mov cr3, {}", in(reg) new_cr3,
174            options(nomem, nostack, preserves_flags));
175
176        KERNEL_CR3 = new_cr3;
177
178        // TSS.RSP0 설정: ring3에서 인터럽트 발생 시 CPU가 사용할 스택
179        // 스택은 높은 주소에서 낮은 방향으로 성장하므로 버퍼 최상단 설정
180        let rsp0 = USER_KERNEL_STACK.as_ptr().add(8192) as u64;
181        gdt::set_tss_rsp0(rsp0);
182        crate::interrupts::init_syscall(rsp0);
183    }
184
185    crate::serial_println!("[paging] Kernel PML4 built and CR3 switched.");
186}

190/// 유저 프로세스 주소 공간 생성 후 ring3으로 진입 (반환하지 않음)
191///
192/// ## 동작 순서
193/// 1. 유저 PML4 생성 (커널 절반 공유)
194/// 2. 유저 코드 프레임 할당 → `code` 바이트 복사 → `USER_CODE_VADDR`에 매핑
195/// 3. 유저 스택 프레임 할당 → `USER_STACK_TOP` 아래에 매핑
196/// 4. 커널 메인 RSP 저장 (longjmp 복귀 지점)
197/// 5. CR3 → 유저 PML4, IRETQ → ring3
198///
199/// ## 유저 코드 규약
200/// - 진입점: `USER_CODE_VADDR` (코드 바이트의 오프셋 0)
201/// - 스택: `USER_STACK_TOP`에서 시작, 아래로 성장
202/// - Syscall: `int 0x80` 사용
203///
204/// ## 반환
205/// 이 함수는 `options(noreturn)` asm으로 끝나며 절대 반환하지 않음.
206/// `syscall_handler()`가 3번째 호출 시 longjmp로 `after_user_demo()`에 도달.
207/// ELF64 바이너리를 유저 주소 공간에 로드한 뒤 ring3으로 진입 (ALPHA 14)
208///
209/// ## 동작 순서
210/// 1. 유저 PML4 생성 (커널 절반 공유)
211/// 2. ELF PT_LOAD 세그먼트별로 물리 프레임 할당 → 파일 내용 복사 → 가상 주소 매핑
212/// 3. 유저 스택 할당 (USER_STACK_TOP 아래)
213/// 4. KERNEL_MAIN_RSP 저장 (longjmp 복귀 지점)
214/// 5. IRETQ → ring3, 진입점 = ELF e_entry
215pub unsafe fn enter_elf(elf_data: &[u8]) -> ! {
216    use crate::elf::Elf64;
217
218    let elf = Elf64::parse(elf_data).expect("[elf] invalid ELF64 binary");
219    let entry = elf.entry();
220
221    crate::serial_println!("[elf] ELF64 parsed: entry={:#x}", entry);
222
223    // ── 유저 PML4 ────────────────────────────────────────────────────────
224    let user_pml4 = alloc_table();
225    let kpml4 = table_at(KERNEL_CR3 & !0xFFF);
226    for i in 256..512usize {
227        (*user_pml4).0[i] = kpml4.0[i];
228    }
229    let user_cr3 = user_pml4 as u64 - hhdm_offset();
230
231    // ── PT_LOAD 세그먼트 매핑 ────────────────────────────────────────────
232    for seg in elf.load_segments() {
233        crate::serial_println!(
234            "[elf] LOAD vaddr={:#x}  filesz={:#x}  memsz={:#x}  flags={:#x}",
235            seg.vaddr, seg.filesz, seg.memsz, seg.flags
236        );
237
238        // 세그먼트가 차지할 페이지 범위 (4KB 정렬)
239        let page_start = seg.vaddr & !0xFFF;
240        let page_end   = (seg.vaddr + seg.memsz as u64 + 0xFFF) & !0xFFF;
241
242        // 세그먼트 플래그 PF_W(2) 있으면 쓰기 허용
243        let pte_flags = PTE_USER | PTE_WRITABLE;
244
245        let mut vpage = page_start;
246        while vpage < page_end {
247            let phys = frame::alloc_frame().expect("OOM: ELF segment page");
248            let frame_virt = (phys + hhdm_offset()) as *mut u8;
249            // 페이지 전체를 0으로 초기화 (BSS 영역 처리)
250            core::ptr::write_bytes(frame_virt, 0, 4096);
251
252            // 이 페이지에 해당하는 파일 데이터 복사
253            // page_off_in_seg: 세그먼트 시작(vaddr)에서 현재 페이지의 상대 오프셋
254            let page_off_in_seg = (vpage.saturating_sub(page_start)) as usize;
255            let file_src_start  = seg.offset + page_off_in_seg;
256            let file_src_end    = file_src_start + 4096;
257            let copy_end        = file_src_end.min(seg.offset + seg.filesz).min(elf_data.len());
258
259            if file_src_start < copy_end {
260                let n = copy_end - file_src_start;
261                // 세그먼트 vaddr이 페이지 경계와 다를 경우 페이지 내 오프셋 조정
262                let dst_off = if vpage < seg.vaddr {
263                    (seg.vaddr - vpage) as usize
264                } else {
265                    0
266                };
267                core::ptr::copy_nonoverlapping(
268                    elf_data.as_ptr().add(file_src_start),
269                    frame_virt.add(dst_off),
270                    n,
271                );
272            }
273
274            map_4k(user_pml4, vpage, phys, pte_flags);
275            vpage += 4096;
276        }
277    }
278
279    // ── 유저 스택 ─────────────────────────────────────────────────────────
280    // i=0: USER_STACK_TOP - 4096 ~ USER_STACK_TOP (가장 높은 페이지, argv/auxv 위치)
281    let mut stack_top_phys = 0u64;
282    for i in 0..USER_STACK_PAGES {
283        let phys  = frame::alloc_frame().expect("OOM: ELF stack page");
284        let vaddr = USER_STACK_TOP - ((i + 1) as u64) * 4096;
285        map_4k(user_pml4, vaddr, phys, PTE_WRITABLE | PTE_USER);
286        if i == 0 { stack_top_phys = phys; }
287    }
288
289    // ── argv / envp / auxv 스택 셋업 ────────────────────────────────────
290    // Linux _start ABI: rsp → [argc][argv[0]..NULL][envp..NULL][auxv pairs][strings]
291    let user_sp = setup_elf_stack(stack_top_phys, hhdm_offset(), USER_STACK_TOP);
292
293    crate::serial_println!("[elf] address space ready. entering ring3 at {:#x} sp={:#x}...",
294        entry, user_sp);
295
296    // ── KERNEL_MAIN_RSP 저장 ──────────────────────────────────────────────
297    asm!(
298        "mov [{save}], rsp",
299        save = in(reg) &raw mut KERNEL_MAIN_RSP,
300        options(nostack, preserves_flags),
301    );
302
303    // ── IRETQ: ring0 → ring3 ─────────────────────────────────────────────
304    asm!(
305        "mov cr3, {user_cr3}",
306        "push {ss}",
307        "push {user_sp}",
308        "push {flags}",
309        "push {cs}",
310        "push {user_ip}",
311        "iretq",
312        user_cr3 = in(reg) user_cr3,
313        ss       = in(reg) gdt::USER_SS_RPL3 as u64,
314        user_sp  = in(reg) user_sp,
315        flags    = in(reg) 0x202u64,
316        cs       = in(reg) gdt::USER_CS_RPL3 as u64,
317        user_ip  = in(reg) entry,
318        options(noreturn),
319    );
320}

322/// Linux _start ABI 스택 셋업.
323///
324/// 스택 레이아웃 (높은 주소 → 낮은 주소):
325/// ```
326/// USER_STACK_TOP
327///   [strings: "prog\0"]           ← 16-byte 정렬
328///   [auxv: AT_CLKTCK=17/100, AT_PAGESZ=6/4096, AT_NULL=0/0]
329///   [envp: NULL]
330///   [argv: ptr_to_prog, NULL]
331///   [argc = 1]                    ← new rsp
332/// ```
333///
334/// 물리 페이지(phys)가 유저 가상 [top-4096, top)에 매핑돼 있으므로
335/// HHDM을 통해 커널에서 직접 쓴다.
336unsafe fn setup_elf_stack(phys: u64, hhdm: u64, top: u64) -> u64 {
337    // HHDM 가상 주소로 페이지 접근
338    let page_hhdm = (hhdm + phys) as *mut u8;
339    // 유저 가상 주소: [top - 4096, top)
340    let page_ubase = top - 4096;
341
342    // cursor: 페이지 내 byte 오프셋 (top에서 내려감)
343    let mut cur: usize = 4096;
344
345    // ── 문자열 영역 ──────────────────────────────────────────────────────
346    // "prog\0"
347    let prog_str = b"prog\0";
348    cur -= prog_str.len();
349    core::ptr::copy_nonoverlapping(prog_str.as_ptr(), page_hhdm.add(cur), prog_str.len());
350    let prog_ptr: u64 = page_ubase + cur as u64;
351
352    // 16-byte 정렬
353    cur &= !0xF;
354
355    // ── write_u64 클로저 ─────────────────────────────────────────────────
356    macro_rules! push {
357        ($v:expr) => {{
358            cur -= 8;
359            *(page_hhdm.add(cur) as *mut u64) = $v as u64;
360        }};
361    }
362
363    // ── auxv (역순: 마지막 pair부터) ─────────────────────────────────────
364    push!(0u64);   // AT_NULL value
365    push!(0u64);   // AT_NULL key
366
367    push!(100u64); // AT_CLKTCK value
368    push!(17u64);  // AT_CLKTCK key
369
370    push!(4096u64);// AT_PAGESZ value
371    push!(6u64);   // AT_PAGESZ key
372
373    // ── envp ──────────────────────────────────────────────────────────────
374    push!(0u64);   // NULL (envp 끝)
375
376    // ── argv ──────────────────────────────────────────────────────────────
377    push!(0u64);      // NULL (argv 끝)
378    push!(prog_ptr);  // argv[0] → "prog"
379
380    // ── argc ──────────────────────────────────────────────────────────────
381    push!(1u64);   // argc = 1
382
383    page_ubase + cur as u64 // 새 rsp (유저 가상 주소)
384}
385
386pub unsafe fn enter_user_demo(code: &[u8]) -> ! {
387    assert!(!code.is_empty() && code.len() <= 4096, "user code must fit in one page");
388
389    // ── 유저 PML4 생성 ───────────────────────────────────────────────────
390    let user_pml4 = alloc_table();
391
392    // 커널 절반 공유: ring3에서 발생한 인터럽트/예외 핸들러가
393    // 커널 코드와 스택에 접근할 수 있어야 하므로 상위 절반 엔트리를 복사.
394    let kpml4 = table_at(KERNEL_CR3 & !0xFFF);
395    for i in 256..512usize {
396        (*user_pml4).0[i] = kpml4.0[i];
397    }
398
399    let user_cr3 = user_pml4 as u64 - hhdm_offset();
400
401    // ── 유저 코드 매핑 ───────────────────────────────────────────────────
402    // 코드 페이지: USER(ring3 접근) + PRESENT, WRITABLE 없음(읽기/실행 전용)
403    // x86_64에서 기본 실행 허용(NX 비트 미사용).
404    let code_phys = frame::alloc_frame().expect("OOM: user code frame");
405    let code_virt = (code_phys + hhdm_offset()) as *mut u8;
406    core::ptr::copy_nonoverlapping(code.as_ptr(), code_virt, code.len());
407    map_4k(user_pml4, USER_CODE_VADDR, code_phys, PTE_USER);
408    // WRITABLE 없음 → 코드 영역은 읽기/실행 전용
409
410    // ── 유저 스택 매핑 ───────────────────────────────────────────────────
411    // 스택 페이지: USER + WRITABLE + PRESENT
412    // 스택은 USER_STACK_TOP 바로 아래 프레임부터 시작.
413    for i in 0..USER_STACK_PAGES {
414        let phys  = frame::alloc_frame().expect("OOM: user stack frame");
415        let vaddr = USER_STACK_TOP - ((i + 1) as u64) * 4096;
416        map_4k(user_pml4, vaddr, phys, PTE_WRITABLE | PTE_USER);
417    }
418
419    crate::serial_println!("[paging] User address space:");
420    crate::serial_println!("  code  → virt={:#x}  ({} bytes)", USER_CODE_VADDR, code.len());
421    crate::serial_println!("  stack → virt={:#x}..{:#x}  ({} pages)",
422        USER_STACK_TOP - (USER_STACK_PAGES as u64) * 4096,
423        USER_STACK_TOP, USER_STACK_PAGES);
424
425    // ── 커널 메인 RSP 저장 ───────────────────────────────────────────────
426    // syscall_handler()가 3회 호출 후 이 RSP로 복원 후 after_user_demo()로 점프.
427    // 저장 시점: 아래 IRETQ 프레임을 push하기 전이므로
428    //            복원 시 RSP는 IRETQ 프레임 이전 상태 → 유효한 커널 스택.
429    asm!(
430        "mov [{save}], rsp",
431        save = in(reg) &raw mut KERNEL_MAIN_RSP,
432        options(nostack, preserves_flags),
433    );
434
435    crate::serial_println!("[paging] Entering ring3 via IRETQ...");
436    crate::serial_println!("         code at virt={:#x}, stack top={:#x}",
437        USER_CODE_VADDR, USER_STACK_TOP);
438
439    // ── IRETQ: ring0 → ring3 ─────────────────────────────────────────────
440    //
441    // ring0에서 ring3으로 전환하려면 IRETQ 명령어 사용.
442    // 스택에 다음 순서로 push (IRETQ가 반대 순서로 pop):
443    //
444    //   [낮은주소] RIP     ← IRETQ 후 ring3 코드가 시작할 주소
445    //              CS      ← ring3 코드 세그먼트 셀렉터 (RPL=3 필수)
446    //              RFLAGS  ← 새 RFLAGS (IF=1로 인터럽트 허용)
447    //              RSP     ← ring3 스택 포인터
448    //   [높은주소] SS      ← ring3 스택 세그먼트 셀렉터 (RPL=3 필수)
449    //
450    // CS/SS의 RPL(Request Privilege Level)=3 필수:
451    //   RPL이 최종 CPL(Current Privilege Level)을 결정함.
452    //   RPL < 3이면 ring3로 전환되지 않음.
453    asm!(
454        "mov cr3, {user_cr3}",      // 유저 페이지 테이블로 전환
455        "push {ss}",                // SS  = 유저 데이터 세그먼트 (RPL=3)
456        "push {user_sp}",           // RSP = 유저 스택 포인터
457        "push {flags}",             // RFLAGS = IF=1 (인터럽트 허용)
458        "push {cs}",                // CS  = 유저 코드 세그먼트 (RPL=3)
459        "push {user_ip}",           // RIP = 유저 코드 진입점
460        "iretq",                    // ring3 전환 (CPL: 0 → 3)
461        user_cr3 = in(reg) user_cr3,
462        ss       = in(reg) gdt::USER_SS_RPL3 as u64,
463        user_sp  = in(reg) USER_STACK_TOP,
464        flags    = in(reg) 0x202u64,  // RFLAGS: IF=1 + 예약비트(bit1)
465        cs       = in(reg) gdt::USER_CS_RPL3 as u64,
466        user_ip  = in(reg) USER_CODE_VADDR,
467        options(noreturn),
468        // noreturn: IRETQ 이후 ring3 실행, 이 asm 블록으로 돌아오지 않음
469        // 복귀는 syscall_handler의 longjmp가 after_user_demo()로 점프
470    );
471}
