1//! PS/2 키보드 드라이버 — 스캔코드 Set 1 → ASCII 변환
2//!
3//! IRQ1 핸들러가 `push_key()`로 ASCII를 버퍼에 쌓으면,
4//! `sys_read(fd=0)`가 `read_key_blocking()`으로 꺼낸다.
5//! 시리얼(COM1)과 병행 동작 — GUI 모드/텍스트 모드 모두 지원.
6
7use core::sync::atomic::{AtomicBool, AtomicU8, AtomicUsize, Ordering};
8
9// ── 링 버퍼 ──────────────────────────────────────────────────────────────────
10
11const BUF_CAP: usize = 64;
12
13// no_std에서 AtomicU8 배열을 const 초기화
14struct KbdBuf {
15    buf:  [AtomicU8;   BUF_CAP],
16    head: AtomicUsize,
17    tail: AtomicUsize,
18}
19
20// SAFETY: 단일 코어 커널 — 인터럽트와 main이 번갈아 접근
21unsafe impl Sync for KbdBuf {}
22
23static KBD: KbdBuf = KbdBuf {
24    buf:  [const { AtomicU8::new(0) }; BUF_CAP],
25    head: AtomicUsize::new(0),
26    tail: AtomicUsize::new(0),
27};
28
29pub fn push_key(ascii: u8) {
30    let tail = KBD.tail.load(Ordering::Relaxed);
31    let next = (tail + 1) % BUF_CAP;
32    if next == KBD.head.load(Ordering::Relaxed) { return; } // 버퍼 풀
33    KBD.buf[tail].store(ascii, Ordering::Relaxed);
34    KBD.tail.store(next, Ordering::Relaxed);
35}
36
37pub fn try_pop() -> Option<u8> {
38    let head = KBD.head.load(Ordering::Relaxed);
39    if head == KBD.tail.load(Ordering::Relaxed) { return None; }
40    let byte = KBD.buf[head].load(Ordering::Relaxed);
41    KBD.head.store((head + 1) % BUF_CAP, Ordering::Relaxed);
42    Some(byte)
43}
44
45/// 버퍼에 데이터가 있는지 확인 (소비하지 않음) — select/poll/epoll용
46pub fn has_key() -> bool {
47    KBD.head.load(Ordering::Relaxed) != KBD.tail.load(Ordering::Relaxed)
48}
49
50/// 키보드 버퍼 또는 시리얼(COM1)에서 한 바이트를 블로킹으로 읽는다.
51pub fn read_key_blocking() -> u8 {
52    loop {
53        if let Some(k) = try_pop() { return k; }
54        if let Some(b) = crate::serial::try_read_byte() { return b; }
55        unsafe { core::arch::asm!("hlt", options(nomem, nostack, preserves_flags)); }
56    }
57}

59// ── Shift / Caps Lock 상태 ────────────────────────────────────────────────────
60
61static SHIFT:    AtomicBool = AtomicBool::new(false);
62static CAPS:     AtomicBool = AtomicBool::new(false);
63
64// ── 스캔코드 Set 1 → ASCII 테이블 (US QWERTY) ────────────────────────────────
65//
66// 인덱스 = 스캔코드(make code, 0x00-0x7F)
67// 0x00 = 없음(특수키), 나머지 = ASCII 값
68// Shift 테이블은 숫자/기호 행과 알파벳 대문자를 포함
69
70const LOWER: [u8; 128] = {
71    let mut t = [0u8; 128];
72    // 0x01: ESC
73    t[0x01] = 0x1B;
74    // 숫자 행
75    t[0x02] = b'1'; t[0x03] = b'2'; t[0x04] = b'3'; t[0x05] = b'4';
76    t[0x06] = b'5'; t[0x07] = b'6'; t[0x08] = b'7'; t[0x09] = b'8';
77    t[0x0A] = b'9'; t[0x0B] = b'0'; t[0x0C] = b'-'; t[0x0D] = b'=';
78    // 0x0E: Backspace, 0x0F: Tab
79    t[0x0E] = 0x08; t[0x0F] = b'\t';
80    // QWERTY 상단
81    t[0x10] = b'q'; t[0x11] = b'w'; t[0x12] = b'e'; t[0x13] = b'r';
82    t[0x14] = b't'; t[0x15] = b'y'; t[0x16] = b'u'; t[0x17] = b'i';
83    t[0x18] = b'o'; t[0x19] = b'p'; t[0x1A] = b'['; t[0x1B] = b']';
84    // 0x1C: Enter, 0x1D: LCtrl
85    t[0x1C] = b'\n';
86    // QWERTY 중단
87    t[0x1E] = b'a'; t[0x1F] = b's'; t[0x20] = b'd'; t[0x21] = b'f';
88    t[0x22] = b'g'; t[0x23] = b'h'; t[0x24] = b'j'; t[0x25] = b'k';
89    t[0x26] = b'l'; t[0x27] = b';'; t[0x28] = b'\'';
90    t[0x29] = b'`'; t[0x2B] = b'\\';
91    // 0x2A: L-Shift, 0x36: R-Shift (별도 처리)
92    // QWERTY 하단
93    t[0x2C] = b'z'; t[0x2D] = b'x'; t[0x2E] = b'c'; t[0x2F] = b'v';
94    t[0x30] = b'b'; t[0x31] = b'n'; t[0x32] = b'm';
95    t[0x33] = b','; t[0x34] = b'.'; t[0x35] = b'/';
96    // 0x39: Space
97    t[0x39] = b' ';
98    // 숫자패드 *
99    t[0x37] = b'*';
100    t
101};
102
103const UPPER: [u8; 128] = {
104    let mut t = [0u8; 128];
105    // Shift + 숫자 행 → 특수문자
106    t[0x02] = b'!'; t[0x03] = b'@'; t[0x04] = b'#'; t[0x05] = b'$';
107    t[0x06] = b'%'; t[0x07] = b'^'; t[0x08] = b'&'; t[0x09] = b'*';
108    t[0x0A] = b'('; t[0x0B] = b')'; t[0x0C] = b'_'; t[0x0D] = b'+';
109    t[0x0E] = 0x08; t[0x0F] = b'\t';
110    // Shift + 알파벳 → 대문자
111    t[0x10] = b'Q'; t[0x11] = b'W'; t[0x12] = b'E'; t[0x13] = b'R';
112    t[0x14] = b'T'; t[0x15] = b'Y'; t[0x16] = b'U'; t[0x17] = b'I';
113    t[0x18] = b'O'; t[0x19] = b'P'; t[0x1A] = b'{'; t[0x1B] = b'}';
114    t[0x1C] = b'\n';
115    t[0x1E] = b'A'; t[0x1F] = b'S'; t[0x20] = b'D'; t[0x21] = b'F';
116    t[0x22] = b'G'; t[0x23] = b'H'; t[0x24] = b'J'; t[0x25] = b'K';
117    t[0x26] = b'L'; t[0x27] = b':'; t[0x28] = b'"';
118    t[0x29] = b'~'; t[0x2B] = b'|';
119    t[0x2C] = b'Z'; t[0x2D] = b'X'; t[0x2E] = b'C'; t[0x2F] = b'V';
120    t[0x30] = b'B'; t[0x31] = b'N'; t[0x32] = b'M';
121    t[0x33] = b'<'; t[0x34] = b'>'; t[0x35] = b'?';
122    t[0x39] = b' ';
123    t[0x37] = b'*';
124    t[0x01] = 0x1B;
125    t
126};

128/// IRQ1 핸들러에서 호출 — 스캔코드를 ASCII로 변환해 버퍼에 push
129pub fn handle_scancode(scancode: u8) {
130    let is_break = scancode >= 0x80;
131    let make     = (scancode & 0x7F) as usize;
132
133    // Shift: 0x2A(L), 0x36(R)
134    if make == 0x2A || make == 0x36 {
135        SHIFT.store(!is_break, Ordering::Relaxed);
136        return;
137    }
138    // Caps Lock 토글 (make only)
139    if !is_break && make == 0x3A {
140        let c = CAPS.load(Ordering::Relaxed);
141        CAPS.store(!c, Ordering::Relaxed);
142        return;
143    }
144
145    if is_break || make >= 128 { return; }
146
147    let shift = SHIFT.load(Ordering::Relaxed);
148    let caps  = CAPS.load(Ordering::Relaxed);
149
150    // 알파벳 키: Shift XOR Caps → 대문자
151    let is_alpha = matches!(make,
152        0x10..=0x19 | 0x1E..=0x26 | 0x2C..=0x32);
153    let use_upper = if is_alpha { shift ^ caps } else { shift };
154
155    let ascii = if use_upper { UPPER[make] } else { LOWER[make] };
156    if ascii != 0 {
157        push_key(ascii);
158    }
159}
